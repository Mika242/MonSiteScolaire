הֲלָכָה כְּרַבִּי יוֹסֵי בֶּן הַמְשׁוּלָּם לְמָה לִי?

אִי אָמַר הִלְכְתָא דְּכוּלֵּיהּ פִּירְקִין, וְלָא אָמַר הִלְכְתָא כְּרַבִּי יוֹסֵי בֶּן הַמְשׁוּלָּם, הֲוָה אָמֵינָא: אַדְּרַבִּי יוֹסֵי בֶּן הַמְשׁוּלָּם קָאֵי, וּמַאי ״בְּכוּלֵּיהּ פִּירְקִין״? דְּרַבִּי יוֹסֵי תַּרְתֵּי אָמַר, וּפְלוּגְתָּא דְּבָרַיְיתָא פְּלוּגְתָּא הִיא.

אַשְׁמְעִינַן הֲלָכָה כְּרַבִּי יוֹסֵי — לְאַשְׁמוֹעִינַן דְּאַדְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל קָאֵי, וּפְלוּגְתָּא דְּבָרַיְיתָא לָאו פְּלוּגְתָּא הִיא.

מַאי בָּרַיְיתָא? דְּתַנְיָא: הַלּוֹקֵחַ בְּהֵמָה מְנִיקָה מִן הַגּוֹי, הַבָּא אַחֲרָיו בְּכוֹר מִסָּפֵק, מִפְּנֵי שֶׁמְּרַחֶמֶת אַף עַל פִּי שֶׁאֵינָהּ יוֹלֶדֶת. רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל אוֹמֵר: דָּבָר בְּחֶזְקָתוֹ.

וְכֵן הָיָה רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל אוֹמֵר: הַנִּכְנָס לְתוֹךְ עֶדְרוֹ בַּלַּיְלָה וְרָאָה כְּעֶשֶׂר כַּחֲמֵשׁ עֶשְׂרֵה, מַבְכִּירוֹת וְשֶׁאֵינָן מַבְכִּירוֹת, יוֹלְדוֹת, לְמָחָר הִשְׁכִּים וּמָצָא זְכָרִים (תְּלוּיוֹת) [תְּלוּיִין] בְּשֶׁאֵין מַבְכִּירוֹת, וּנְקֵבוֹת תְּלוּיוֹת בְּמַבְכִּירוֹת — אֵינוֹ חוֹשֵׁשׁ שֶׁמָּא בְּנָהּ שֶׁל זוֹ בָּא לוֹ אֵצֶל זוֹ.

אִיבַּעְיָא לְהוּ: דָּבָר בְּחֶזְקָתוֹ דְּקָאָמַר רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל, דְּאֵינָהּ מְרַחֶמֶת אֶלָּא אִם כֵּן יוֹלֶדֶת — הָא הֵיכָא דְּאוֹלִיד חָיְישִׁינַן לְרַחוֹמֵי, אוֹ דִלְמָא: דִּידַהּ מְרַחֲמָא, דְּלָא דִּידַהּ לָא מְרַחֲמָא לֵיהּ?

לְמַאי נָפְקָא מִינַּהּ? לְמִלְקֵא עֲלַהּ מִשּׁוּם ״אוֹתוֹ וְאֶת בְּנוֹ״. אִי אָמְרַתְּ דִּידַהּ מְרַחֲמָא, דְּלָא דִּידַהּ לָא מְרַחֲמָא — לָקֵי, וְאִי אָמְרַתְּ דְּלָא דִּידַהּ נָמֵי מְרַחֲמָא — לָא לָקֵי.

מַאי? תָּא שְׁמַע, רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל אוֹמֵר: הַלּוֹקֵחַ בְּהֵמָה מִן הַגּוֹי — אֵינוֹ חוֹשֵׁשׁ שֶׁמָּא בְּנָהּ שֶׁל אַחֶרֶת הָיָה. מִי קָתָנֵי ״הוּא״? ״הָיָה״ קָתָנֵי! הָכִי קָאָמַר: אֵינוֹ חוֹשֵׁשׁ שֶׁמָּא בְּנָהּ שֶׁל אַחֶרֶת, אֶלָּא אִם כֵּן הָיָה לָהּ וָלָד מֵעִיקָּרָא.

תָּא שְׁמַע: נִכְנַס בְּתוֹךְ עֶדְרוֹ וְרָאָה אֶת הַמַּבְכִּירוֹת מְנִיקוֹת, וְאֶת שֶׁאֵינָן מַבְכִּירוֹת מְנִיקוֹת — אֵינוֹ חוֹשֵׁשׁ שֶׁמָּא בְּנָהּ שֶׁל זוֹ בָּא לוֹ אֵצֶל זוֹ, אוֹ שֶׁמָּא בְּנָהּ שֶׁל זוֹ בָּא לוֹ אֵצֶל זוֹ.

אַמַּאי? לֵיחוּשׁ דִּלְמָא רַחוֹמֵי רַחֵים! בִּמְקוֹם דִּידַהּ, לָא שָׁבְקָא דִּידַהּ וּמְרַחֲמָא דְּלָא דִּידַהּ.

תָּא שְׁמַע: דָּבָר בְּחֶזְקָתוֹ כּוּ׳. מַאי לָאו רֵישָׁא דּוּמְיָא דְּסֵיפָא? מָה סֵיפָא וַדַּאי דִּידַהּ, אַף רֵישָׁא וַדַּאי דִּידַהּ.

מִידֵּי אִירְיָא? הַאי כִּדְאִיתֵאּ, וְהַאי כִּדְאִיתֵאּ. וּמַאי ״וְכֵן״? אַפְּטוּרָה דִּבְכוֹרָה.

אָמַר רַבָּה בַּר בַּר חָנָה אָמַר רַבִּי יוֹחָנָן: רָאָה חֲזִיר שֶׁכָּרוּךְ אַחַר רָחֵל, פְּטוּרָה מִן הַבְּכוֹרָה, וְאָסוּר בַּאֲכִילָה ״עַד יָבֹא וְיוֹרֶה צֶדֶק לָכֶם״.

פָּטוּר מִן הַבְּכוֹרָה — כְּמַאן? כְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל, וְאָסוּר בַּאֲכִילָה — כְּרַבָּנַן.

וְתוּ, אִי כְּרַבָּנַן, מַאי ״עַד יָבֹא וְיוֹרֶה צֶדֶק לָכֶם״? ״עַד שֶׁיִּוָּדַע לְךָ דָּבָר״ מִיבְּעֵי לֵיהּ!

וְכִי תֵּימָא: מְסַפְּקָא לֵיהּ אִי הֲלָכָה כְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל אִי כְּרַבָּנַן, אִי מְסַפְּקָא לֵיהּ — אַמַּאי פְּטוּרָה מִן הַבְּכוֹרָה? וְתוּ: מִי מְסַפְּקָא לֵיהּ? וְהָאָמַר רַבָּה בַּר בַּר חָנָה אָמַר רַבִּי יוֹחָנָן: כׇּל מָקוֹם שֶׁשָּׁנָה רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל בְּמִשְׁנָתֵינוּ — הֲלָכָה כְּמוֹתוֹ, חוּץ מֵעָרֵב, וְצַיְדָן, וּרְאָיָה אַחֲרוֹנָה!

לְעוֹלָם פְּשִׁיטָא לֵיהּ הֲלָכָה כְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל, מִיהוּ מְסַפְּקָא לֵיהּ אִי סָבַר רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל יוֹלֶדֶת מְרַחֶמֶת, אוֹ אֵינָהּ מְרַחֶמֶת. אִי הָכִי, עַד דְּאַשְׁמְעִינַן חֲזִיר, לַישְׁמְעִינַן טָלֶה, וּלְמִילְקֵי עֲלֵיהּ מִשּׁוּם ״אוֹתוֹ וְאֶת בְּנוֹ״!

חֲזִיר אִיצְטְרִיכָא לֵיהּ, דְּאִי אַשְׁמְעִינַן טָלֶה, הֲוָה אָמֵינָא: אִם תִּמְצָא לוֹמַר דְּסָבַר יוֹלֶדֶת מְרַחֲמָא — דְּמִינַהּ, דְּלָא מִינַהּ — לָא. אַשְׁמְעִינַן חֲזִיר, דִּמְרַחֶמֶת, וַאֲפִילּוּ דְּלָאו מִינַהּ נָמֵי, דְּאִיכָּא לְמֵימַר דִּלְמָא מְרַחֲמָא.

וְהַיְינוּ דְּאָמַר רַבִּי יוֹחָנָן: בָּעֵי

אַחַאי בְּרִיבִּי: רָאָה חֲזִיר שֶׁכָּרוּךְ אַחַר רָחֵל, מַהוּ? מַאי קָמִיבַּעְיָא לֵיהּ? אִילֵּימָא לִבְכוֹרָה, וְקָמִיבַּעְיָא לֵיהּ אִי הִלְכְתָא כְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל אִי הִלְכְתָא כְּרַבָּנַן, תִּיבְּעֵי לֵיהּ טָלֶה!

כִּי קָמִיבַּעְיָא לֵיהּ, לִבְכוֹרָה וְאַלִּיבָּא דְּרַבָּנַן, לַאֲכִילָה וְאַלִּיבָּא דְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל.

לִבְכוֹרָה, וַאֲפִילּוּ לְרַבָּנַן דְאָמְרִי מְרַחֲמָא, כִּי מְרַחֲמָא דְּמִינַהּ, דְּלָא מִינַהּ — לָא, אוֹ דִלְמָא דְּלָא מִינַהּ נָמֵי מְרַחֲמָא?

לַאֲכִילָה, וַאֲפִילּוּ לְרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל — אִם תִּמְצָא לוֹמַר סָבַר: יוֹלֶדֶת מְרַחֲמָא דְּמִינַהּ, דְּלָאו מִינַהּ — לָא; אוֹ דִלְמָא: דְּלָאו מִינַהּ נָמֵי מְרַחֲמָא? תֵּיקוּ.

מַתְנִי׳ רַבִּי יוֹסֵי בֶּן הַמְשׁוּלָּם אוֹמֵר: הַשּׁוֹחֵט אֶת הַבְּכוֹר, עוֹשֶׂה מָקוֹם בְּקוֹפִיץ מִיכָּן וּמִיכָּן, וְתוֹלֵשׁ אֶת הַשֵּׂעָר, וּבִלְבַד שֶׁלֹּא יְזִיזֶנּוּ מִמְּקוֹמוֹ, וְכֵן תּוֹלֵשׁ אֶת הַשֵּׂעָר לִרְאוֹת מְקוֹם מוּם.

גְּמָ׳ אָמַר רַב: הֲלָכָה כְּרַבִּי יוֹסֵי בֶּן הַמְשׁוּלָּם. בְּעוֹ מִינֵּיהּ מֵרַב הוּנָא: כְּנֶגְדּוֹ בְּיוֹם טוֹב מַהוּ? טַעְמָא דְּרַבִּי יוֹסֵי בֶּן הַמְשׁוּלָּם מִשּׁוּם דְּקָסָבַר תּוֹלֵשׁ לָאו הַיְינוּ גּוֹזֵז, וּבְיוֹם טוֹב אָסוּר, דְּהָוֵה לֵיהּ עוֹקֵר דָּבָר מִגִּידּוּלוֹ.

אוֹ דִלְמָא בְּעָלְמָא סָבַר רַבִּי יוֹסֵי בֶּן הַמְשׁוּלָּם תּוֹלֵשׁ הַיְינוּ גּוֹזֵז, וְהַיְינוּ טַעְמָא דִּשְׁרֵי, דְּהָוֵה לֵיהּ דָּבָר שֶׁאֵין מִתְכַּוֵּין, וּבְיוֹם טוֹב דָּבָר שֶׁאֵין מִתְכַּוֵּין מוּתָּר?

אֲמַר לְהוּ: זִילוּ שַׁיְילוּהּ לְרַב חֲנַנְאֵל, אִי אָמַר הֲלָכָה כְּרַבִּי יוֹסֵי בֶּן הַמְשׁוּלָּם, אֶיפְשׁוֹט לְכוּ. אֲתוֹ שַׁיְילוּהּ, אֲמַר לְהוּ: הָכִי אָמַר רַב: הֲלָכָה כְּרַבִּי יוֹסֵי בֶּן הַמְשׁוּלָּם. אֲתוֹ לְקַמֵּיהּ דְּרַב הוּנָא, אֲמַר לְהוּ: כְּנֶגְדּוֹ בְּיוֹם טוֹב מוּתָּר. אִיתְּמַר נָמֵי, אָמַר רַב חֲנַנְיָה בַּר שֶׁלֶמְיָא מִשְּׁמֵיהּ דְּרַב: כְּנֶגְדּוֹ בְּיוֹם טוֹב

… pourquoi ai-je besoin qu'il dise que la halakha est comme Rabbi Yossi ben HaMechoulam ?

La Guemara explique : si Rav avait dit seulement que la halakha est comme la Michna anonyme dans tout le chapitre, sans dire explicitement que la halakha est comme Rabbi Yossi ben HaMechoulam, j'aurais dit : Rav se réfère précisément à l'avis de Rabbi Yossi ben HaMechoulam. Et que signifie l'expression : dans tout le chapitre, qui indique plus d'un cas ? Cela vient de ce que Rabbi Yossi a dit deux choses distinctes dans le chapitre. Et cela voudrait dire, à propos de la règle de Rav, que si une Michna est contredite par une opinion divergente citée dans une baraïta, c'est considéré comme une controverse.

Rav a donc enseigné que la halakha est comme Rabbi Yossi pour nous apprendre que sa déclaration sur tout le chapitre se réfère à l'avis de Rabban Chimon ben Gamliel, et qu'une opinion divergente citée dans une baraïta n'est pas considérée comme une controverse.

Quelle est la baraïta dont on a parlé ? Comme on l'enseigne dans une baraïta : celui qui achète à un non-Juif une bête allaitante : le petit qui naît après celui qu'elle allaite est traité comme un premier-né par doute, car une bête peut avoir pitié et allaiter un petit même si elle n'a pas mis bas du tout. Rabban Chimon ben Gamliel dit : la chose reste dans sa présomption, c'est-à-dire que la bête allaitante est présumée être la mère du petit qu'elle allaite.

La baraïta poursuit : et de même, Rabban Chimon ben Gamliel disait : celui qui entre au milieu de son troupeau la nuit et voit une dizaine ou une quinzaine de bêtes, dont certaines ont un petit qui est un premier-né et d'autres un petit qui n'est pas un premier-né, toutes ayant mis bas, et qui se lève le lendemain et trouve les petits mâles en train de téter auprès des mères dont le petit n'est pas un premier-né, et les petites femelles auprès des mères dont le petit est un premier-né, n'a pas à craindre que le petit de celle-ci soit venu auprès de celle-là. Il peut supposer que chaque bête allaite son propre petit.

On a posé une question aux Sages : quel est le sens exact de l'expression « la chose reste dans sa présomption », qu'a dite Rabban Chimon ben Gamliel ? Cela veut-il dire qu'une bête n'a pitié que si elle a mis bas, et alors, là où elle a mis bas, nous craignons qu'elle ait pitié d'un petit qui n'est pas le sien ? Ou bien qu'une bête a pitié de son propre petit, mais n'a pas pitié de celui qui n'est pas le sien ?

La Guemara demande : quelle différence cela fait-il en pratique ? Dans les deux cas, la bête allaitante est présumée avoir mis bas, et son futur petit ne sera pas un premier-né. La Guemara répond : la différence porte sur le fait d'être flagellé pour avoir abattu la bête à cause de l'interdit d'abattre une bête et son petit le même jour. Si tu dis qu'une bête a pitié seulement de son propre petit et n'a pas pitié d'un petit qui n'est pas le sien, on est flagellé si l'on abat volontairement la bête et le petit qu'elle allaite. Mais si tu dis qu'elle a aussi pitié du petit d'une autre, on n'est pas flagellé dans ce cas.

La Guemara revient à sa question de départ. Quel est le sens des paroles de Rabban Chimon ben Gamliel ? Viens et écoute une preuve de la Michna : Rabban Chimon ben Gamliel dit : celui qui achète une bête à un non-Juif n'a pas à craindre que le petit allaité ait été le petit d'une autre. Apparemment, Rabban Chimon ben Gamliel pense qu'une bête n'allaite que son propre petit. La Guemara rejette cette preuve : la Michna enseigne-t-elle qu'on ne craint pas que ce soit le petit d'une autre ? Non, elle enseigne qu'on ne craint pas que ce fût le petit d'une autre, et voici ce que la Michna dit : de toute façon, le prochain petit n'est pas un premier-né, car on n'a pas à craindre que la bête allaite un petit d'une autre, sauf si elle avait d'abord un petit à elle.

La Guemara propose une autre preuve de la suite de la Michna : celui qui entre au milieu de son troupeau et voit des bêtes qui ont mis bas pour la première fois qui allaitent, et des bêtes qui n'ont pas mis bas pour la première fois qui allaitent aussi, n'a pas à craindre que le petit de celle-ci soit venu auprès de celle-là, ou que le petit de celle-là soit venu auprès de celle-ci.

La Guemara explique la preuve : mais pourquoi pas ? Craignons-nous que la bête ait peut-être pitié d'un petit qui n'est pas le sien. Apparemment, Rabban Chimon ben Gamliel pense qu'une bête n'allaite que son propre petit, même si elle a déjà mis bas. La Guemara rejette cette preuve : même si une bête a pitié d'un petit qui n'est pas le sien, là où elle a son propre petit, elle ne laisse pas de côté le sien pour avoir pitié de celui qui n'est pas le sien.

La Guemara suggère encore : viens et écoute une preuve de la baraïta : la chose reste dans sa présomption. D'après les mots « et de même », la Guemara déduit : n'est-ce pas que la première partie est semblable à la dernière ? De même que dans la dernière partie le petit qui tète est certainement celui de la bête qui l'allaite, de même dans la première partie il est certainement le sien, bien que la mère ait pu mettre bas auparavant ?

La Guemara rejette cette suggestion : les cas sont-ils comparables ? Celui-ci est tel qu'il est, et celui-là tel qu'il est, c'est-à-dire qu'ils concernent des cas différents. Dans la deuxième partie, où tous les petits se tiennent devant leurs mères, il est certain que chaque mère n'allaite que son petit. À l'inverse, dans la première partie, on ne sait pas si ce petit est le sien. Mais alors, que signifie « et de même » ? Cela veut dire que dans les deux parties la bête allaitante est exemptée du premier-né, car même si ce petit n'est pas à elle, elle ne l'allaiterait pas si elle n'avait pas déjà mis bas.

La Guemara analyse un cas semblable. Rabba bar bar 'Hana dit que Rabbi Yo'hanan dit : si quelqu'un a vu un petit ressemblant à un porc qui suivait une brebis, la brebis est exemptée du premier-né. Et le « porc » est interdit à la consommation « jusqu'à ce qu'il vienne et vous enseigne la justice » (Hochéa 10:12), c'est-à-dire jusqu'à la venue d'Éliyahou le prophète qui tranchera la halakha.

La Guemara soulève une difficulté : d'un côté, la brebis est exemptée du premier-né, ce qui correspond à l'avis de Rabban Chimon ben Gamliel, pour qui une bête qui allaite un petit est présumée en être la mère. Mais d'un autre côté, le « porc » est interdit à la consommation, ce qui correspond à l'avis des Sages, selon lesquels une bête peut allaiter le petit d'une autre.

Et de plus, si la halakha est comme les Sages et qu'on a un doute sur l'identité de la mère, pourquoi citer le verset : « jusqu'à ce qu'il vienne et vous enseigne la justice » ? Ce verset indique que le statut du « porc » est un doute halakhique que seul Éliyahou le prophète peut résoudre. Il aurait fallu dire : jusqu'à ce que la chose soit connue de vous, pour un doute de fait sur la question de savoir si le « porc » est bien le petit de la brebis.

Et si tu dis qu'on ne sait pas si la halakha est comme Rabban Chimon ben Gamliel ou comme les Sages, ce n'est pas possible : si c'est douteux, pourquoi la brebis est-elle exemptée du premier-né ? Et de plus, est-ce un doute ? Rabba bar bar 'Hana ne dit-il pas que Rabbi Yo'hanan dit : partout où Rabban Chimon ben Gamliel a enseigné dans notre Michna, la halakha est comme lui, sauf pour trois cas : la garantie (Bava Batra 174a), l'incident de Tsaïdan (Guitin 74a) et la preuve dans la dernière controverse (Sanhédrin 31a) ?

La Guemara répond : en réalité, il est évident pour Rabba bar bar 'Hana que la halakha est comme Rabban Chimon ben Gamliel. Mais il est douteux que Rabban Chimon ben Gamliel pense qu'une bête qui a mis bas peut avoir pitié d'un petit qui n'est pas le sien, ou qu'elle n'a pas pitié du petit d'une autre. La Guemara rejette cette explication : s'il en est ainsi, au lieu d'enseigner le cas d'un animal ayant l'apparence d'un porc, il aurait dû enseigner le cas, plus courant, d'une brebis allaitant un agneau. Et au lieu de s'interroger sur l'interdit de consommer un animal impur, il aurait dû se demander si on est flagellé à cause de l'interdit d'abattre une bête et son petit le même jour.

La Guemara explique : il était nécessaire que Rabba bar bar 'Hana enseigne le cas du porc, car, s'il avait enseigné le cas d'un agneau, j'aurais dit : même si tu dis que, selon Rabban Chimon ben Gamliel, une bête qui a mis bas peut avoir pitié d'un petit d'une autre, cela ne vaut que pour son espèce ; mais pour un petit qui n'est pas de son espèce, elle n'a pas pitié. Rabba bar bar 'Hana nous enseigne donc le cas du porc pour montrer qu'une bête peut avoir pitié, et même d'un petit qui n'est pas de son espèce, car on peut dire que peut-être elle a eu pitié.

Et c'est ce que Rabba bar bar 'Hana dit au nom de Rabbi Yo'hanan : une question a été posée par A'haï le Distingué [beribbi] : si quelqu'un voit un animal ressemblant à un porc qui suit une brebis, quelle est la halakha ? La Guemara demande : quelle question pose-t-il ? Si nous disons qu'il s'interroge sur le premier-né du futur petit, et qu'il demande donc si la halakha est comme Rabban Chimon ben Gamliel ou comme les Sages, alors qu'on pose la question sur le cas courant d'un agneau, plutôt que sur le cas inhabituel d'un animal ressemblant à un porc.

La Guemara répond : en réalité, quand A'haï le Distingué a posé sa question, c'était à propos des conséquences de chaque avis : il s'interrogeait sur le premier-né, en supposant que la halakha est comme les Sages. Il s'interrogeait aussi sur la consommation, en supposant que la halakha est comme Rabban Chimon ben Gamliel.

La Guemara développe : il s'interrogeait sur le premier-né, même selon les Sages qui disent qu'une bête peut avoir pitié d'un autre petit. Peut-être que la pitié ne vaut que pour un petit de son espèce, mais que pour un petit qui n'est pas de son espèce, elle n'a certainement pas pitié, et le fait qu'elle l'allaite prouve que c'est le sien. Ou peut-être qu'elle a aussi pitié d'un petit qui n'est pas de son espèce. Si elle n'a pitié que de son espèce, la brebis est exemptée du premier-né ; si sa pitié s'étend à d'autres espèces, elle n'est pas exemptée.

A'haï le Distingué s'interrogeait aussi sur la consommation, même selon Rabban Chimon ben Gamliel : si tu dis qu'il pense qu'une bête qui a mis bas peut avoir pitié du petit d'une autre, peut-être que cela ne vaut que pour un petit de son espèce, mais pas pour un petit qui n'est pas de son espèce, ce qui signifie que le « porc » est licite. Ou peut-être qu'elle a aussi pitié d'un petit qui n'est pas de son espèce, auquel cas le « porc » peut être interdit et ne doit pas être mangé. Après avoir expliqué la question d'A'haï le Distingué, la Guemara conclut : qu'elle reste sans réponse.

GUEMARA : Rav dit : la halakha est comme Rabbi Yossi ben HaMechoulam. Les Sages posèrent une question à Rav Houna : quelle est la halakha dans le cas parallèle de celui qui veut abattre un animal profane un jour de fête ? La réponse dépend de la raison de l'avis de Rabbi Yossi ben HaMechoulam : permet-il de dégager le poil au couperet parce qu'il pense qu'arracher n'est pas tondre, et que, contrairement au poil du premier-né, il est interdit d'arracher du poil un jour de fête, car c'est déraciner quelque chose de sa croissance ?

Ou peut-être que, en général, Rabbi Yossi ben HaMechoulam pense qu'arracher, c'est tondre. Et la raison pour laquelle c'est permis de dégager avec un couperet dans le cas du premier-né est qu'il s'agit d'un acte non intentionnel, c'est-à-dire d'un acte fait dans un but permis et dont une conséquence interdite peut résulter par inadvertance, puisqu'il n'a pas l'intention d'arracher le poil ; et un jour de fête, un acte non intentionnel est permis ?

Rav Houna leur dit : allez demander à un disciple de Rav, Rav 'Hananel, et si Rav a bien dit que la halakha est comme Rabbi Yossi ben HaMechoulam, je vous expliquerai la solution. Ils allèrent et demandèrent à Rav 'Hananel, qui leur dit : voici ce que dit Rav : la halakha est comme Rabbi Yossi ben HaMechoulam. Ils vinrent chez Rav Houna, qui leur dit : dans le cas parallèle d'un animal profane que l'on veut abattre un jour de fête, il est permis d'arracher le poil pour permettre l'abattage correct. On a aussi dit que Rav 'Hanania bar Chelamia dit au nom de Rav : dans le cas parallèle d'un animal profane que l'on veut abattre un jour de fête, … (la phrase se poursuit au daf suivant)

why do I need him to state that the halakha is in accordance with the opinion of Rabbi Yosei ben HaMeshullam?

The Gemara explains: Had Rav said only that the halakha is in accordance with the opinion of the anonymous tanna in the mishnayot throughout the whole chapter, but did not say explicitly that the halakha is in accordance with the opinion of Rabbi Yosei ben HaMeshullam, I would say: Rav is referring specifically to the opinion of Rabbi Yosei ben HaMeshullam. And what is the meaning of the phrase: Throughout the whole chapter, which indicates he is speaking of more than one case? This is referring to the fact that Rabbi Yosei said two separate statements in the chapter. And this would indicate with regard to Rav’s statement that if a mishna is contradicted by a dissenting opinion cited in a baraita, that is considered a dispute.

Rav taught that the halakha is in accordance with the opinion of Rabbi Yosei in order to teach us that his statement with regard to the whole chapter is referring to the opinion of Rabban Shimon ben Gamliel, and with regard to Rav’s statement, if a mishna is contradicted by a dissenting opinion cited in a baraita, that is not considered a dispute.

What is the baraita mentioned previously? As it is taught in a baraita: In the case of one who purchases a nursing female animal from a gentile, the young that is born after the offspring currently being nursed is treated as a firstborn due to the uncertainty whether it is the animal’s offspring, because an animal might be merciful and nurse another animal even though it has not given birth at all. Rabban Shimon ben Gamliel says: The matter stands in its presumptive status, i.e., the nursing animal is assumed to be the mother of the nursing offspring.

The baraita continues: And similarly, Rabban Shimon ben Gamliel would say: One who enters amid his flock at night and sees about ten or about fifteen animals, some of whose offspring is a firstborn and others whose offspring is not a firstborn, and all were giving birth, and on the next day he arose and found that the male young were nursing from the mothers whose offspring is not a firstborn, and the female young were nursing from the mothers whose offspring is a firstborn, he does not need to be concerned that perhaps the young of this mother came to nurse from that mother. Rather, he may assume that each animal is nursing its own offspring.

A dilemma was raised before the Sages: What is the precise meaning of the phrase: The matter stands in its presumptive status, which Rabban Shimon ben Gamliel stated? Does this mean that an animal does not show mercy unless it has given birth, and consequently, in a case where the animal has given birth we are concerned for the possibility that it might be merciful and nurse an offspring that is not its own? Or perhaps it means that an animal might be merciful only toward its own offspring, but it does not show mercy to an animal that is not its own?

The Gemara asks: What difference is there in practice based on this distinction? Either way the nursing animal is assumed to have given birth and its future offspring will not be considered a firstborn. The Gemara answers that there is a difference with regard to whether or not one is flogged for slaughtering it due to violating the prohibition against slaughtering a mother and its offspring on the same day. If you say an animal is merciful only toward its own offspring but is not merciful toward an animal that is not its own, then one is flogged if he deliberately slaughters an animal and the offspring that it is nursing. But if you say it is also merciful toward an animal that is not its own, then one is not flogged in such a case.

The Gemara returns to its initial inquiry. What is the precise meaning of Rabban Shimon ben Gamliel’s statement? Come and hear a proof from the mishna, as Rabban Shimon ben Gamliel says: In the case of one who purchases a nursing female animal from a gentile, he does not need to be concerned lest the offspring currently being nursed was the offspring of another animal. Evidently, Rabban Shimon ben Gamliel holds that an animal nurses only its own offspring. The Gemara rejects this proof: Does the mishna teach that one does not need to be concerned that perhaps it is the offspring of another animal? No, it teaches that one does not need to be concerned that perhaps it was the offspring of another animal, and this is what it is saying: Whichever way you look at it, the next offspring is not a firstborn, as one does not need to be concerned that perhaps it was the offspring of another animal that it is nursing, as it would not nurse unless it already had an offspring initially.

The Gemara suggests: Come and hear a proof from the continuation of the mishna: In the case of one who enters amid his flock and sees mother animals that gave birth for the first time that were nursing, and also sees mother animals that gave birth not for the first time that were also nursing, he does not need to be concerned that perhaps the offspring of this animal came to that animal to be nursed, or that perhaps the offspring of that animal came to this animal to be nursed.

The Gemara explains the proof: But why not? Let us be concerned that perhaps the nursing animal is merciful toward an offspring that is not its own. Evidently, Rabban Shimon ben Gamliel holds that an animal nurses only its own offspring, even if it had previously given birth. The Gemara rejects this proof: Even if an animal does show mercy toward offspring that is not its own, in a place where it has its own offspring, it does not set aside its own offspring and show mercy toward offspring that is not its own.

The Gemara further suggests: Come and hear a proof from Rabban Shimon ben Gamliel’s statement in the baraita: If one purchases a nursing animal from a gentile, the matter stands in its presumptive status. From the phrase: And similarly, in the continuation of the baraita, the Gemara infers: What, is it not so that the first clause is similar to the latter clause? Just as in the latter clause the nursing young is certainly considered to be the nursing animal’s own offspring, so too, in the first clause it is certainly considered to be the nursing animal’s own offspring, despite the fact that the animal mother might have given birth previously.

The Gemara rejects this suggestion: Are the cases comparable? This first clause of the baraita stands as it is and that second clause of the baraita stands as it is, i.e., they are referring to different cases. In the second clause, where all the offspring are standing before their mothers, it is certain that each mother will nurse only its own child. Conversely, in the first clause it is uncertain whether that child is its own. But if so, what is the meaning of the phrase: And similarly? It means both clauses state that the nursing animal is exempt from having its future offspring counted a firstborn, since even if that offspring is not the mother’s, the mother would not nurse it unless it had previously given birth.

The Gemara analyzes a similar case. Rabba bar bar Ḥana says that Rabbi Yoḥanan says: If one saw a young animal resembling a pig that was trailing after a ewe, the ewe is exempt from having its future offspring counted a firstborn. And the pig-like animal is forbidden for consumption “until he comes and instructs righteousness to you” (Hosea 10:12), i.e., until Elijah the prophet arrives and determines the halakha.

The Gemara raises a difficulty with regard to Rabba bar bar Ḥana’s statement: On the one hand, the ewe is exempt from having its future offspring counted a firstborn, which is in accordance with the opinion of Rabban Shimon ben Gamliel that an animal that nurses from another is presumed to be its offspring. But on the other hand, the pig-like animal is forbidden for consumption, which is in accordance with the opinion of the Rabbis that an animal might nurse the young of another.

And furthermore, if the halakha is in accordance with the opinion of the Rabbis, and there is a concern with regard to the identity of the mother of the offspring, what is the reason for citing the verse: “Until he comes and instructs righteousness to you”? By citing it, the baraita is indicating that the status of the pig-like animal is a halakhic uncertainty that can be resolved only by Elijah the prophet. Instead, it should have stated: Until the matter becomes known to you, in reference to the empirical uncertainty as to whether the pig-like animal is the offspring of the ewe.

And if you would say that it is uncertain whether the halakha is in accordance with the opinion of Rabban Shimon ben Gamliel or in accordance with the opinion of the Rabbis, that cannot be correct: If that matter is uncertain, then why is the ewe exempt from having its future offspring counted a firstborn? And furthermore, is this question of whose opinion is accepted as the halakha an uncertainty at all? But doesn’t Rabba bar bar Ḥana say that Rabbi Yoḥanan says: Wherever Rabban Shimon ben Gamliel taught a halakha in the corpus of our Mishna the halakha is in accordance with his opinion, except for the following three cases: The responsibility of the guarantor (see Bava Batra 174a), and the incident that occurred in the city of Tzaidan (see Gittin 74a), and the dispute with regard to evidence in the final disagreement (see Sanhedrin 31a)?

The Gemara answers: Actually, it is obvious to Rabba bar bar Ḥana that the halakha is in accordance with the opinion of Rabban Shimon ben Gamliel. But it is uncertain if Rabban Shimon ben Gamliel holds that an animal that has given birth might be merciful toward offspring that is not its own, or if he holds that it is not merciful toward the offspring of another. The Gemara rejects this claim: If that is so, instead of teaching a case involving an animal with the appearance of a pig, he should have taught the more common case of a mother ewe nursing a baby lamb. And accordingly, rather than inquiring with regard to the prohibition against eating a non-kosher animal, he should have inquired as to whether or not one is flogged for it due to violating the prohibition against slaughtering a mother and its offspring on the same day.

The Gemara explains: It was necessary for Rabba bar bar Ḥana to teach the case of a pig, as, had he taught a case involving a lamb, I would say: Even if you say that Rabban Shimon ben Gamliel holds that an animal that has given birth might be merciful toward the offspring of another, this is so only with regard to its own species, but with regard to an animal that is not of its own species, it would not show mercy and would not nurse the offspring. Therefore, Rabba bar bar Ḥana teaches us the case of a pig, to convey that an animal might be merciful, and this is also true even where the offspring is not of its own species, as it could be said that perhaps the animal was merciful in such a case as well.

And this is the meaning of that which Rabba bar bar Ḥana said that Rabbi Yoḥanan said: A dilemma was raised

by Aḥai the Distinguished [beribbi]: If one saw an animal with the appearance of a pig that was trailing after a ewe, what is the halakha? The Gemara asks: What is the dilemma he is raising? If we say he is inquiring with regard to the firstborn status of the future offspring, and he is therefore essentially asking if the halakha is in accordance with the opinion of Rabban Shimon ben Gamliel or if the halakha is in accordance with the opinion of the Rabbis, then let the dilemma be raised with respect to the more common case of a lamb, rather than the irregular case of a pig-like animal.

The Gemara answers: Actually, when Aḥai the Distinguished raised the dilemma it was with respect to the ramifications of each opinion: He inquired with regard to the future offspring’s firstborn status, and this aspect of the dilemma was raised under the assumption that the halakha is in accordance with the opinion of the Rabbis. He also inquired with regard to whether or not the pig-like animal is permitted for consumption, and this aspect of the dilemma was raised under the assumption that the halakha is in accordance with the opinion of Rabban Shimon ben Gamliel.

The Gemara elaborates: He inquired with regard to the offspring’s firstborn status, and this question applies even according to the opinion of the Rabbis, who say that an animal might be merciful toward other young animals. The Gemara explains the dilemma: One can say that when an animal is merciful this applies only toward an animal that is of its own species, but to a young animal that is not of its own species a mother animal certainly does not show mercy. Therefore, the fact that it nursed proves that it is its offspring. Or perhaps an animal might also be merciful toward an offspring that is not of its own species. If it is merciful only to its own species, the ewe is exempt from having its future offspring counted a firstborn; if its mercy extends to other species then it is not exempt.

And Aḥai the Distinguished also inquired with regard to whether the pig-like animal is permitted for consumption, and this question applies even according to the opinion of Rabban Shimon ben Gamliel: If you say he holds that an animal that has given birth might be merciful and nurse another mother’s offspring, perhaps that is so only with regard to an animal that is of its own species, but to an animal that is not of its own species it is not merciful, which means the pig-like animal is kosher. Or perhaps a mother might also be merciful toward an offspring that is not of its own species, in which case the pig-like animal might not be kosher and therefore may not be eaten. After explaining the dilemma of Aḥai the Distinguished, the Gemara concludes: The dilemma shall stand unresolved.

MISHNA: Rabbi Yosei ben HaMeshullam says: Since it is prohibited by Torah law to shear a firstborn, as it states: “And you shall not shear the firstborn of your flock” (Deuteronomy 15:19), one who is slaughtering a firstborn, and must clear hair or wool from the area of the neck in order to facilitate proper slaughter, clears space by uprooting the hair with a cleaver [bekofitz] from here and from there, on either side of the neck, although he thereby plucks out the hair. He may clear space in this manner provided that he does not move the plucked hair from its place; it must remain intermingled with the rest of the hair so it will appear that he did not shear the animal. And likewise, one plucks the hair to enable one of the Sages to examine the place of a blemish and thereby determine whether it is permitted to slaughter the firstborn outside the Temple.

Enfin, la Michna 3:3 (fiche séparée) ouvre la question de l'arrachage du poil d'un premier-né, que la Guemara étend à l'abattage d'un animal profane un jour de fête.

1. Pourquoi Rav précise-t-il Rabbi Yossi ben HaMechoulam ? La difficulté laissée ouverte au daf 23 : si Rav a dit que la halakha suit la Michna anonyme dans tout le chapitre, pourquoi aurait-il besoin de dire aussi qu'elle suit Rabbi Yossi ben HaMechoulam ? La Guemara raisonne par l'absurde :

1. Imaginons que Rav ait dit seulement « la halakha est comme la Michna anonyme dans tout le chapitre ».
2. L'expression « dans tout le chapitre » suggère plusieurs cas. Or Rabbi Yossi ben HaMechoulam a dit deux choses dans le chapitre (dégager au couperet, et arracher pour voir le défaut). On aurait donc cru que Rav parle de lui.
3. Dans ce cas, la règle exclurait Rabban Chimon ben Gamliel, et une opinion contraire dans une baraïta compterait comme une controverse.
4. En disant explicitement « comme Rabbi Yossi », Rav oriente sa règle générale vers Rabban Chimon ben Gamliel, et nous apprend qu'une opinion divergente rapportée dans une baraïta n'est pas une controverse.

2. La baraïta sur la bête allaitante. La baraïta que la Guemara cite pour parler de « controverse en baraïta » traite de celui qui achète à un non-Juif une bête allaitante. Selon l'avis anonyme, le petit suivant est un premier-né douteux, car une bête peut avoir « pitié » (merah'amet) et allaiter le petit d'une autre même sans avoir mis bas. Rabban Chimon ben Gamliel répond que la chose reste dans sa présomption : on suppose que la bête allaitante est la mère du petit. Il ajoute un cas de troupeau, la nuit, où les petits ont changé de place, et dit que même alors on ne craint pas d'échange. Les Tossefot soulignent que le cas est pris la nuit pour montrer que même quand les mères ne reconnaissent pas leurs petits et peuvent se tromper, on ne craint pas.

3. Que veut dire « sa présomption » ? La Guemara pose deux lectures possibles de l'avis de Rabban Chimon ben Gamliel :

LectureCe que cela impliquePour « une bête et son petit »
A. Une bête n'a pitié que si elle a mis bas ; quand elle a mis bas, on craint qu'elle ait pitié d'un autre petitUn petit qui tète auprès d'une bête qui a déjà mis bas peut ne pas être le sienPas de flagellation (doute)
B. Une bête n'a pitié que de son propre petitUn petit qui tète auprès d'elle est certainement le sienFlagellation si on les abat le même jour

Pour le premier-né, la différence est nulle : dans les deux lectures la bête est exemptée, puisqu'elle a au moins un autre petit (Rachi). La différence porte donc sur l'interdit d'abattre une bête et son petit le même jour (Vayikra 22:28), puni de flagellation si c'est volontaire.

4. Trois preuves, trois rejets. La Guemara tente de trancher entre les deux lectures en s'appuyant sur des textes. Chaque fois, elle rejette la preuve :

Preuve proposéePourquoi elle semble trancherPourquoi elle est rejetée
La Michna : on ne craint pas que ce fût le petit d'une autreUne bête n'allaite que son propre petitLa Michna dit « fût » : elle parle seulement du fait qu'elle a mis bas avant, pas de l'identité du petit qui tète
La fin de la Michna : on ne craint pas que le petit de celle-ci soit venu auprès de celle-làMême une bête qui a déjà mis bas n'allaite pas un autre petitLà où elle a son propre petit, elle ne le laisse pas pour un autre
Le « et de même » de la baraïta : la première partie ressemble à la dernièreDans les deux cas, le petit est certainement le sienLes deux cas sont différents ; « et de même » signifie seulement que dans les deux la bête est exemptée du premier-né

5. Le « porc » derrière la brebis. Rabba bar bar 'Hana rapporte, au nom de Rabbi Yo'hanan : si un petit ressemblant à un porc suit une brebis, la brebis est exemptée du premier-né et le « porc » est interdit à la consommation jusqu'à la venue d'Éliyahou. La Guemara relève trois difficultés :

1. L'exemption ressemble à l'avis de Rabban Chimon ben Gamliel, et l'interdit à l'avis des Sages : l'enseignement mélange deux avis opposés.
2. Si c'est comme les Sages, le doute est de fait (est-ce son petit ?), et il aurait fallu dire « jusqu'à ce que la chose soit connue », et non « jusqu'à ce qu'il enseigne la justice ».
3. On ne peut pas dire que la halakha entre Rabban Chimon ben Gamliel et les Sages est douteuse : Rabbi Yo'hanan établit que la halakha est comme Rabban Chimon ben Gamliel dans notre Michna, sauf trois cas (la garantie, l'incident de Tsaïdan, la preuve dans la dernière controverse).

La Guemara conclut que la halakha comme Rabban Chimon ben Gamliel est certaine pour Rabba bar bar 'Hana ; ce qui est douteux, c'est ce qu'il pense de la pitié d'une bête qui a mis bas. Le cas du porc est choisi pour une raison précise : avec un agneau, on aurait pu croire que la pitié d'une bête ne vaut que pour son espèce. Le porc montre qu'elle peut s'étendre à d'autres espèces.

6. La question d'A'haï le Distingué. Elle porte sur le même cas et comporte deux volets, sur le plan du premier-né et sur celui de la consommation :

VoletSelon quiLe doute
Premier-néLes Sages (qui craignent la pitié)La pitié s'étend-elle à une autre espèce ? Si non, le « porc » est le sien et la brebis est exemptée ; si oui, le petit suivant est un premier-né douteux
ConsommationRabban Chimon ben Gamliel (une bête qui a mis bas a pitié)La pitié s'étend-elle à une autre espèce ? Si non, le « porc » est licite ; si oui, il peut être interdit

La Guemara ne tranche pas : la question « reste » (teiku).

7. La Michna 3:3 et la question de Rav Houna. Rav dit que la halakha est comme Rabbi Yossi ben HaMechoulam. On pose alors une question à Rav Houna sur le cas parallèle d'un animal profane abattu un jour de fête (Yom Tov). La réponse dépend de la raison de Rabbi Yossi :

Raison possible de Rabbi YossiConséquence un jour de fête
Arracher n'est pas tondre (donc permis pour le premier-né)Interdit un jour de fête, car on déracine quelque chose de sa croissance
Arracher est tondre, mais c'est un acte non intentionnel (davar chéèn mitkavène)Permis un jour de fête, car un acte non intentionnel y est permis

Rav Houna ne répond pas tout de suite : il veut d'abord s'assurer que Rav a bien dit « la halakha est comme Rabbi Yossi ». Rav 'Hananel, disciple de Rav, le confirme. Rav Houna répond alors que c'est permis. Les Tossefot expliquent cette exigence : si Rav n'avait pas tranché comme Rabbi Yossi, on aurait interdit de dégager le poil pour abattre le premier-né, et de même on n'aurait pas permis de le faire un jour de fête pour abattre un animal profane. Le daf s'arrête sur un enseignement de Rav 'Hanania bar Chelamia au nom de Rav sur le même cas, qui se poursuit au daf 25.

Rachi explique d'abord que la phrase « pourquoi ai-je besoin que la halakha soit comme Rabbi Yossi ben HaMechoulam ? » vient de ce qu'on sait déjà, par la Michna anonyme, que la halakha est comme lui, puisque les Sages ne le contredisent pas. Pour la suite, il précise que « dans tout le chapitre » indique plus d'un cas parce que Rabbi Yossi a dit deux choses dans la Michna (faire de la place au couperet, et arracher pour voir le défaut), alors que Rav ne dit pas « sauf controverse » pour Rabban Chimon ben Gamliel, car sur lui il y a une controverse dans la baraïta. Tossefot objectent que Rabbi Yossi ben HaMechoulam lui-même est contredit d'une façon ou d'une autre, puisque Rav a eu besoin de trancher comme lui, et répondent qu'il n'y a peut-être de controverse ni dans la Michna ni dans la baraïta, mais que Rav savait que les Sages le contredisaient.

Sur la baraïta de la bête allaitante, Rachi définit merah'emet comme « aimer le petit de sa compagne ». Pour le troupeau, il explique que les petits mâles qui viennent de naître sont accrochés aux bêtes âgées, déjà exemptées du premier-né, et que les petites femelles sont accrochées aux bêtes qui ont mis bas pour la première fois. Il n'y a donc rien pour le cohen : aucun mâle n'est premier-né, et les femelles ne sont pas soumises à la loi du premier-né. Tossefot s'étonnent que la baraïta parle d'un troupeau « la nuit » et répondent que c'est pour montrer qu'on ne craint pas de mélange, même quand les mères se trompent plus facilement.

Pour le doute sur « sa présomption », Rachi explique pourquoi la différence porte sur la flagellation : pour le premier-né, la bête est dans tous les cas exemptée, puisque si ce petit n'est pas le sien, elle en avait un autre. Pour « une bête et son petit », le petit accroché à elle est le sien, d'où la flagellation, ou bien c'est un doute et on ne flagelle pas. Pour la preuve de la Michna, il précise que « fût » signifie que la Michna ne dit pas que la bête n'a pas pitié, mais qu'elle n'allaite un autre petit que si elle en a eu un autre. Pour « et de même », il explique que dans les deux parties la bête est exemptée du premier-né, mais que, pour « une bête et son petit », la première partie reste un doute.

Sur le « porc » derrière la brebis, Rachi explique que « jusqu'à ce qu'il vienne et enseigne la justice » signifie jusqu'à ce qu'Éliyahou vienne dire s'il est permis ou interdit. L'exemption correspond à Rabban Chimon : la bête n'allaite pas un petit qui n'est pas le sien, et ce « porc » lui-même n'est pas consacré, car c'est un petit déformé. L'interdit correspond aux Sages, qui craignent la pitié ; si c'était comme Rabban Chimon, ce serait un impur né d'une bête pure, et il serait permis (cf. daf 5b). Tossefot se demandent pourquoi l'exemple est un porc et non un chameau, et répondent que, comme l'exemple de la brebis est un petit bétail, le porc est cité, ou que l'exemple est choisi pour valoir selon tous les avis.

Sur la réponse que la halakha est comme Rabban Chimon, Rachi explique que Rabba bar bar 'Hana est certain de l'exemption, car si ce petit n'est pas à elle, la bête en avait un autre qu'elle a perdu, et c'est pour cela qu'elle allaite celui-ci. Il précise que, pour la consommation, il peut y avoir interdit, car il est possible qu'elle ait eu un autre petit, mort, et qu'elle ait pitié de celui-ci. Pour le « porc » dans la Guemara, il explique que l'on aurait cru la pitié limitée à l'espèce, d'où l'importance de cet exemple.

Sur la Michna 3:3 dans la Guemara, Rachi explique « pour le couperet » comme « pour une bête profane, faut-il faire de la place pour l'abattage ou non ? », et « n'est-ce pas tondre ? » comme le fait que c'est la tonte du premier-né qui est interdite (Devarim 15:19). Pour l'acte non intentionnel, il explique que l'homme n'a pas l'intention d'arracher : il dégage seulement le poil, et si une partie s'arrache, elle s'arrache. Tossefot discutent longuement la structure du doute. Ils expliquent que « arracher » dans la Michna s'entend sans intention, et que la question du jour de fête porte sur le point de savoir si, quand l'arrachage est un interdit de la Torah (déraciner de sa croissance), un acte non intentionnel est permis. Ils notent aussi une difficulté avec l'avis habituel de Rav, qui suit Rabbi Yehouda pour qui un acte non intentionnel est interdit, et proposent que, dans le cas du premier-né, Rav suive Rabbi Yossi parce qu'« arracher n'est pas tondre », et qu'un jour de fête l'arrachage est un déracinement fait « de façon inhabituelle » (kilachar yad), donc permis.

Enfin, Rachi précise que Rav 'Hananel est le disciple de Rav, et que « je vous expliquerai » signifie « je vous expliquerai la raison ». Tossefot demandent pourquoi Rav Houna insiste tant, alors qu'un jour de fête le dégagement est permis dans tous les cas (acte non intentionnel ou déracinement inhabituel), et répondent que, si la halakha n'était pas comme Rabbi Yossi, on aurait interdit pour le premier-né par prudence, de même que pour la préparation de l'abattage un jour de fête.

D'après Tosafot, Rashi
⚖️ Loi pratique

Les lois du premier-né animal concernent surtout l'époque du Temple, mais deux points de ce daf ont une portée pratique. D'abord, l'interdit d'abattre une bête et son petit le même jour (Vayikra 22:28) reste en vigueur, et la discussion sur la « pitié » des animaux montre comment on traite le doute sur l'identité de la mère. Ensuite, la règle de méthode de Rabbi Yo'hanan, selon laquelle la halakha suit Rabban Chimon ben Gamliel dans notre Michna sauf trois cas (la garantie, l'incident de Tsaïdan, la preuve dans la dernière controverse), est un outil classique pour la décision. Pour les lois de Yom Tov et de l'abattage d'un animal profane, il faut consulter un rabbin compétent.
