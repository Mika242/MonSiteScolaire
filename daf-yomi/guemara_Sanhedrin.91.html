אֲמַרָה לֵיהּ בְּרַתֵּיה: שִׁבְקֵיהּ, וַאֲנָא מַהְדַּרְנָא לֵיהּ. שְׁנֵי יוֹצְרִים יֵשׁ בְּעִירֵנוּ: אֶחָד יוֹצֵר מִן הַמַּיִם, וְאֶחָד יוֹצֵר מִן הַטִּיט. אֵיזֶה מֵהֶן מְשׁוּבָּח? אָמַר לָהּ: זֶה שֶׁיּוֹצֵר מִן הַמַּיִם. אָמְרָה לוֹ: מִן הַמַּיִם צָר, מִן הַטִּיט לֹא כׇּל שֶׁכֵּן?

דְּבֵי רַבִּי יִשְׁמָעֵאל תָּנָא: קַל וָחוֹמֶר מִכְּלֵי זְכוּכִית, מָה כְּלֵי זְכוּכִית שֶׁעֲמָלָן בְּרוּחַ בָּשָׂר וָדָם - נִשְׁבְּרוּ יֵשׁ לָהֶן תַּקָּנָה, בָּשָׂר וָדָם שֶׁבְּרוּחוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא – עַל אַחַת כַּמָּה וְכַמָּה.

אֲמַר לֵיהּ הָהוּא מִינָא לְרַבִּי אַמֵּי: אָמְרִיתוּ דְּשָׁכְבֵי חָיִי. וְהָא הָווּ עַפְרָא, וְעַפְרָא מִי קָא חָיֵי?! אֲמַר לֵיהּ: אֶמְשׁוֹל לְךָ מָשָׁל. לְמָה הַדָּבָר דּוֹמֶה? לְמֶלֶךְ בָּשָׂר וָדָם שֶׁאָמַר לַעֲבָדָיו: לְכוּ וּבְנוּ לִי פַּלְטֵרִין גְּדוֹלִים בִּמְקוֹם שֶׁאֵין מַיִם וְעָפָר. הָלְכוּ וּבָנוּ אוֹתָן. לְיָמִים נָפְלוּ. אָמַר לָהֶם: חִזְרוּ וּבְנוּ אוֹתָן בִּמְקוֹם שֶׁיֵּשׁ עָפָר וָמַיִם. אָמְרוּ לוֹ: אֵין אָנוּ יְכוֹלִין.

כָּעַס עֲלֵיהֶם, וְאָמַר לָהֶן: בִּמְקוֹם שֶׁאֵין מַיִם וְעָפָר בְּנִיתֶם, עַכְשָׁיו שֶׁיֵּשׁ מַיִם וְעָפָר – עַל אַחַת כַּמָּה וְכַמָּה! וְאִם אִי אַתָּה מַאֲמִין, צֵא לַבִּקְעָה וּרְאֵה עַכְבָּר שֶׁהַיּוֹם חֶצְיוֹ בָּשָׂר וְחֶצְיוֹ אֲדָמָה, לְמָחָר הִשְׁרִיץ וְנַעֲשָׂה כֻּלּוֹ בָּשָׂר. שֶׁמָּא תֹּאמַר לִזְמַן מְרוּבֶּה? עֲלֵה לָהָר וּרְאֵה שֶׁהַיּוֹם אֵין בּוֹ אֶלָּא חִלָּזוֹן אֶחָד, לְמָחָר יָרְדוּ גְּשָׁמִים וְנִתְמַלֵּא כּוּלּוֹ חֶלְזוֹנוֹת.

אֲמַר לֵיהּ הָהוּא מִינָא לִגְבִיהָא בֶּן פְּסִיסָא: וַוי לְכוֹן חַיָּיבַיָּא, דְּאָמְרִיתוּן מִיתֵי חָיִין! דְּחָיִין מָיְתִי, דְּמִיתֵי חָיִין? אֲמַר לֵיהּ: וַוי לְכוֹן חַיָּיבַיָּא, דְּאָמְרִיתוּן מִיתֵי לָא חָיִין! דְּלָא הֲווֹ חָיִי, דַּהֲווֹ חַיֵּי – לֹא כׇּל שֶׁכֵּן? אֲמַר לֵיהּ: חַיָּיבַיָּא קָרֵית לִי? אִי קָאֵימְנָא, בָּעֵיטְנָא בָּךְ וּפָשֵׁיטְנָא לְעַקְמוּתָךְ מִינָּךְ! אֲמַר לֵיהּ: אִם אַתָּה עוֹשֶׂה כֵּן, רוֹפֵא אוּמָּן תִּקָּרֵא, וְשָׂכָר הַרְבֵּה תִּטּוֹל.

תָּנוּ רַבָּנַן: בְּעֶשְׂרִים וְאַרְבָּעָה בְּנִיסָן אִיתְנְטִילוּ דֵּימוֹסְנָאֵי מִיהוּדָה וּמִירוּשָׁלַיִם. כְּשֶׁבָּאוּ בְּנֵי אַפְרִיקִיָּא לָדוּן עִם יִשְׂרָאֵל לִפְנֵי אֲלֶכְּסַנְדְּרוֹס מוֹקְדוֹן, אָמְרוּ לוֹ: אֶרֶץ כְּנַעַן שֶׁלָּנוּ הִיא, דִּכְתִיב ״אֶרֶץ כְּנַעַן לִגְבֻלֹתֶיהָ״, וּכְנַעַן אֲבוּהוֹן דְּהָנְהוּ אִינָשֵׁי הֲוָה.

אֲמַר לְהוּ גְּבִיהָא בֶּן פְּסִיסָא לַחֲכָמִים: תְּנוּ לִי רְשׁוּת וְאֵלֵךְ וְאָדוּן עִמָּהֶן לִפְנֵי אֲלֶכְּסַנְדְּרוֹס מוֹקְדוֹן. אִם יְנַצְּחוּנִי, אִמְרוּ: הֶדְיוֹט שֶׁבָּנוּ נִצַּחְתֶּם, וְאִם אֲנִי אֲנַצֵּחַ אוֹתָם, אִמְרוּ לָהֶם: תּוֹרַת מֹשֶׁה נִצְּחַתְכֶם. נָתְנוּ לוֹ רְשׁוּת, וְהָלַךְ וְדָן עִמָּהֶם.

אָמַר לָהֶם: מֵהֵיכָן אַתֶּם מְבִיאִים רְאָיָיה? אָמְרוּ לוֹ: מִן הַתּוֹרָה. אָמַר לָהֶן: אַף אֲנִי לֹא אָבִיא לָכֶם רְאָיָיה אֶלָּא מִן הַתּוֹרָה, שֶׁנֶּאֱמַר: ״וַיֹּאמֶר אָרוּר כְּנָעַן עֶבֶד עֲבָדִים יִהְיֶה לְאֶחָיו״. עֶבֶד שֶׁקָּנָה נְכָסִים – עֶבֶד לְמִי וּנְכָסִים לְמִי? וְלֹא עוֹד, אֶלָּא שֶׁהֲרֵי כַּמָּה שָׁנִים שֶׁלֹּא עֲבַדְתּוּנוּ!

אָמַר לָהֶם אֲלֶכְּסַנְדְּרוֹס מַלְכָּא: הַחְזִירוּ לוֹ תְּשׁוּבָה! אָמְרוּ לוֹ: תְּנוּ לָנוּ זְמַן שְׁלֹשָׁה יָמִים. נָתַן לָהֶם זְמַן. בָּדְקוּ וְלֹא מָצְאוּ תְּשׁוּבָה. מִיָּד בָּרְחוּ, וְהִנִּיחוּ שְׂדוֹתֵיהֶן כְּשֶׁהֵן זְרוּעוֹת, וְכַרְמֵיהֶן כְּשֶׁהֵן נְטוּעוֹת. וְאוֹתָהּ שָׁנָה שְׁבִיעִית הָיְתָה.

שׁוּב פַּעַם אַחַת בָּאוּ בְּנֵי מִצְרַיִם לָדוּן עִם יִשְׂרָאֵל לִפְנֵי אֲלֶכְּסַנְדְּרוֹס מוֹקְדוֹן. אָמְרוּ לוֹ: הֲרֵי הוּא אוֹמֵר ״וַה׳ נָתַן אֶת חֵן הָעָם בְּעֵינֵי מִצְרַיִם וַיַּשְׁאִלוּם״. תְּנוּ לָנוּ כֶּסֶף וְזָהָב שֶׁנְּטַלְתֶּם מִמֶּנּוּ.

אָמַר גְּבִיהָא בֶּן פְּסִיסָא לַחֲכָמִים: תְּנוּ לִי רְשׁוּת וְאֵלֵךְ וְאָדוּן עִמָּהֶן לִפְנֵי אֲלֶכְּסַנְדְּרוֹס. אִם יְנַצְּחוּנִי, אִמְרוּ לָהֶם: הֶדְיוֹט שֶׁבָּנוּ נִצַּחְתֶּם, וְאִם אֲנִי אֲנַצֵּחַ אוֹתָם, אִמְרוּ לָהֶם: תּוֹרַת מֹשֶׁה רַבֵּינוּ נִצְּחַתְכֶם. נָתְנוּ לוֹ רְשׁוּת וְהָלַךְ וְדָן עִמָּהֶן.

אָמַר לָהֶן: מֵהֵיכָן אַתֶּם מְבִיאִין רְאָיָיה? אָמְרוּ לוֹ: מִן הַתּוֹרָה. אָמַר לָהֶן: אַף אֲנִי לֹא אָבִיא לָכֶם רְאָיָיה אֶלָּא מִן הַתּוֹרָה, שֶׁנֶּאֱמַר: ״וּמוֹשַׁב בְּנֵי יִשְׂרָאֵל אֲשֶׁר יָשְׁבוּ בְּמִצְרָיִם שְׁלֹשִׁים שָׁנָה וְאַרְבַּע מֵאוֹת שָׁנָה״. תְּנוּ לָנוּ שְׂכַר עֲבוֹדָה שֶׁל שִׁשִּׁים רִיבּוֹא שֶׁשִּׁיעְבַּדְתֶּם בְּמִצְרַיִם שְׁלֹשִׁים שָׁנָה וְאַרְבַּע מֵאוֹת שָׁנָה.

אָמַר לָהֶן אֲלֶכְּסַנְדְּרוֹס מוֹקְדוֹן: הַחְזִירוּ לוֹ תְּשׁוּבָה! אָמְרוּ לוֹ: תְּנוּ לָנוּ זְמַן שְׁלֹשָׁה יָמִים. נָתַן לָהֶם זְמַן. בָּדְקוּ וְלֹא מָצְאוּ תְּשׁוּבָה. מִיָּד הִנִּיחוּ שְׂדוֹתֵיהֶן כְּשֶׁהֵן זְרוּעוֹת, וְכַרְמֵיהֶן כְּשֶׁהֵן נְטוּעוֹת, וּבָרְחוּ. וְאוֹתָהּ שָׁנָה שְׁבִיעִית הָיְתָה.

וְשׁוּב פַּעַם אַחַת בָּאוּ בְּנֵי יִשְׁמָעֵאל וּבְנֵי קְטוּרָה לָדוּן עִם יִשְׂרָאֵל לִפְנֵי אֲלֶכְּסַנְדְּרוֹס מוֹקְדוֹן. אָמְרוּ לוֹ: אֶרֶץ כְּנַעַן שֶׁלָּנוּ וְשֶׁלָּכֶם, דִּכְתִיב ״וְאֵלֶּה תֹּלְדֹת יִשְׁמָעֵאל בֶּן אַבְרָהָם״ וּכְתִיב ״אֵלֶּה תּוֹלְדֹת יִצְחָק בֶּן אַבְרָהָם״.

אָמַר לָהֶן גְּבִיהָא בֶּן פְּסִיסָא לַחֲכָמִים: תְּנוּ לִי רְשׁוּת וְאֵלֵךְ וְאָדוּן עִמָּהֶם לִפְנֵי אֲלֶכְּסַנְדְּרוֹס מוֹקְדוֹן. אִם יְנַצְּחוּנִי אִמְרוּ: הֶדְיוֹט שֶׁבָּנוּ נִצַּחְתֶּם, וְאִם אֲנִי אֲנַצֵּחַ אוֹתָם אִמְרוּ לָהֶם: תּוֹרַת מֹשֶׁה רַבֵּינוּ נִצְּחַתְכֶם. נָתְנוּ לוֹ רְשׁוּת, הָלַךְ וְדָן עִמָּהֶן.

אָמַר לָהֶם: מֵהֵיכָן אַתֶּם מְבִיאִין רְאָיָיה? אָמְרוּ לוֹ: מִן הַתּוֹרָה. אָמַר לָהֶן: אַף אֲנִי לֹא אָבִיא רְאָיָיה אֶלָּא מִן הַתּוֹרָה, שֶׁנֶּאֱמַר: ״וַיִּתֵּן אַבְרָהָם אֶת כׇּל אֲשֶׁר לוֹ לְיִצְחָק. וְלִבְנֵי הַפִּילַגְשִׁים אֲשֶׁר לְאַבְרָהָם נָתַן אַבְרָהָם מַתָּנֹת״. אָב שֶׁנָּתַן אֲגִטִּין לְבָנָיו בְּחַיָּיו וְשִׁיגֵּר זֶה מֵעַל זֶה, כְּלוּם יֵשׁ לָזֶה עַל זֶה כְּלוּם? מַאי מַתָּנוֹת? אָמַר רַבִּי יִרְמְיָה בַּר אַבָּא: מְלַמֵּד שֶׁמָּסַר לָהֶם שֵׁם טוּמְאָה.

אֲמַר לֵיהּ אַנְטוֹנִינוּס לְרַבִּי: גּוּף וּנְשָׁמָה יְכוֹלִין לִפְטוֹר עַצְמָן מִן הַדִּין. כֵּיצַד? גּוּף אוֹמֵר: נְשָׁמָה חָטָאת, שֶׁמִּיּוֹם שֶׁפֵּירְשָׁה מִמֶּנִּי הֲרֵינִי מוּטָּל כְּאֶבֶן דּוּמָם בַּקֶּבֶר. וּנְשָׁמָה אוֹמֶרֶת: גּוּף חָטָא, שֶׁמִּיּוֹם שֶׁפֵּירַשְׁתִּי מִמֶּנּוּ הֲרֵינִי פּוֹרַחַת בָּאֲוִיר כְּצִפּוֹר. אֲמַר לֵיהּ: אֶמְשׁוֹל לְךָ מָשָׁל. לְמָה הַדָּבָר דּוֹמֶה? לְמֶלֶךְ בָּשָׂר וָדָם שֶׁהָיָה לוֹ פַּרְדֵּס נָאֶה, וְהָיָה בּוֹ

בַּכּוּרוֹת נָאוֹת, וְהוֹשִׁיב בּוֹ שְׁנֵי שׁוֹמְרִים: אֶחָד חִיגֵּר וְאֶחָד סוֹמֵא. אָמַר לוֹ חִיגֵּר לַסּוֹמֵא: בַּכּוּרוֹת נָאוֹת אֲנִי רוֹאֶה בַּפַּרְדֵּס. בֹּא וְהַרְכִּיבֵנִי וּנְבִיאֵם לְאׇכְלָם. רָכַב חִיגֵּר עַל גַּבֵּי סוֹמֵא, וֶהֱבִיאוּם וַאֲכָלוּם.

לְיָמִים בָּא בַּעַל פַּרְדֵּס. אָמַר לָהֶן: בַּכּוּרוֹת נָאוֹת הֵיכָן הֵן? אָמַר לוֹ חִיגֵּר: כְּלוּם יֵשׁ לִי רַגְלַיִם לְהַלֵּךְ בָּהֶן? אָמַר לוֹ סוֹמֵא: כְּלוּם יֵשׁ לִי עֵינַיִם לִרְאוֹת? מָה עָשָׂה? הִרְכִּיב חִיגֵּר עַל גַּבֵּי סוֹמֵא, וְדָן אוֹתָם כְּאֶחָד.

אַף הַקָּדוֹשׁ בָּרוּךְ הוּא מֵבִיא נְשָׁמָה וְזוֹרְקָהּ בַּגּוּף, וְדָן אוֹתָם כְּאֶחָד, שֶׁנֶּאֱמַר: ״יִקְרָא אֶל הַשָּׁמַיִם מֵעָל וְאֶל הָאָרֶץ לָדִין עַמּוֹ״. ״יִקְרָא אֶל הַשָּׁמַיִם מֵעָל״ – זוֹ נְשָׁמָה, ״וְאֶל הָאָרֶץ לָדִין עַמּוֹ״ – זֶה הַגּוּף.

אֲמַר לֵיהּ אַנְטוֹנִינוּס לְרַבִּי: מִפְּנֵי מָה חַמָּה יוֹצְאָה בַּמִּזְרָח וְשׁוֹקַעַת בְּמַעֲרָב? אֲמַר לֵיהּ: אִי הֲוָה אִיפְּכָא נָמֵי הָכִי הֲוָה אָמְרַתְּ לִי. אֲמַר לֵיהּ, הָכִי קָאָמֵינָא לָךְ: מִפְּנֵי מָה שׁוֹקַעַת בַּמַּעֲרָב?

אֲמַר לֵיהּ: כְּדֵי לִיתֵּן שָׁלוֹם לְקוֹנָהּ, שֶׁנֶּאֱמַר ״וּצְבָא הַשָּׁמַיִם לְךָ מִשְׁתַּחֲוִים״. אֲמַר לֵיהּ: וְתֵיתֵי עַד פַּלְגָא דְרָקִיעַ, וְתִתֵּן שְׁלָמָא, וְתֵיעוֹל! מִשּׁוּם פּוֹעֲלִים וּמִשּׁוּם עוֹבְרֵי דְרָכִים.

וְאָמַר לוֹ אַנְטוֹנִינוּס לְרַבִּי: נְשָׁמָה מֵאֵימָתַי נִיתְּנָה בָּאָדָם, מִשְּׁעַת פְּקִידָה אוֹ מִשְּׁעַת יְצִירָה? אָמַר לוֹ: מִשְּׁעַת יְצִירָה. אָמַר לוֹ: אֶפְשָׁר חֲתִיכָה שֶׁל בָּשָׂר עוֹמֶדֶת שְׁלֹשָׁה יָמִים בְּלֹא מֶלַח וְאֵינָהּ מַסְרַחַת? אֶלָּא מִשְּׁעַת פְּקִידָה. אָמַר רַבִּי: דָּבָר זֶה לִמְּדַנִי אַנְטוֹנִינוּס, וּמִקְרָא מְסַיְּיעוֹ שֶׁנֶּאֱמַר ״וּפְקֻדָּתְךָ שָׁמְרָה רוּחִי״.

וְאָמַר לוֹ אַנְטוֹנִינוּס לְרַבִּי: מֵאֵימָתַי יֵצֶר הָרָע שׁוֹלֵט בָּאָדָם, מִשְּׁעַת יְצִירָה אוֹ מִשְּׁעַת יְצִיאָה? אָמַר לוֹ: מִשְּׁעַת יְצִירָה. אָמַר לוֹ: אִם כֵּן, בּוֹעֵט בִּמְעֵי אִמּוֹ וְיוֹצֵא! אֶלָּא מִשְּׁעַת יְצִיאָה. אָמַר רַבִּי: דָּבָר זֶה לִמְּדַנִי אַנְטוֹנִינוּס, וּמִקְרָא מְסַיְּיעוֹ, שֶׁנֶּאֱמַר: ״לַפֶּתַח חַטָּאת רֹבֵץ״.

רֵישׁ לָקִישׁ רָמֵי: כְּתִיב ״בָּם עִוֵּר וּפִסֵּחַ הָרָה וְיֹלֶדֶת יַחְדָּיו״, וּכְתִיב ״אָז יְדַלֵּג כָּאַיָּל פִּסֵּחַ וְתָרֹן לְשׁוֹן אִלֵּם כִּי נִבְקְעוּ בַמִּדְבָּר מַיִם וּנְחָלִים בָּעֲרָבָה״. הָא כֵּיצַד? עוֹמְדִין בְּמוּמָן וּמִתְרַפְּאִין.

עוּלָּא רָמֵי: כְּתִיב ״בִּלַּע הַמָּוֶת לָנֶצַח וּמָחָה ה׳ דִּמְעָה מֵעַל כׇּל פָּנִים״, וּכְתִיב ״כִּי הַנַּעַר בֶּן מֵאָה שָׁנָה יָמוּת לֹא יִהְיֶה מִשָּׁם עוֹד עוּל יָמִים״. לָא קַשְׁיָא: כָּאן – בְּיִשְׂרָאֵל, כָּאן – בְּאוּמּוֹת הָעוֹלָם. וְאוּמּוֹת הָעוֹלָם מַאי בָּעוּ הָתָם? הָנָךְ דִּכְתִיב בְּהוּ ״וְעָמְדוּ זָרִים וְרָעוּ צֹאנְכֶם וּבְנֵי נֵכָר אִכָּרֵיכֶם וְכֹרְמֵיכֶם״.

רַב חִסְדָּא רָמֵי: כְּתִיב ״וְחָפְרָה הַלְּבָנָה וּבוֹשָׁה הַחַמָּה כִּי מָלַךְ ה׳ צְבָאוֹת״, וּכְתִיב ״וְהָיָה אוֹר הַלְּבָנָה כְּאוֹר הַחַמָּה וְאוֹר הַחַמָּה יִהְיֶה שִׁבְעָתַיִם כְּאוֹר שִׁבְעַת הַיָּמִים״. לָא קַשְׁיָא: כָּאן לִימוֹת הַמָּשִׁיחַ, כָּאן לָעוֹלָם הַבָּא.

וְלִשְׁמוּאֵל, דְּאָמַר: אֵין בֵּין הָעוֹלָם הַזֶּה לִימוֹת הַמָּשִׁיחַ אֶלָּא שִׁיעְבּוּד גָּלִיּוֹת בִּלְבַד, לָא קַשְׁיָא: כָּאן בְּמַחֲנֵה צַדִּיקִים, כָּאן בְּמַחֲנֵה שְׁכִינָה.

רָבָא רָמֵי: כְּתִיב ״אֲנִי אָמִית וַאֲחַיֶּה״, וּכְתִיב ״מָחַצְתִּי וַאֲנִי אֶרְפָּא״. אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא: מַה שֶּׁאֲנִי מֵמִית – אֲנִי מְחַיֶּה, וַהֲדַר מַה שֶּׁמָּחַצְתִּי – וַאֲנִי אֶרְפָּא.

תָּנוּ רַבָּנַן: ״אֲנִי אָמִית וַאֲחַיֶּה״ – יָכוֹל שֶׁתְּהֵא מִיתָה בְּאֶחָד וְחַיִּים בְּאֶחָד, כְּדֶרֶךְ שֶׁהָעוֹלָם נוֹהֵג? תַּלְמוּד לוֹמַר: ״מָחַצְתִּי וַאֲנִי אֶרְפָּא״ – מָה מְחִיצָה וּרְפוּאָה בְּאֶחָד, אַף מִיתָה וְחַיִּים בְּאֶחָד. מִיכָּן תְּשׁוּבָה לָאוֹמְרִין אֵין תְּחִיַּית הַמֵּתִים מִן הַתּוֹרָה.

תַּנְיָא, אָמַר רַבִּי מֵאִיר: מִנַּיִין לִתְחִיַּית הַמֵּתִים מִן הַתּוֹרָה? שֶׁנֶּאֱמַר: ״אָז יָשִׁיר מֹשֶׁה וּבְנֵי יִשְׂרָאֵל אֶת הַשִּׁירָה הַזֹּאת לַה׳״. ״שָׁר״ לֹא נֶאֱמַר, אֶלָּא ״יָשִׁיר״ – מִכָּאן לִתְחִיַּית הַמֵּתִים מִן הַתּוֹרָה. כַּיּוֹצֵא בַּדָּבָר, אַתָּה אוֹמֵר: ״אָז יִבְנֶה יְהוֹשֻׁעַ מִזְבֵּחַ לַה׳״. ״בָּנָה״ לֹא נֶאֱמַר, אֶלָּא ״יִבְנֶה״ – מִכָּאן לִתְחִיַּית הַמֵּתִים מִן הַתּוֹרָה.

אֶלָּא מֵעַתָּה, ״אָז יִבְנֶה שְׁלֹמֹה בָּמָה לִכְמוֹשׁ שִׁקֻּץ מוֹאָב״, הָכִי נָמֵי דְּיִבְנֶה? אֶלָּא מַעֲלֶה עָלָיו הַכָּתוּב כְּאִילּוּ בָּנָה.

אָמַר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי: מִנַּיִין לִתְחִיַּית הַמֵּתִים מִן הַתּוֹרָה? שֶׁנֶּאֱמַר: ״אַשְׁרֵי יוֹשְׁבֵי בֵיתֶךָ עוֹד יְהַלְלוּךָ סֶּלָה״. ״הִילְּלוּךְ״ לֹא נֶאֱמַר, אֶלָּא ״יְהַלְלוּךָ״ – מִכָּאן לִתְחִיַּית הַמֵּתִים מִן הַתּוֹרָה. וְאָמַר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי: כׇּל הָאוֹמֵר שִׁירָה בְּעוֹלָם הַזֶּה זוֹכֶה וְאוֹמְרָהּ לָעוֹלָם הַבָּא, שֶׁנֶּאֱמַר: ״אַשְׁרֵי יוֹשְׁבֵי בֵיתֶךָ עוֹד יְהַלְלוּךָ סֶּלָה״.

אָמַר רַבִּי חִיָּיא בַּר אַבָּא אָמַר רַבִּי יוֹחָנָן: מִנַּיִין לִתְחִיַּית הַמֵּתִים מִן הַתּוֹרָה? שֶׁנֶּאֱמַר: ״קוֹל צֹפַיִךְ נָשְׂאוּ קוֹל יַחְדָּו יְרַנֵּנוּ וְגוֹ״. ״רִינְּנוּ״ לֹא נֶאֱמַר, אֶלָּא ״יְרַנֵּנוּ״ – מִכָּאן לִתְחִיַּית הַמֵּתִים מִן הַתּוֹרָה. וְאָמַר רַבִּי חִיָּיא בַּר אַבָּא אָמַר רַבִּי יוֹחָנָן: עֲתִידִין כׇּל הַנְּבִיאִים כּוּלָּן אוֹמְרִים שִׁירָה בְּקוֹל אֶחָד, שֶׁנֶּאֱמַר: ״קוֹל צֹפַיִךְ נָשְׂאוּ קוֹל יַחְדָּו יְרַנֵּנוּ״.

(Début de 91a : suite du débat sur la résurrection des morts, face à l'empereur.) La fille de l'empereur dit à Rabban Gamliel : laisse-le, je lui répondrai par une parabole. Il y a dans notre ville deux artisans : l'un façonne des vases avec de l'eau, l'autre avec de la boue. Lequel est le plus remarquable ? L'empereur répondit : celui qui façonne avec l'eau. Elle dit : s'il le fait avec de l'eau, à plus forte raison avec de la boue. De même, si D.ieu a pu créer le monde à partir de l'eau, il peut certainement ressusciter les hommes à partir de la poussière.

L'école de Rabbi Yichmaël a enseigné la résurrection par un raisonnement a fortiori à partir des vases de verre : si des vases façonnés par le souffle d'un être de chair et de sang peuvent être réparés quand ils se brisent (on les fond et on les souffle de nouveau), à plus forte raison l'homme, dont l'âme vient du souffle du Saint, béni soit-Il.

Un hérétique dit à Rabbi Ami : vous dites que les morts vivront, mais ils sont poussière ; la poussière revit-elle ? Rabbi Ami répondit par une parabole : un roi de chair et de sang dit à ses serviteurs de lui bâtir un grand palais dans un endroit sans eau ni terre ; ils le firent. Plus tard, le palais s'effondra, et le roi leur dit de le rebâtir dans un endroit avec eau et terre ; ils dirent qu'ils ne le pouvaient pas. Le roi se fâcha : si vous l'avez fait sans eau ni terre, à plus forte raison avec ! De même l'homme, que D.ieu a créé à partir de rien, à plus forte raison peut-Il le ressusciter à partir de la poussière. Et si tu ne crois pas qu'un être puisse naître de la poussière, va dans la vallée et regarde un animal (le akhbar) qui est aujourd'hui moitié chair et moitié terre et demain sera tout chair ; et pour ne pas dire que cela prend du temps, monte sur une montagne et vois que, aujourd'hui, il n'y a qu'un seul animal de cette sorte, et demain la terre sera pleine d'eux.

Un hérétique dit à Geviha ben Pesisa : malheur à vous, les impies, qui dites que les morts vivront : ceux qui vivent meurent, comment les morts revivraient-ils ? Geviha répondit : malheur à vous, impies, qui dites que les morts ne vivront pas : si ceux qui n'existaient pas ont vécu, à plus forte raison ceux qui ont déjà vécu. L'hérétique, furieux : tu m'as traité d'impie ? Si je me lève, je t'aplatirai ta bosse. Geviha répondit en plaisantant : si tu le fais, on dira que tu es un médecin expert et tu prendras de gros honoraires.

Geviha ben Pesisa devant Alexandre le Macédonien. On a enseigné dans la Meguilat Ta'anit : le vingt-quatre Nissan est un jour de joie, car les usurpateurs furent chassés de Judée et de Jérusalem. Les gens d'Afrikya vinrent plaider contre Israël devant Alexandre : « La terre de Canaan est à nous, comme il est écrit : “ce pays qui vous tombera en héritage, le pays de Canaan” (Nombres 34:2), et Canaan est notre ancêtre. » Geviha ben Pesisa dit aux Sages : donnez-moi la permission d'aller plaider ; si je suis battu, dites qu'ils ont battu un simple homme, et si je gagne, dites que c'est la Torah de Moïse qui les a battus. Il dit aux gens d'Afrikya : d'où tirez-vous votre preuve ? De la Torah. Moi aussi : « Maudit soit Canaan, il sera esclave des esclaves de ses frères » (Genèse 9:25). L'esclave qui a acquis des biens, à qui appartiennent l'esclave et ses biens ? À son maître. Et cela fait des années que vous ne nous avez pas servis. Alexandre leur demanda de répondre ; ils demandèrent trois jours, ne trouvèrent pas de réponse, et s'enfuirent en abandonnant leurs champs ensemencés et leurs vignes plantées. Cette année-là étant une année sabbatique, Israël put en consommer les produits.

Une autre fois, les Égyptiens vinrent plaider : « Il est écrit : “L'Éternel fit trouver grâce au peuple aux yeux des Égyptiens, qui leur prêtèrent” (Exode 12:36). Rendez-nous l'argent et l'or. » Geviha plaida de même : « Le séjour des enfants d'Israël en Égypte fut de quatre cent trente ans » (Exode 12:40) : donnez-nous le salaire des six cent mille hommes de plus de vingt ans que vous avez asservis pendant quatre cent trente ans. Ils demandèrent trois jours, ne trouvèrent pas de réponse, et s'enfuirent ; c'était encore une année sabbatique.

Une autre fois, les fils d'Ismaël et les fils de Ketoura vinrent plaider : « La terre de Canaan est à nous comme à vous : “Voici les générations d'Ismaël, fils d'Abraham” (Genèse 25:12), et “Voici les générations d'Isaac, fils d'Abraham” (25:19). » Geviha répondit : « Abraham donna tout ce qu'il avait à Isaac. Aux fils des concubines qu'avait Abraham, il fit des présents, et il les envoya loin de son fils Isaac, vers l'orient » (Genèse 25:5-6). Un père qui a donné un acte à ses fils de son vivant et qui a éloigné l'un d'eux, l'éloigné a-t-il encore des prétentions ? La Guemara demande : quels étaient ces présents ? Rabbi Yirmeya bar Abba dit : Abraham leur transmit le nom d'un esprit d'impureté (shem tum'a), pour la sorcellerie.

Antoninos et Rabbi (Yehouda haNassi). Antoninos dit à Rabbi : le corps et l'âme peuvent chacun se dégager du jugement. Le corps dit : l'âme a fauté, car depuis qu'elle m'a quitté je gis comme une pierre muette dans la tombe. L'âme dit : le corps a fauté, car depuis que je l'ai quitté, je vole dans les airs comme un oiseau. Rabbi répondit par une parabole : un roi avait un beau verger, et y plaça deux gardiens, un boiteux et un aveugle. Le boiteux dit à l'aveugle : je vois de belles figues précoces ; mets-moi sur tes épaules, je te guiderai. Ils les mangèrent. Quand le propriétaire demanda où étaient les figues, le boiteux dit : ai-je des jambes pour marcher ? L'aveugle dit : ai-je des yeux pour voir ? Le propriétaire plaça le boiteux sur les épaules de l'aveugle et les jugea ensemble. De même, le Saint, béni soit-Il, ramène l'âme dans le corps au jour du jugement et les juge ensemble, selon « Il appelle les cieux d'en haut et la terre pour juger Son peuple » (Psaumes 50:4) : les cieux, c'est l'âme ; la terre, le corps.

Antoninos demanda : pourquoi le soleil se lève-t-il à l'est et se couche-t-il à l'ouest ? Rabbi répondit : s'il en était autrement, tu me poserais la même question. Et pourquoi se couche-t-il toujours à l'ouest ? Pour saluer son Créateur, selon « l'armée des cieux se prosterne devant Toi » (Néhémie 9:6) ; la Présence divine est à l'ouest (le Saint des Saints est à l'ouest du Temple). Antoninos objecta : qu'il aille jusqu'au milieu du ciel, se prosterne et revienne à l'est. Rabbi répondit : il se couche à l'ouest à cause des ouvriers et des voyageurs, qui savent ainsi qu'il est temps de rentrer ou de trouver une auberge.

Antoninos demanda : depuis quand l'âme est-elle placée dans l'homme ? Dès la conception ou dès la formation de l'embryon, quarante jours plus tard ? Rabbi répondit : dès la formation. Antoninos : est-ce possible ? Un morceau de viande peut-il tenir trois jours sans sel sans pourrir ? C'est donc dès la conception. Rabbi dit : Antoninos m'a enseigné cela, et un verset l'appuie : « Ta providence a gardé mon esprit » (Job 10:12), dès la pekida. Antoninos demanda : depuis quand le mauvais penchant domine-t-il l'homme ? Rabbi : dès la formation. Antoninos : alors il ferait donner des coups au fœtus dans le ventre de sa mère pour sortir. C'est donc dès la sortie du ventre. Rabbi dit : Antoninos m'a enseigné cela, et un verset l'appuie : « Le péché est tapi à la porte » (Genèse 4:7).

Les contradictions sur la résurrection. Rech Lakich : « Je les ramènerai... l'aveugle et le boiteux » (Jérémie 31:7) indique que des gens auront des défauts ; « alors le boiteux bondira comme un cerf » (Isaïe 35:6) indique qu'ils n'en auront pas. Réponse : les morts ressusciteront avec leurs défauts, puis seront guéris. Oulla : « Il engloutira la mort à jamais » (Isaïe 25:8) et « le plus jeune mourra à cent ans » (Isaïe 65:20) : le premier verset concerne Israël, le second les nations. Rav 'Hisda : « la lune rougira et le soleil aura honte » (Isaïe 24:23) et « la lumière de la lune sera comme celle du soleil » (Isaïe 30:26) : l'un concerne les jours du Machia'h, l'autre le monde futur ; selon Chmouel, l'un concerne le camp des justes et l'autre le camp de la Présence divine. Rava : « Je fais mourir et je fais vivre » (Deutéronome 32:39) : ce que je tue, je ressuscite ; et ce que j'ai blessé, je le guérirai.

La résurrection dans la Torah. Une baraïta : « Je fais mourir et je fais vivre » peut signifier la mort d'un et la vie d'un autre ; mais « j'ai blessé et je guérirai » montre que, comme la blessure et la guérison se font chez la même personne, la mort et la vie aussi : c'est une réponse à ceux qui disent qu'il n'y a pas de résurrection dans la Torah. Rabbi Méir : « Alors Moïse et les enfants d'Israël chanteront (yashir) ce cantique » (Exode 15:1) : le futur indique que Moïse reviendra à la vie ; de même « Alors Josué bâtira (yivne) un autel » (Josué 8:30). La Guemara objecte avec « Alors Salomon bâtira un autel à Kemosh » (I Rois 11:7), qui ne prouve rien : le texte lui impute la faute comme s'il l'avait bâti, puisqu'il n'a pas empêché ses épouses. Rabbi Yehoshoua ben Lévi : « Heureux ceux qui habitent Ta maison, ils Te loueront encore » (Psaumes 84:5) : yehallelou-kha (futur). Et celui qui récite un cantique dans ce monde mérite de le réciter dans le monde futur. Rabbi 'Hiya bar Abba, au nom de Rabbi Yo'hanan : « Tes guetteurs élèvent la voix, ils chantent (yeranenou) ensemble » (Isaïe 52:8). Tous les prophètes chanteront d'une seule voix.

Conclusion. Rav 'Hana bar Bizna, au nom de Rabbi Chim'on 'Hassida : même les fœtus dans le ventre de leur mère le maudissent (Proverbes 11:26).

The daughter of the emperor said to Rabban Gamliel: Leave him, and I will respond to him with a parable. She said: There are two craftsmen in our city; one fashions vessels from water, and one fashions vessels from mortar. Which is more noteworthy? The emperor said to her: It is that craftsman that fashions vessels from water. His daughter said to him: If he fashions a vessel from the water, all the more so is it not clear that he can fashion vessels from mortar? The school of Rabbi Yishmael taught that resurrection of the dead a fortiori from glass vessels: If concerning glass vessels, which are fashioned by the breath of those of flesh and blood, who blow and form the vessels, and yet if they break they can be repaired, as they can be melted and subsequently blown again, then with regard to those of flesh and blood, whose souls are a product of the breath of the Holy One, Blessed be He, all the more so can God restore them to life.

The Gemara relates that a certain heretic said to Rabbi Ami: You say that the dead will live. Aren’t they dust? And does dust come to life? Rabbi Ami said to him: I will tell you a parable. To what is this matter comparable? It is comparable to a flesh-and-blood king who said to his servants: Go and construct for me a great palace [palterin] in a place where there is no water and earth available. They went and constructed it. Sometime later, the palace collapsed. The king said to them: Return to your labor and construct the palace in a place where there is earth and water available. They said to him: We are unable to do so.

The king became angry at them and said to them: If in a place where there is no water and earth available you constructed a palace, now that there is water and earth available all the more so should you be able to do so. Similarly, concerning man, whom God created ex nihilo, all the more so will God be able to resurrect him from dust. And if you do not believe that a being can be created from dust, go out to the valley and see an akhbar, a creature that today is half flesh and half earth, and tomorrow the being will develop and all of it will become flesh. Lest you say that creation of living creatures is a matter that develops over an extended period, ascend a mountain and see that today there is only one snail there; then ascend tomorrow, after rain will have fallen, and see that it will be entirely filled with snails.

The Gemara relates that a certain heretic said to Geviha ben Pesisa: Woe unto you, the wicked, as you say: The dead will come to life. The way of the world is that those who are alive die. How can you say that the dead will come to life? Geviha ben Pesisa said to him: Woe unto you, the wicked, as you say: The dead will not come to life. If those who were not in existence come to life, is it not reasonable all the more so that those who were once alive will come to life again? The heretic said to Geviha ben Pesisa angrily: You called me wicked? If I stand, I will kick you and flatten your hump, as Geviha ben Pesisa was a hunchback. Geviha ben Pesisa said to him jocularly: If you do so, you will be called an expert doctor and will take high wages for your services.

The lame person said: Do I have any legs with which I would be able to walk and take the figs? The blind person said: Do I have any eyes with which I would be able to see the way to the figs? What did the owner of the orchard do? He placed the lame person upon the shoulders of the blind person just as they did when they stole the figs, and he judged them as one.

So too, the Holy One, Blessed be He, brings the soul on the day of judgment and casts it back into the body, as they were when they sinned, and He judges them as one, as it is stated: “He calls to the heavens above and to the earth that He may judge His people” (Psalms 50:4). “He calls to the heavens above”; this is the soul, which is heavenly. “And to the earth that He may judge His people”; this is the body, which is earthly.

The Gemara relates another exchange. Antoninos said to Rabbi Yehuda HaNasi: For what reason does the sun emerge in the east and set in the west? Rabbi Yehuda HaNasi said to him: If it were the reverse, you would have also said that to me, as the sun must emerge from one direction and set in the other. Antoninos said to him: This is what I am saying to you: For what reason does the sun set in the west and not occasionally deviate and set elsewhere?

Rabbi Yehuda HaNasi said to him: The sun always sets in the west in order to greet its Creator, as it is stated: “And the hosts of heaven worship You” (Nehemiah 9:6). Setting is a form of worship; it is as though the sun is bowing to God. The Divine Presence rests in the west, as is evident from the fact that the Holy of Holies in the Temple, in which the Ark, the resting place of the Divine Presence, is located in the west. Antoninos said to him: If so, let the sun come until the midpoint of the sky, set slightly and greet its Creator, and return and enter its place of origin in the east and set there. Rabbi Yehuda HaNasi answered him: The sun sets in the west due to workers and due to travelers, as, if the sun did not proceed from east to west with the light of day gradually waning, they would not know that it is time to return home or to find an inn.

And Antoninos said to Rabbi Yehuda HaNasi: From when is the soul placed in a person? Is it from the moment of conception or from the moment of the formation of the embryo, forty days after conception? Rabbi Yehuda HaNasi said to him: It is from the moment of the formation of the embryo. Antoninos said to him: That is inconceivable. Is it possible that a piece of meat could stand for even three days without salt as a preservative and would not rot? The embryo could not exist for forty days without a soul. Rather, the soul is placed in man from the moment of conception. Rabbi Yehuda HaNasi said: Antoninos taught me this matter, and there is a verse that supports him, as it is stated: “And Your Providence [pekudatekha] has preserved my spirit” (Job 10:12) indicating that it is from the moment of conception [pekida] that the soul is preserved within a person.

And Antoninos said to Rabbi Yehuda HaNasi: From when does the evil inclination dominate a person? Is it from the moment of the formation of the embryo or from the moment of emergence from the womb? Rabbi Yehuda HaNasi said to him: It is from the moment of the formation of the embryo. Antoninos said to him: If so, the evil inclination would cause the fetus to kick his mother’s innards and emerge from the womb. Rather, the evil inclination dominates a person from the moment of emergence from the womb. Rabbi Yehuda HaNasi said: Antoninos taught me this matter, and there is a verse that supports him, as it is stated: “Sin crouches at the entrance” (Genesis 4:7), indicating that it is from the moment of birth, when the newborn emerges from the entrance of his mother’s womb, that the evil inclination lurks.

§ Reish Lakish raises a contradiction between two verses written with regard to the resurrection of the dead. It is written: “I will bring them from the north country and gather them from the ends of the earth, and with them the blind and the lame, the woman with child and the woman giving birth together” (Jeremiah 31:7), indicating that at the end of days there will still be people with physical defects. And it is written: “Then shall the lame man leap as a deer and the tongue of the mute sing; for in the wilderness shall waters break out, and streams in the desert” (Isaiah 35:6), indicating that at the end of days there will be no people with physical defects. How so? When resurrected, the dead will arise still afflicted with their defects, and they will then be healed.

Ulla raises a contradiction. It is written: “He will swallow death forever; and the Lord God will wipe tears from all faces” (Isaiah 25:8), indicating that death will no longer exist at the end of days. And it is written: “There shall be no more an infant a few days old then…for the youngest shall die one hundred years old” (Isaiah 65:20). The Gemara answers that this contradiction is not difficult. The verse here, in Isaiah chapter 25, is written with regard to the Jewish people, who will live forever after resurrection; the verse there, in Isaiah chapter 65, is written with regard to gentiles, who will ultimately die after an extremely long life. The Gemara asks: And what do gentiles seek, i.e., why will they merit to live, in that era? The Gemara answers that the verse is referring to those gentiles about whom it is written: “And strangers shall stand and feed your flocks, and aliens shall be your plowmen and vinedressers” (Isaiah 61:5).

Rav Ḥisda raises a contradiction. It is written: “Then the moon shall be confounded and the sun ashamed, when the Lord of hosts will reign in Mount Zion and in Jerusalem, and before His elders shall be His glory” (Isaiah 24:23), indicating that the sun and the moon will no longer shine at the end of days. And it is written: “And the light of the moon shall be as the light of the sun, and the light of the sun shall be sevenfold, as the light of seven days” (Isaiah 30:26), indicating that the sun and the moon will exist then and they will shine more brightly. The Gemara answers that this is not difficult. The verse here, in Isaiah chapter 30, is written with regard to the days of the Messiah, when the sun and moon will shine more brightly; the verse there, in Isaiah chapter 24, is written with regard to the World-to-Come, when the only light will be the light of God.

The Gemara asks: And according to Shmuel, who says: The difference between this world and the messianic era is only subjugation of the exiles, as during that era the Jewish people will be freed from that subjugation, how is the contradiction resolved? The Gemara answers that even according to Shmuel this contradiction is not difficult. The verse here, in Isaiah chapter 30, is written with regard to the light in the camp of the righteous; the verse there, in Isaiah chapter 24, the verse is written with regard to the camp of the Divine Presence, when the only light will be the light of God.

Rava raises a contradiction. It is written: “I will kill and I will bring to life” (Deuteronomy 32:39), indicating that God is capable of reviving the dead. And it is written immediately afterward: “I wounded and I will heal,” which indicates that God will only heal the wounded. Rather, it should be understood: The Holy One, Blessed be He, is saying: What I kill, I bring to life, indicating that God revives the dead. And then what I wounded, I will heal.

5. La résurrection dans la Torah

La Guemara cherche un fondement dans les temps verbaux : le futur de « alors Moïse chantera » (Exode 15:1) et « alors Josué bâtira » (Josué 8:30) suggère un retour à la vie (Rabbi Méir) ; le futur de « ils Te loueront » (Psaumes 84:5, Rabbi Yehoshoua ben Lévi) et de « ils chanteront ensemble » (Isaïe 52:8, Rabbi 'Hiya bar Abba) mène à la même idée.

6. Ne pas retenir la halakha

Le daf se termine sur un avertissement : empêcher un élève d'apprendre est comme le priver de son héritage (Rav Yehouda au nom de Rav) ; même les fœtus maudissent celui qui retient la halakha (Rav 'Hana bar Bizna au nom de Rabbi Chim'on 'Hassida).

Rachi sur la fille de l'empereur : c'est une parabole : si deux artisans étaient dans notre ville, l'un fabriquant des vases avec de l'eau, et ainsi de suite ; le Saint, béni soit-Il, forme l'homme d'une goutte putride qui est comme de l'eau, et à plus forte raison de la poussière, qui est entre Ses mains. Sur les vases de verre : leur fabrication se fait avec le souffle de l'artisan (un tube dans lequel il souffle), et on peut les fondre et les refaire.

Rachi sur « akhbar » : il en existe des espèces qui ne naissent pas par reproduction. Sur « Geviha ben Pesisa » : c'est son nom, et il avait une courbure du dos, comme on le voit plus loin. Sur « malheur à vous, impies » : « ceux que nous voyons vivants, nous les voyons mourir : comment penser que les morts revivent ? » ; et la réponse de Geviha : ceux qui n'ont jamais existé sont formés et naissent, ceux qui ont déjà vécu, à plus forte raison.

Rachi sur « les usurpateurs (dimousnaï) » : des revendicateurs violents qui voulaient prendre une part en Judée et à Jérusalem. La Meguilat Ta'anit compte tous ces jours où des miracles ont été faits pour Israël et qu'on a fixés comme jours de fête : le vingt-quatre Nissan est donc un jour de fête et il est interdit d'y faire des éloges funèbres. Sur « agatin » : des écrits qui empêchent l'un de prendre dans la part de l'autre, comme Abraham envers Ismaël, qui ne peut rien prétendre de ce qu'il a donné à Isaac. Sur « nom d'impureté » : la sorcellerie et les œuvres des démons. Sur « figues précoces (bekhorot) » : les figues primeurs.

Rachi sur « Il appelle les cieux d'en haut » : le Saint appelle l'âme, qui vient de là-haut, pour entrer dans le corps de l'homme, et appelle aussi le corps, qui vient de la terre. Sur le coucher du soleil : pour saluer son Créateur ; la Présence divine est à l'ouest, et lorsque le soleil y arrive, il se prosterne devant le Saint qui l'a créé, et c'est pourquoi il se couche et entre en étant penché. Sur « qu'il vienne jusqu'au milieu du ciel » : si c'était son intention, il irait au milieu du ciel et se prosternerait vers l'ouest, comme les hommes.

Rachi sur « dès la formation (yetsira) » : quand l'embryon est entièrement formé en chair, nerfs et os ; sur « dès la pekida » : dès que l'ange examine la goutte et l'amène devant D.ieu pour décider de son sort (comme dans Nida 16b), après quoi l'âme et la vie y sont jetées. Sur « dès la sortie » : quand il sort du ventre de sa mère.

Rachi sur Rech Lakich : « ils auront leurs défauts » car ils ressuscitent avec leur défaut, puis sont guéris (« le boiteux bondira comme un cerf »). Sur Oulla (« il engloutira la mort ») : dans le monde futur, comme ils vivent, ils ne meurent plus ; et sur « le jeune homme de cent ans » : quand quelqu'un mourra à cent ans, on dira qu'un enfant est mort. Sur Rav 'Hisda : « la lune rougira » signifie que leur lumière sera plus faible, et « la lumière de la lune sera comme celle du soleil » : l'un pour les jours du Machia'h, l'autre pour le monde futur, à cause de l'éclat des justes. Sur Rava : « Je fais mourir » signifie que, comme je fais mourir un homme, je le ressuscite ; s'il est mort avec un défaut, il ressuscite avec un défaut, puis est guéri.

Rachi sur « chantera (yashir) » : cela laisse entendre le futur. Sur « d'une seule voix » : chantant ensemble. Sur « celui qui retient la halakha » : qui refuse de l'enseigner ; « depuis les six jours de la Création » : car il est écrit « Au commencement D.ieu créa » à cause de la Torah, qui est « le commencement », et Israël est appelé « prémices de Sa récolte ».

⚖️ Loi pratique

Ce daf traite surtout de croyance et de pensée juive ; il ne contient presque pas de loi pratique directe, mais plusieurs enseignements restent fondamentaux.

À retenir

  • La résurrection des morts est un principe de la foi juive, que la Guemara fonde sur plusieurs versets de la Torah et des Prophètes (Sanhédrin 91a-91b).
  • Le corps et l'âme sont jugés ensemble, comme le boiteux sur les épaules de l'aveugle (Sanhédrin 91a-b).
  • Enseigner la halakha. Celui qui empêche un élève d'apprendre est comparé à celui qui le prive de son héritage (Rav Yehouda au nom de Rav).
  • Le 24 Nissan est cité par la Meguilat Ta'anit comme un jour de fête (Rachi).
