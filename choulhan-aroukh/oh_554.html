תִּשְׁעָה בְּאָב אָסוּר בִּרְחִיצָה וְסִיכָה, וּנְעִילַת הַסַנְדָּל וְתַשְׁמִישׁ הַמִּטָּה, וְאָסוּר לִקְרוֹת בְּתוֹרָה נְבִיאִים וּכְתוּבִים, וְלִשְׁנוֹת בַּמִּשְׁנָה וּבַמִּדְרָשׁ וּבַגְּמָרָא בַּהֲלָכוֹת וּבְאַגָּדוֹת, מִשּׁוּם שֶׁנֶּאֱמַר פִּקּוּדֵי ה' יְשָׁרִים מְשַׁמְּחֵי לֵב (תְּהִלִּים יט, ט) וְתִינוֹקוֹת שֶׁל בֵּית רַבָּן בְּטֵלִים בּוֹ אֲבָל קוֹרֵא הוּא בְּאִיּוֹב וּבַדְּבָרִים הָרָעִים שֶׁבְּיִרְמִיָּה; וְאִם יֵשׁ בֵּינֵיהֶם פְּסוּקֵי נֶחָמָה, צָרִיךְ לְדַלְּגָם.

וּמֻתָּר לִלְמֹד מִדְרַשׁ אֵיכָה וּפֶרֶק אֵלּוּ מְגַלְּחִין, וְכֵן לִלְמֹד פֵּרוּשׁ אֵיכָה וּפֵרוּשׁ אִיּוֹב.

יֵשׁ מִי שֶׁאוֹסֵר לִלְמֹד עַל יְדֵי הִרְהוּר.

וּמֻתָּר לִקְרוֹת כָּל סֵדֶר הַיּוֹם, וּפָרָשַׁת הַקָּרְבָּנוֹת, וּמִשְׁנַת אֵיזֶהוּ מְקוֹמָן, וּמִדְרַשׁ רַבִּי יִשְׁמָעֵאל. הַגָּה: וּמֻתָּר לַחֲזֹר הַפָּרָשָׁה בְּתִשְׁעָה בְּאָב (מִנְהָגִים).

עֻבָּרוֹת וּמֵינִיקוֹת מִתְעַנּוֹת בְּט' בְּאָב, וּמַשְׁלִימוֹת כְּדֶרֶךְ שֶׁמִּתְעַנּוֹת וּמַשְׁלִימוֹת בְּיוֹם כִּפּוּר; אֲבָל בְּג' צוֹמוֹת אֲחֵרִים פְּטוּרִים מִלְּהִתְעַנּוֹת, וְאַף עַל פִּי כֵן רָאוּי שֶׁלֹּא תֹּאכַלְנָה לְהִתְעַנֵּג בְּמַאֲכָל וּבְמִשְׁתֶּה, אֶלָּא כִּדֵי קִיּוּם הַוָּלָד וְעַיֵּן לְעֵיל סי' תק"נ סָעִיף א'.

חַיָּה כָּל שְׁלֹשִׁים יוֹם, וְכֵן חוֹלֶה שֶׁהוּא צָרִיךְ לֶאֱכֹל, אֵין צָרִיךְ אֹמֶד אֶלָּא מַאֲכִילִין אוֹתוֹ מִיָּד, דְּבִמְקוֹם חֹלִי לֹא גָּזְרוּ רַבָּנָן. הַגָּה: וּמִיהוּ נוֹהֲגִין לְהִתְעַנּוֹת כָּל זְמַן שֶׁאֵין לָהֶם צַעַר גָּדוֹל שֶׁהָיָה לָחוּשׁ לְסַכָּנָה, וְהַמֵּקֵל לֹא הִפְסִיד.

רְחִיצָה אֲסוּרָה בְּט' בְּאָב, בֵּין בְּחַמִּין בֵּין בְּצוֹנֵן; אֲפִלּוּ לְהוֹשִׁיט אֶצְבָּעוֹ בְּמַיִם, אָסוּר.

טְבִילָה שֶׁל מִצְוָה בִּזְמַנָּהּ, מֻתֶּרֶת; אֲבָל בַּזְּמַן הַזֶּה אֵין טְבִילָה בִּזְמַנָּהּ, הִלְכָּךְ לֹא תִּטְבֹּל בּוֹ; וְכֵן נָהֲגוּ.

אִם הָיוּ יָדָיו מְלֻכְלָכוֹת בְּטִיט וּבְצוֹאָה, מֻתָּר לִרְחֹץ לְהַעֲבִיר הַלִּכְלוּךְ, וְלֹא יִטֹּל כָּל יָדָיו אֶלָּא לְפִי הַצֹּרֶךְ לְהַעֲבִיר הַלִּכְלוּךְ. הַגָּה: וְאִם עָשָׂה צְרָכָיו, אִם מֻתָּר לִרְחֹץ, דִּינוֹ כְּמוֹ בְּיוֹם כִּפּוּר וְעַיֵּן לְקַמָּן סי' תרי"ג ס"ג.

נוֹטֵל אָדָם יָדָיו שַׁחֲרִית, וְצָרִיךְ לִזָּהֵר שֶׁלֹּא יִטֹּל יָדָיו אֶלָּא עַד סוֹף קִשְׁרֵי אֶצְבְּעוֹתָיו.

וּלְאַחַר שֶׁנִּגֵּב יָדָיו וַעֲדַיִן לַחוֹת קְצָת, מַעֲבִירָם עַל עֵינָיו; וְאִם הָיָה לִפְלוּף עַל גַּבֵּי עֵינָיו וְדַרְכּוֹ לִרְחֲצָם בְּמַיִם, רוֹחֵץ וּמַעֲבִירוֹ וְאֵינוֹ חוֹשֵׁשׁ, דַּהֲוָה לֵיהּ כְּטִיט וּכְצוֹאָה שֶׁרוֹחֵץ כְּדַרְכּוֹ וְאֵינוֹ חוֹשֵׁשׁ.

הַהוֹלֵךְ לְהַקְבִּיל פְּנֵי רַבּוֹ אוֹ אָבִיו, אוֹ מִי שֶׁגָּדוֹל מִמֶּנּוּ, אוֹ לְצָרְכֵי מִצְוָה, עוֹבֵר בְּמַיִם עַד צַוָּארוֹ וְאֵינוֹ חוֹשֵׁשׁ; וְכֵן בַּחֲזָרָה, מֻתָּר.

הַהוֹלֵךְ לִשְׁמֹר פֵּרוֹתָיו, עוֹבֵר בְּמַיִם עַד צַוָּארוֹ וְאֵינוֹ חוֹשֵׁשׁ; אֲבָל בַּחֲזָרָה, אָסוּר.

הַבָּא מִן הַדֶּרֶךְ וְרַגְלָיו כֵּהוֹת, מֻתָּר לִרְחֹץ בְּמַיִם. הַגָּה: וּמֻתָּר לִשְׁרוֹת מַפָּה בְּמַיִם בְּעֶרֶב תִּשְׁעָה בְּאָב וּמוֹצִיאָהּ מִן הַמַּיִם וְהִיא מִתְנַגֶּבֶת, וּמְקַנֵּחַ בָּהּ פָּנָיו יָדָיו וְרַגְלָיו; אֲפִלּוּ אֵינוֹ עוֹשֶׂה רַק לְתַעֲנוּג, שָׁרֵי, כֵּיוָן שֶׁהִיא נְגוּבָה (טוּר).

סִיכָה אֵינָהּ אֲסוּרָה אֶלָּא שֶׁל תַּעֲנוּג; אֲבָל מִי שֶׁיֵּשׁ לוֹ חֲטָטִין בְּרֹאשׁוֹ, סָךְ כְּדַרְכּוֹ וְאֵינוֹ חוֹשֵׁשׁ.

נְעִילַת הַסַנְדָּל, דַּוְקָא שֶׁל עוֹר; אֲבָל שֶׁל בֶּגֶד אוֹ שֶׁל עֵץ אוֹ שֶׁל שַׁעַם (פִּי' קְלִפֵּי עֵץ) וְגֶמִּי, מֻתָּר. וְשֶׁל עֵץ מְחֻפֶּה עוֹר, אָסוּר.

אָבֵל וּמְנֻדֶּה שֶׁמְּהַלְּכִים בַּדֶּרֶךְ מֻתָּרִים בִּנְעִילַת הַסַנְדָּל, וּכְשֶׁיַּגִּיעוּ לָעִיר יַחֲלֹצוּ, וְכֵן בְּתִשְׁעָה בְּאָב. הַגָּה: וְכֵן בְּמָקוֹם שֶׁדָּרִים בֵּין הָעַכּוּ"ם, לֹא יַחֲלֹץ כִּי אִם בִּרְחוֹב הַיְּהוּדִים, וְכֵן נָהֲגוּ (טוּר וְהַגָּהוֹת מַיְמוֹנִי וְהַגָּהוֹת מָרְדְּכַי רפ"ק).

יֵשׁ מִי שֶׁאוֹמֵר שֶׁלֹּא יִישַׁן בְּלֵיל ט' בְּאָב עִם אִשְׁתּוֹ בַּמִּטָּה, וְנָכוֹן הַדָּבָר מִשּׁוּם לֵךְ לֵךְ אַמְרִינָן נְזִירָא.

אִם חָל תִּשְׁעָה בְּאָב בְּשַׁבָּת, מֻתָּר בְּכֻלָּן אֲפִלּוּ בְּתַשְׁמִישׁ הַמִּטָּה. הַגָּה: וְיֵשׁ אוֹסְרִים בְּתַּשְׁמִישׁ הַמִּטָּה, וְכֵן נוֹהֲגִין (טוּר בְּשֵׁם ר"י וְאָגוּר וּמִנְהָגִים).

אֵין שְׁאֵלַת שָׁלוֹם לַחֲבֵרוֹ בְּתִשְׁעָה בְּאָב, וְהֶדְיוֹטוֹת שֶׁאֵינָם יוֹדְעִים וְנוֹתְנִים שָׁלוֹם, מְשִׁיבִים לָהֶם בְּשָׂפָה רָפָה וּבְכֹבֶד רֹאשׁ.

יֵשׁ מִי שֶׁאוֹמֵר שֶׁלֹּא יֵלֵךְ וִיטַיֵּל בַּשּׁוּק, כְּדֵי שֶׁלֹּא יָבֹא לִידֵי שְׂחוֹק וְקַלּוּת וְהִתּוּל.

מָקוֹם שֶׁנָּהֲגוּ לַעֲשׂוֹת מְלָאכָה בְּט' בְּאָב, עוֹשִׂין; בְּמָקוֹם שֶׁנָּהֲגוּ שֶׁלֹּא לַעֲשׂוֹת, אֵין עוֹשִׂין; וּבְכָל מָקוֹם תַּלְמִידֵי חֲכָמִים בְּטֵלִים, וְכָל הָרוֹצֶה לַעֲשׂוֹת עַצְמוֹ תַּלְמִיד חָכָם לְעִנְיָן זֶה עוֹשֶׂה; וַאֲפִלּוּ בְּמָקוֹם שֶׁנָּהֲגוּ שֶׁלֹּא לַעֲשׂוֹת, מֻתָּר עַל יְדֵי עַכּוּ"ם, אֲפִלּוּ בְּבֵיתוֹ; וּפְרַקְמַטְיָא לְהַרְוִיחַ וּלְהִשְׂתַּכֵּר בְּמָקוֹם שֶׁנָּהֲגוּ שֶׁלֹּא לַעֲשׂוֹת מְלָאכָה, אָסוּר; וּבְמָקוֹם שֶׁנָּהֲגוּ לַעֲשׂוֹת, מֻתָּר אֶלָּא שֶׁמְּמַעֵט שֶׁאֲפִלּוּ מִשֶּׁנִּכְנַס אָב אָנוּ מְמַעֲטִין מִלִּשָּׂא וְלִתֵּן. הַגָּה: וְלֹא נָהֲגוּ בְּאִסּוּר מְלָאכָה כִּי אִם עַד חֲצוֹת (מִנְהָגִים). וְנָהֲגוּ לְהַחְמִיר עַד חֲצוֹת בְּכָל מְלָאכָה שֶׁיֵּשׁ בָּהּ שִׁהוּי קְצָת, אֲפִלּוּ מַעֲשֵׂה הֶדְיוֹט; אֲבָל דָּבָר שֶׁאֵין בָּהּ שִׁהוּי, כְּגוֹן הַדְלָקַת נֵרוֹת אוֹ קְשִׁירָה וְכַדּוֹמֶה, מֻתֶּרֶת. וְלַחְלֹב הַפָּרוֹת טוֹב לַעֲשׂוֹת עַל יְדֵי עַכּוּ"ם, אִם אֶפְשָׁר בְּעַכּוּ"ם.

וּמְלֶאכֶת דָּבָר הָאָבֵד מֻתָּר, כְּדֶרֶךְ שֶׁאָמְרוּ בְּחֻלּוֹ שֶׁל מוֹעֵד.

כָּל הָעוֹשֶׂה מְלָאכָה בְּתִשְׁעָה בְּאָב, אֵינוֹ רוֹאֶה סִימַן בְּרָכָה מֵאוֹתָהּ מְלָאכָה.

כָּל הָאוֹכֵל וְשׁוֹתֶה בְּתִשְׁעָה בְּאָב, אֵינוֹ רוֹאֶה בְּשִׂמְחַת יְרוּשָׁלַיִם; וְכָל הַמִּתְאַבֵּל עַל יְרוּשָׁלַיִם, זוֹכֶה וְרוֹאֶה בְּשִׂמְחָתָהּ. וְכָל הָאוֹכֵל בָּשָׂר אוֹ שׁוֹתֶה יַיִן בַּסְעֻדָּה הַמַּפְסֶקֶת, עָלָיו הַכָּתוּב אוֹמֵר: וַתְּהִי עֲוֹנֹתָם עַל עַצְמוֹתָם (יְחֶזְקֵאל לב, כז).

דברים האסורים בתשעה באב. ובו כה סעיפים:
תשעה באב (א)אסור ברחיצה וסיכה ונעילת הסנדל ותשמיש המיטה ואסור לקרות בתורה נביאים וכתובים ולשנות במשנה ובמדרש ובגמ' בהלכות ובאגדות משום שנאמר פקודי ה' ישרים משמחי לב ותינוקות של בית רבן בטלים בו אבל קורא הוא באיוב ובדברים (ב)הרעים שבירמיה ואם יש ביניהם פסוקי נחמה צריך לדלגם:

ומותר ללמוד מדרש (ג)איכה ופרק אלו (ד)מגלחין וכן ללמוד פירוש איכה ופירוש איוב:

יש מי שאוסר ללמוד (ה)ע"י הרהור:

Les interdits du 9 Av (25 se'ifim).

1. Tich'a beAv, il est interdit de se laver, de s'oindre, de chausser des sandales, d'avoir des relations conjugales, et il est interdit de lire dans la Torah, les Prophètes et les Écrits, d'étudier la Michna, le Midrach, le Talmud, les lois et les récits, car il est écrit « les préceptes de l'Éternel sont droits, réjouissent le cœur », et les enfants de l'école sont libérés ; mais on peut lire Iyov et les passages de malheur de Yirmiyahou, et si des versets de consolation s'y trouvent, il faut les sauter.

2. On peut étudier le Midrach d'Eikha et le chapitre « Ceux qui se rasent », ainsi que le commentaire d'Eikha et celui d'Iyov.

3. Certains interdisent d'étudier même en pensée.

4. On peut lire tout l'ordre du jour, le passage des sacrifices, la michna « Quel est leur lieu » et le Midrach de Rabbi Yichmaël. Rema : on peut répéter la paracha le 9 Av (Minhaguim).

5. Les femmes enceintes et allaitantes jeûnent à Tich'a beAv et achèvent comme à Yom Kippour ; mais pour les trois autres jeûnes elles sont exemptées ; néanmoins il est bon qu'elles ne mangent pas pour se délecter mais seulement pour la survie de l'enfant (voir 550:1).

6. Une accouchée jusqu'à trente jours, et un malade qui a besoin de manger, n'ont pas besoin d'estimation : on les fait manger tout de suite, car dans le cas de maladie les Sages n'ont pas décrété. Rema : on a pourtant l'usage de jeûner tant qu'il n'y a pas de grande souffrance faisant craindre un danger, et celui qui allège n'a rien perdu.

7. La toilette est interdite à Tich'a beAv, à l'eau chaude comme froide, même passer son doigt dans l'eau.

8. L'immersion de mitsva en son temps est permise, mais de nos jours il n'y a pas d'immersion en son temps ; elle ne s'immerge donc pas, et l'on a l'usage ainsi.

9. Si ses mains sont salies de boue ou d'excréments, il lui est permis de se laver pour enlever la saleté, sans laver toute la main, mais seulement ce qu'il faut. Rema : et s'il a fait ses besoins, voir pour la toilette comme à Yom Kippour (613:3).

10. On se lave les mains le matin, en veillant à ne les laver que jusqu'aux articulations des doigts.

11. Après les avoir essuyées, quand elles sont encore un peu humides, on les passe sur ses yeux ; et s'il y a du mucus sur les yeux et qu'il les lave d'habitude à l'eau, il les lave et l'enlève : c'est comme de la boue ou de l'excrément.

12. Celui qui va saluer son maître ou son père ou plus grand que lui, ou pour une mitsva, traverse l'eau jusqu'au cou sans souci, et au retour c'est permis aussi.

13. Celui qui va garder ses fruits traverse l'eau jusqu'au cou sans souci, mais au retour c'est interdit.

14. Celui qui revient de la route et dont les pieds sont fatigués peut les laver. Rema : on peut tremper un linge dans l'eau la veille de Tich'a beAv, le sortir de l'eau, et il sèche ; on s'en frotte le visage et les pieds, même pour son plaisir, puisqu'il est sec (Tour).

15. L'onction n'est interdite que pour le plaisir ; celui qui a des plaies à la tête s'oint comme d'habitude sans souci.

16. Les chaussures de cuir seulement ; celles de tissu, de bois ou de liège et de jonc sont permises ; celles de bois recouvertes de cuir sont interdites.

17. L'endeuillé et l'excommunié qui marchent sur la route peuvent chausser des sandales, et arrivés en ville ils les ôtent ; de même à Tich'a beAv. Rema : de même, là où ils vivent parmi des non-Juifs, on n'ôte ses chaussures que dans la rue des Juifs, et c'est l'usage.

18. Selon certains, on ne dort pas la nuit de Tich'a beAv avec sa femme dans le lit, et c'est bien, car on dit au nazir : « va, va autour de la vigne, n'approche pas ».

19. Si Tich'a beAv tombe un Chabbat, tout est permis, même les relations conjugales. Rema : certains interdisent les relations, et l'usage est ainsi (Tour au nom de Rabbénou Yits'haq, Agour, Minhaguim).

20. On ne salue pas son prochain à Tich'a beAv ; si des gens simples, ne sachant pas, saluent, on leur répond d'une voix faible et avec gravité.

21. Selon certains, on ne se promène pas dans le marché, pour ne pas en venir au rire, à la légèreté et à la moquerie.

22. Là où l'on a l'usage de travailler le 9 Av, on travaille ; là où l'on a l'usage de ne pas travailler, on ne travaille pas ; partout les sages chôment, et quiconque veut se faire sage pour cette règle chôme ; même là où l'on ne travaille pas, c'est permis par un non-Juif, même chez soi ; et le commerce pour gagner et profiter, là où l'on a l'usage de ne pas travailler, est interdit, et là où l'on a l'usage de travailler, permis, en diminuant…

23. Le travail d'une perte est permis, comme on l'a dit pour 'Hol haMoed.

24. Quiconque travaille le 9 Av ne voit pas de bénédiction dans ce travail.

25. Quiconque mange et boit le 9 Av ne verra pas la joie de Jérusalem ; quiconque pleure Jérusalem mérite d'en voir la joie ; et quiconque mange de la viande ou boit du vin au repas de séparation, sur lui il est écrit : « Que leurs fautes soient sur leurs os ».

Tisha b'Av is forbidden for washing, anointing, wearing leather shoes, and marital relations. It is also forbidden to read from the Torah, Nevi'im, and Ksuvim and to learn mishna and midrash and gemara and halacha and aggada, because it says, "The precepts of God are right, gladdening the heart" (Tehillim 19:9). Schoolchildren are idle on it. One may read Iyov and the bad things which are in Yirmiyah, but if there are between them passages of consolation, one must skip them.

It is permitted to learn the exegesis of Eichah and Perek Ein Megalchin (third and final perek of Moed Katan). Likewise, one may learn the interpretations of Eichah and Iyov.

There are those who prohibit learning by thought (and not out loud).

It is permitted to read the entire Order of the Day (of prayers) - the paragraphs of the Sacrifices, the Mishnah (Chapter) of Eizu Mekoman, and the teaching of Rabbi Yishmael. Rema: It is permitted to review the Torah reading (of Tisha B'av) on Tisha B'av.

Pregnant and nursing women fast on the 9th of Av just as they fast and complete the fast on Yom Kippur. On the other 3 fasts they are exempt from fasting, nevertheless it is proper that eating and drinking should be solely for the purpose of sustaining the baby, instead of deriving enjoyment from the food and drink. (Also see prior in Siman 550, Paragraph 1)

A chayah (=a new mother) all 30 days (after delivery) and a sick person that needs to eat, it is not necessary to assess; rather, we feed to them immediately because in a place of sickness, the Rabbis did not decree. Rema: However, we are accustomed to fast the whole time that there is not a great pain which would cause to worry for danger; however, a lenient approach doesn't lose.

Washing is forbidden on Tisha B'av whether with warm or cold (water) even to dip one's fingers into water is forbidden.

Mitzvah immersions in their proper time are allowed, but today we do not immerse in the proper time; therefore, one should not immerse on it (Tisha B'av) and so the is the custom.

If one's hand are dirty with dirt or excrement, it is allowed to wash to get off the dirt, yet he should not wash his whole hand rather according to the need to get off the dirt. Rema: If one does their bodily needs (i.e. goes to the bathroom) if it is allowed to wash, the law is like Yom Kippur and further see Siman 613 Paragraph 3.

A person who pours (the water) on his hands in the morning (for the ritual hand-washing) needs to to be careful that he not pour on his hand except to the end of the joints of the fingers.

And after he dries his hands, and they are still a little moist, he can pass them on his eyes (to clean) if there is pus on his eyes and his normal matters to wash them with water and pass over. And he doesn't have to be concerned because that it is like dirt and excrement that he washes normally and there is no concern.

One who is going to visit his teacher, father, or one who is bigger (in Torah knowledge) than him, or for a Mitzvah need, can cross in water until his neck and he has no concern. And same on the return, it is allowed.

One who is going to guard his fruits, can cross in water until his neck and he has no concern. But on the return, it is forbidden.

One who returns from the road and his feet are darkened, it is allowed to wash them with water. Rema: It is permitted to soak a cloth in water on the eve of Tisha B'av and take out from the water and let dry. And (on Tisha B'av) wipe with it one's face and feet even if one does only for pleasure it is fine since it is (Halachically) dry [Tur].

Anointing is not forbidden except for pleasure but one who has scabs on their head can normally anoint and he has no concern.

(The prohibition of) wearing shoes is specifically by leather but of garment or wood or cork explanation - word pieces or silk is allowed. Of wood covered by leather is forbidden.

A mourner or an excommunicant that is going on the road is allowed to wear shoes and when he reaches the city, he takes them off. And so too on Tisha B'av. Rema: And same in a place that we dwell amongst Gentiles, he does not take off unless on a Jewish street, and so is the custom. [Tur and Haagos Maimoni, Haagos Mordechai beginning of the first Chapter].

There is one who says not to sleep on the night of Tisha B'av with his wife in (the same) bed. And the matter is fitting to do because "go go" (i.e. away from the orchard of grapes) that we say to a Nazir (who is forbidden in eating grape products).

If Tisha B'av falls out on Shabbos, it is permitted in all (i.e. all acts with his wife) even marital relations are allowed. Rema: There are those that forbid marital relations, and so we are accustomed [Tur in the name of the Ri, Agur, and Minhagim.]

He should not greet "Shalom" to his friend on Tisha B'av. And fools that do not know and give "Shalom", we respond to them in a quiet tone and with solemnity.

There is one who says not to go and leisurely walk in the market (on Tisha B'av) in order not to come to laughter, irreverence, and mockery.

A place that is accustomed to do work on Tisha B'av, one can do. A place that is accustomed that they do not do work on Tisha B'av, one should not do. In all places, Talmud Chachim (Torah Sages) are idle (i.e. from work), and all who want to make themselves a Talmud Chacham regarding this matter, one can do. Even in a place that is accustomed not to do work, it is allowed via a Gentile even in his (the Jew's) house. (Regarding) doing business for profit or to get wages (i.e. as opposed to for preventing loss etc.), in a place that is accustomed not to do work, it is forbidden. In a place that is accustomed to do work, it is permitted. However, that it should be minimized, as even upon the entry of (the month of) Av, we reduce transactions. Rema: It is not the custom in forbidding work except until Halachic Noon [Minhagim]. It is the custom to be strict until Halachic Noon in all work that there is some delay even a non-professional act. But something that does not have delay like lighting a candle, tying (a knot) and similar are permissible. Milking cows, it is good to do via a Gentile if possible with a Gentile.

Work that is a matter of loss (Davar Haavid) is permitted like the way that it is said by Chol HaMoed.

All that do work on Tisha B'av, one will not see a sign of blessing from that work.

All that eat and drink on Tisha B'av, will not see the joy of Jerusalem. And all that mourn on Jerusalem, will merit and see it in her joy. And that eat meat and drink wine at the Parting Meal, on him the verse says their iniquities are on their bones (Ezekiel 32:27).

Traduction française de ce commentaire à venir — traduction anglaise (éventuellement partielle) affichée. L'onglet « Michna Beroura » en explique l'essentiel en français.

Le siman 554 est le centre de la section. À Tich'a beAv, cinq choses sont interdites, comme à Yom Kippour : se laver, s'oindre, chausser des sandales (de cuir), les relations conjugales, et l'étude de la Torah (qui « réjouit le cœur ») ; on lit seulement ce qui est triste (Iyov, Eikha, les malheurs de Yirmiyahou) ; les enfants de l'école sont libérés.

Pour la toilette, on peut laver ce qui est sali, les mains au réveil jusqu'aux articulations, et l'on traverse l'eau pour aller saluer son maître ou pour une mitsva. Les femmes enceintes ou allaitantes jeûnent et achèvent ; une accouchée (trente jours) ou un malade qui a besoin de manger mangent sans estimation. L'immersion rituelle d'une femme n'a pas lieu le 9 Av.

On ne se salue pas, on ne se promène pas au marché ; le travail suit l'usage du lieu (partout les sages chôment) ; qui travaille ne voit pas de bénédiction dans son travail ; qui mange le 9 Av ne verra pas la joie de Jérusalem, et qui la pleure la verra. Si Tich'a beAv tombe un Chabbat, tout est permis (usage : sauf les relations).

La Michna Beroura explique que tout cela est interdit tout le jour, même au crépuscule, jusqu'à la sortie des étoiles, et la prière d'Arvit avant ne change rien ; que « les malheurs » (Yirmiyahou) ne permettent pas les malheurs des non-Juifs ; qu'avec les enfants, on n'étudie même pas les malheurs, mais on peut leur enseigner la destruction, simple récit qui brise le cœur de l'enfant ; que le Midrach d'Eikha et les récits de destruction (Guittin, Sanhédrin) sont permis, ainsi que ce que le Yosippon relate ; que « Ceux qui se rasent » (Moed Katan) est permis à l'étude, mais la discussion halakhique est interdite ; que, même en pensée, l'étude est interdite, car la pensée donne aussi de la joie, d'où l'interdiction d'une explication, d'une difficulté ou d'une solution, même dans les passages permis ; que « l'ordre du jour » comprend le Chéma et ses bénédictions et tout ce qui touche à la prière ; que l'on dit des Psaumes et le Chant de l'Unité à Min'ha ; que « répéter la paracha » vise le chef de prière lecteur, même à Min'ha, car le matin on lit « Quand tu auras des enfants », qui est du sujet ; que l'accouchée de trente jours est présumée faible, comme un malade sans danger ; que « un malade » vise celui dont l'esprit est dérangé et qui guérit de jour en jour : il mange viande et vin toute la semaine et ne jeûne pas ; que « pas d'estimation » signifie qu'il n'est pas nécessaire d'estimer si le jeûne mène au danger : même sans danger, on les nourrit ; que l'usage de jeûner vaut même si elle dit « j'ai besoin de manger », mais seulement après sept jours (dans les sept jours, même si elle dit qu'elle n'en a pas besoin, elle ne jeûne pas) ; que celui qui allège n'a rien perdu, a fortiori un homme faible et malade ; qu'aujourd'hui, les femmes attendent sept jours propres sur une goutte de sang, donc l'immersion n'est pas en son temps et elle ne s'immerge pas, mais se lave et se frotte la veille et, à la sortie du 9 Av, se frotte un peu avant l'immersion ; que l'on ne s'interdit que la toilette de plaisir, d'où les femmes qui cuisinent et lavent la viande peuvent aussi laver leurs mains ; que la toilette des mains le matin est faite à cause de l'esprit mauvais qui se pose sur les mains, comme de la boue, mais le Hagra en doute ; qu'on passe aussi les mains sur le visage et les pieds, car il n'y a pas assez d'eau pour mouiller ; que, pour aller voir son maître, c'est permis même en semaine, car il y a une mitsva d'entendre de lui des paroles de Torah ; que celui qui garde ses fruits est permis à cause de la perte d'argent ; que celui dont les pieds sont fatigués peut les laver, car ce n'est pas du plaisir mais un remède ; que l'onction est l'usage d'une substance habituelle, comme huile, graisse ou savon, même sur un seul membre ; que la jeune mariée, jusqu'à trente jours après son mariage, peut se laver le visage et s'oindre ; que les chaussures de tissu ou de bois sont permises même si elles protègent le pied et ont la forme de chaussures ; que de bois recouvert de cuir est interdit même si les semelles sont de bois ; que sur la route c'est permis, car marcher pieds nus longtemps est pénible, mais dans un champ proche de la ville c'est interdit, de même au cimetière, sauf boue ; qu'on y met de la terre dans les chaussures ; qu'on ne les ôte pas parmi les non-Juifs parce qu'ils se moquent ; que, pour les relations, on interdit de toucher à sa femme comme à une niddah (voir 615), mais de jour on peut être indulgent ; que « va, va autour de la vigne » est le mot adressé au nazir ; que, si Tich'a beAv tombe un Chabbat, c'est permis car la Braïta dit qu'il ne s'abstient de rien à sa table, mais certains l'interdisent, comme pour celui qui enterre son mort un jour de fête, et l'usage est d'être strict, le bedeau annonçant que les choses privées s'observent ; mais si la nuit d'immersion tombe ce Chabbat, on peut s'appuyer sur la première opinion ; que le salut, ou « bonjour », est interdit, de même envoyer un cadeau ; qu'aux ignorants il convient d'apprendre qu'on ne salue pas aujourd'hui, pour qu'ils n'en conçoivent pas d'inimitié ; que l'on ne travaille pas afin de ne pas détourner son esprit du deuil, et que la nuit aussi c'est interdit, mais tout travail sans délai est permis ; que « travaille » signifie sans que cela paraisse de l'arrogance : on dirait qu'il n'avait rien à faire ; que, dans notre pays, où l'on a l'usage de ne pas travailler, c'est permis par un non-Juif, mais construire est interdit même par un non-Juif, car cela se remarque ; que le dommage d'une perte, si une foire ou une caravane arrive, on peut acheter d'elle par un non-Juif, avant midi aussi, et vendre a fortiori ; que même celui qui travaille dans un lieu où c'est l'usage, même après midi, s'il s'y consacre et distrait son esprit du deuil, ne verra pas de bénédiction dans le gain de cet argent ; et que celui qui mange et boit, même une femme enceinte ou allaitante ou quelqu'un de faible de nature qui souffre du jeûne, ne verra pas la joie de Jérusalem, sauf un vrai malade.
