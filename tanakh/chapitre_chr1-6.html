בְּנֵ֖י לֵוִ֑י גֵּרְשֹׁ֕ם קְהָ֖ת וּמְרָרִֽי׃

וְאֵ֛לֶּה שְׁמ֥וֹת בְּֽנֵי־גֵרְשׁ֖וֹם לִבְנִ֥י וְשִׁמְעִֽי׃

וּבְנֵ֖י קְהָ֑ת עַמְרָ֣ם וְיִצְהָ֔ר וְחֶבְר֖וֹן וְעֻזִּיאֵֽל׃ {ס}        

בְּנֵ֥י מְרָרִ֖י מַחְלִ֣י וּמֻשִׁ֑י וְאֵ֛לֶּה מִשְׁפְּח֥וֹת הַלֵּוִ֖י לַאֲבֹתֵיהֶֽם׃

לְֽגֵרְשׁ֑וֹם לִבְנִ֥י בְנ֛וֹ יַ֥חַת בְּנ֖וֹ זִמָּ֥ה בְנֽוֹ׃

יוֹאָ֤ח בְּנוֹ֙ עִדּ֣וֹ בְנ֔וֹ זֶ֥רַח בְּנ֖וֹ יְאׇתְרַ֥י בְּנֽוֹ׃

בְּנֵ֖י קְהָ֑ת עַמִּינָדָ֣ב בְּנ֔וֹ קֹ֥רַח בְּנ֖וֹ אַסִּ֥יר בְּנֽוֹ׃

אֶלְקָנָ֥ה בְנ֛וֹ וְאֶבְיָסָ֥ף בְּנ֖וֹ וְאַסִּ֥יר בְּנֽוֹ׃

תַּ֤חַת בְּנוֹ֙ אוּרִיאֵ֣ל בְּנ֔וֹ עֻזִּיָּ֥ה בְנ֖וֹ וְשָׁא֥וּל בְּנֽוֹ׃

וּבְנֵי֙ אֶלְקָנָ֔ה עֲמָשַׂ֖י וַאֲחִימֽוֹת׃

אֶלְקָנָ֑ה (בנו) [בְּנֵי֙] אֶלְקָנָ֔ה צוֹפַ֥י בְּנ֖וֹ וְנַ֥חַת בְּנֽוֹ׃

אֱלִיאָ֥ב בְּנ֛וֹ יְרֹחָ֥ם בְּנ֖וֹ אֶלְקָנָ֥ה בְנֽוֹ׃

וּבְנֵ֧י שְׁמוּאֵ֛ל הַבְּכֹ֥ר וַשְׁנִ֖י וַאֲבִיָּֽה׃ {ס}        

בְּנֵ֥י מְרָרִ֖י מַחְלִ֑י לִבְנִ֥י בְנ֛וֹ שִׁמְעִ֥י בְנ֖וֹ עֻזָּ֥ה בְנֽוֹ׃

שִׁמְעָ֥א בְנ֛וֹ חַגִּיָּ֥ה בְנ֖וֹ עֲשָׂיָ֥ה בְנֽוֹ׃ {פ}

וְאֵ֗לֶּה אֲשֶׁ֨ר הֶעֱמִ֥יד דָּוִ֛יד עַל־יְדֵי־שִׁ֖יר בֵּ֣ית יְהֹוָ֑ה מִמְּנ֖וֹחַ הָאָרֽוֹן׃

וַיִּֽהְי֨וּ מְשָֽׁרְתִ֜ים לִפְנֵ֨י מִשְׁכַּ֤ן אֹֽהֶל־מוֹעֵד֙ בַּשִּׁ֔יר עַד־בְּנ֧וֹת שְׁלֹמֹ֛ה אֶת־בֵּ֥ית יְהֹוָ֖ה בִּירוּשָׁלָ֑͏ִם וַיַּעַמְד֥וּ כְמִשְׁפָּטָ֖ם עַל־עֲבוֹדָתָֽם׃

וְאֵ֥לֶּה הָעֹמְדִ֖ים וּבְנֵיהֶ֑ם מִבְּנֵי֙ הַקְּהָתִ֔י הֵימָן֙ הַמְשׁוֹרֵ֔ר בֶּן־יוֹאֵ֖ל בֶּן־שְׁמוּאֵֽל׃

בֶּן־אֶלְקָנָה֙ בֶּן־יְרֹחָ֔ם בֶּן־אֱלִיאֵ֖ל בֶּן־תּֽוֹחַ׃

בֶּן־[צוּף֙] (ציף) בֶּן־אֶלְקָנָ֔ה בֶּן־מַ֖חַת בֶּן־עֲמָשָֽׂי׃

בֶּן־אֶלְקָנָה֙ בֶּן־יוֹאֵ֔ל בֶּן־עֲזַרְיָ֖ה בֶּן־צְפַנְיָֽה׃

בֶּן־תַּ֙חַת֙ בֶּן־אַסִּ֔יר בֶּן־אֶבְיָסָ֖ף בֶּן־קֹֽרַח׃

בֶּן־יִצְהָ֣ר בֶּן־קְהָ֔ת בֶּן־לֵוִ֖י בֶּן־יִשְׂרָאֵֽל׃ {ס}        

וְאָחִ֣יו אָסָ֔ף הָעֹמֵ֖ד עַל־יְמִינ֑וֹ אָסָ֥ף בֶּן־בֶּרֶכְיָ֖הוּ בֶּן־שִׁמְעָֽא׃

בֶּן־מִיכָאֵ֥ל בֶּן־בַּעֲשֵׂיָ֖ה בֶּן־מַלְכִּיָּֽה׃

בֶּן־אֶתְנִ֥י בֶן־זֶ֖רַח בֶּן־עֲדָיָֽה׃

בֶּן־אֵיתָ֥ן בֶּן־זִמָּ֖ה בֶּן־שִׁמְעִֽי׃

בֶּן־יַ֥חַת בֶּן־גֵּרְשֹׁ֖ם בֶּן־לֵוִֽי׃ {ס}        

וּבְנֵ֧י מְרָרִ֛י אֲחֵיהֶ֖ם עַֽל־הַשְּׂמֹ֑אול אֵיתָן֙ בֶּן־קִישִׁ֔י בֶּן־עַבְדִּ֖י בֶּן־מַלּֽוּךְ׃

בֶּן־חֲשַׁבְיָ֥ה בֶן־אֲמַצְיָ֖ה בֶּן־חִלְקִיָּֽה׃

בֶּן־אַמְצִ֥י בֶן־בָּנִ֖י בֶּן־שָֽׁמֶר׃

בֶּן־מַחְלִי֙ בֶּן־מוּשִׁ֔י בֶּן־מְרָרִ֖י בֶּן־לֵוִֽי׃ {ס}        

וַאֲחֵיהֶ֖ם הַלְוִיִּ֑ם נְתוּנִ֕ים לְכׇ֨ל־עֲבוֹדַ֔ת מִשְׁכַּ֖ן בֵּ֥ית הָאֱלֹהִֽים׃

וְֽאַהֲרֹ֨ן וּבָנָ֜יו מַקְטִירִ֨ים עַל־מִזְבַּ֤ח הָעוֹלָה֙ וְעַל־מִזְבַּ֣ח הַקְּטֹ֔רֶת לְכֹ֕ל מְלֶ֖אכֶת קֹ֣דֶשׁ הַקֳּדָשִׁ֑ים וּלְכַפֵּר֙ עַל־יִשְׂרָאֵ֔ל כְּכֹל֙ אֲשֶׁ֣ר צִוָּ֔ה מֹשֶׁ֖ה עֶ֥בֶד הָאֱלֹהִֽים׃ {פ}

וְאֵ֖לֶּה בְּנֵ֣י אַהֲרֹ֑ן אֶלְעָזָ֥ר בְּנ֛וֹ פִּֽינְחָ֥ס בְּנ֖וֹ אֲבִישׁ֥וּעַ בְּנֽוֹ׃

בֻּקִּ֥י בְנ֛וֹ עֻזִּ֥י בְנ֖וֹ זְרַֽחְיָ֥ה בְנֽוֹ׃

מְרָי֥וֹת בְּנ֛וֹ אֲמַרְיָ֥ה בְנ֖וֹ אֲחִיט֥וּב בְּנֽוֹ׃

צָד֥וֹק בְּנ֖וֹ אֲחִימַ֥עַץ בְּנֽוֹ׃ {ס}        

וְאֵ֙לֶּה֙ מֽוֹשְׁבוֹתָ֔ם לְטִירוֹתָ֖ם בִּגְבוּלָ֑ם לִבְנֵ֤י אַֽהֲרֹן֙ לְמִשְׁפַּ֣חַת הַקְּהָתִ֔י כִּ֥י לָהֶ֖ם הָיָ֥ה הַגּוֹרָֽל׃

וַֽיִּתְּנ֥וּ לָהֶ֛ם אֶת־חֶבְר֖וֹן בְּאֶ֣רֶץ יְהוּדָ֑ה וְאֶת־מִגְרָשֶׁ֖יהָ סְבִיבֹתֶֽיהָ׃

וְאֶת־שְׂדֵ֥ה הָעִ֖יר וְאֶת־חֲצֵרֶ֑יהָ נָתְנ֖וּ לְכָלֵ֥ב בֶּן־יְפֻנֶּֽה׃ {ס}        

וְלִבְנֵ֣י אַהֲרֹ֗ן נָֽתְנוּ֙ אֶת־עָרֵ֣י הַמִּקְלָ֔ט אֶת־חֶבְר֥וֹן וְאֶת־לִבְנָ֖ה וְאֶת־מִגְרָשֶׁ֑יהָ וְאֶת־יַתִּ֥ר וְאֶֽת־אֶשְׁתְּמֹ֖עַ וְאֶת־מִגְרָשֶֽׁיהָ׃

וְאֶת־חִילֵז֙ וְאֶת־מִגְרָשֶׁ֔יהָ אֶת־דְּבִ֖יר וְאֶת־מִגְרָשֶֽׁיהָ׃

וְאֶת־עָשָׁן֙ וְאֶת־מִגְרָשֶׁ֔יהָ וְאֶת־בֵּ֥ית שֶׁ֖מֶשׁ וְאֶת־מִגְרָשֶֽׁיהָ׃ {ס}        

וּמִמַּטֵּ֣ה בִנְיָמִ֗ן אֶת־גֶּ֤בַע וְאֶת־מִגְרָשֶׁ֙יהָ֙ וְאֶת־עָלֶ֣מֶת וְאֶת־מִגְרָשֶׁ֔יהָ וְאֶת־עֲנָת֖וֹת וְאֶת־מִגְרָשֶׁ֑יהָ כׇּל־עָרֵיהֶ֛ם שְׁלֹשׁ־עֶשְׂרֵ֥ה עִ֖יר בְּמִשְׁפְּחוֹתֵיהֶֽם׃ {ס}        

וְלִבְנֵ֨י קְהָ֜ת הַנּֽוֹתָרִ֗ים מִמִּשְׁפַּ֣חַת הַמַּטֶּ֡ה מִֽ֠מַּחֲצִ֠ית מַטֵּ֨ה חֲצִ֧י מְנַשֶּׁ֛ה בַּגּוֹרָ֖ל עָרִ֥ים עָֽשֶׂר׃ {פ}

וְלִבְנֵ֨י גֵרְשׁ֜וֹם לְמִשְׁפְּחוֹתָ֗ם מִמַּטֵּ֣ה יִ֠שָּׂשכָ֠ר וּמִמַּטֵּ֨ה אָשֵׁ֜ר וּמִמַּטֵּ֣ה נַפְתָּלִ֗י וּמִמַּטֵּ֤ה מְנַשֶּׁה֙ בַּבָּשָׁ֔ן עָרִ֖ים שְׁלֹ֥שׁ עֶשְׂרֵֽה׃ {ס}        

לִבְנֵ֨י מְרָרִ֜י לְמִשְׁפְּחוֹתָ֗ם מִמַּטֵּ֣ה רְ֠אוּבֵ֠ן וּֽמִמַּטֵּה־גָ֞ד וּמִמַּטֵּ֤ה זְבֻלוּן֙ בַּגּוֹרָ֔ל עָרִ֖ים שְׁתֵּ֥ים עֶשְׂרֵֽה׃ {ס}        

וַיִּתְּנ֥וּ בְנֵֽי־יִשְׂרָאֵ֖ל לַלְוִיִּ֑ם אֶת־הֶעָרִ֖ים וְאֶת־מִגְרְשֵׁיהֶֽם׃

וַיִּתְּנ֣וּ בַגּוֹרָ֗ל מִמַּטֵּ֤ה בְנֵֽי־יְהוּדָה֙ וּמִמַּטֵּ֣ה בְנֵֽי־שִׁמְע֔וֹן וּמִמַּטֵּ֖ה בְּנֵ֣י בִנְיָמִ֑ן אֵ֚ת הֶעָרִ֣ים הָאֵ֔לֶּה אֲשֶׁר־יִקְרְא֥וּ אֶתְהֶ֖ם בְּשֵׁמֽוֹת׃ {ס}        

וּמִֽמִּשְׁפְּח֖וֹת בְּנֵ֣י קְהָ֑ת וַֽיְהִי֙ עָרֵ֣י גְבוּלָ֔ם מִמַּטֵּ֖ה אֶפְרָֽיִם׃

וַיִּתְּנ֨וּ לָהֶ֜ם אֶת־עָרֵ֧י הַמִּקְלָ֛ט אֶת־שְׁכֶ֥ם וְאֶת־מִגְרָשֶׁ֖יהָ בְּהַ֣ר אֶפְרָ֑יִם וְאֶת־גֶּ֖זֶר וְאֶת־מִגְרָשֶֽׁיהָ׃

וְאֶֽת־יׇקְמְעָם֙ וְאֶת־מִגְרָשֶׁ֔יהָ וְאֶת־בֵּ֥ית חוֹר֖וֹן וְאֶת־מִגְרָשֶֽׁיהָ׃

וְאֶת־אַיָּלוֹן֙ וְאֶת־מִגְרָשֶׁ֔יהָ וְאֶת־גַּת־רִמּ֖וֹן וְאֶת־מִגְרָשֶֽׁיהָ׃

וּמִֽמַּחֲצִית֙ מַטֵּ֣ה מְנַשֶּׁ֔ה אֶת־עָנֵר֙ וְאֶת־מִגְרָשֶׁ֔יהָ וְאֶת־בִּלְעָ֖ם וְאֶת־מִגְרָשֶׁ֑יהָ לְמִשְׁפַּ֥חַת לִבְנֵֽי־קְהָ֖ת הַנּוֹתָרִֽים׃ {פ}

לִבְנֵי֮ גֵּרְשׁוֹם֒ מִמִּשְׁפַּ֗חַת חֲצִי֙ מַטֵּ֣ה מְנַשֶּׁ֔ה אֶת־גּוֹלָ֥ן בַּבָּשָׁ֖ן וְאֶת־מִגְרָשֶׁ֑יהָ וְאֶת־עַשְׁתָּר֖וֹת וְאֶת־מִגְרָשֶֽׁיהָ׃

וּמִמַּטֵּ֣ה יִשָּׂשכָ֔ר אֶת־קֶ֖דֶשׁ וְאֶת־מִגְרָשֶׁ֑יהָ אֶת־דָּבְרַ֖ת וְאֶת־מִגְרָשֶֽׁיהָ׃

וְאֶת־רָאמוֹת֙ וְאֶת־מִגְרָשֶׁ֔יהָ וְאֶת־עָנֵ֖ם וְאֶת־מִגְרָשֶֽׁיהָ׃ {ס}        

וּמִמַּטֵּ֣ה אָשֵׁ֔ר אֶת־מָשָׁ֖ל וְאֶת־מִגְרָשֶׁ֑יהָ וְאֶת־עַבְדּ֖וֹן וְאֶת־מִגְרָשֶֽׁיהָ׃

וְאֶת־חוּקֹק֙ וְאֶת־מִגְרָשֶׁ֔יהָ וְאֶת־רְחֹ֖ב וְאֶת־מִגְרָשֶֽׁיהָ׃ {ס}        

וּמִמַּטֵּ֣ה נַפְתָּלִ֗י אֶת־קֶ֤דֶשׁ בַּגָּלִיל֙ וְאֶת־מִגְרָשֶׁ֔יהָ וְאֶת־חַמּ֖וֹן וְאֶת־מִגְרָשֶׁ֑יהָ וְאֶת־קִרְיָתַ֖יִם וְאֶת־מִגְרָשֶֽׁיהָ׃ {ס}        

לִבְנֵ֣י מְרָרִי֮ הַנּוֹתָרִים֒ מִמַּטֵּ֣ה זְבֻל֔וּן אֶת־רִמּוֹנ֖וֹ וְאֶת־מִגְרָשֶׁ֑יהָ אֶת־תָּב֖וֹר וְאֶת־מִגְרָשֶֽׁיהָ׃

Lévi, Aharon et la lignée sacerdotale (6:1-15)

Fils de Lévi : Guershon, Kehat et Merari. Fils de Kehat : Amram, Yitshar, Hevron et Ouziel. Enfants d'Amram : Aharon, Moshé et Myriam. Fils d'Aharon : Nadav, Avihou, Elazar et Itamar. Elazar engendra Pinhas ; Pinhas engendra Avishoua ; Avishoua engendra Bouki ; Bouki engendra Ouzi ; Ouzi engendra Zerahya ; Zerahya engendra Merayot ; Merayot engendra Amarya ; Amarya engendra Ahitouv ; Ahitouv engendra Tsadok ; Tsadok engendra Ahimaats ; Ahimaats engendra Azarya ; Azarya engendra Yohanan ; Yohanan engendra Azarya, qui exerça la prêtrise dans la maison que Shelomo bâtit à Jérusalem. Azarya engendra Amarya, Amarya engendra Ahitouv, Ahitouv engendra Tsadok, Tsadok engendra Shalloum, Shalloum engendra Hilkiya, Hilkiya engendra Azarya, Azarya engendra Seraya, Seraya engendra Yehotsadak. Yehotsadak partit en exil quand l'Éternel fit déporter Yehouda et Jérusalem par la main de Nabuchodonosor.

Les lévites et les chantres de David (6:16-48)

Fils de Lévi : Guershom, Kehat et Merari. Voici les noms des fils de Guershom : Livni et Shimeï. Fils de Kehat : Amram, Yitshar, Hevron et Ouziel. Fils de Merari : Mahli et Moushi. Voici ceux que David établit pour le service du chant dans la maison de l'Éternel, depuis que l'arche eut un lieu de repos. Ils exerçaient leur service de chant devant le tabernacle de la tente d'assignation, jusqu'à ce que Shelomo bâtisse la maison de l'Éternel à Jérusalem ; ils faisaient leur service selon leur règle. Voici ceux qui se tenaient là avec leurs fils : des fils des Kehatites, Heman le chantre, fils de Yoël, fils de Shemouel, fils d'Elkana... ; et son frère Asaf, qui se tenait à sa droite, Asaf fils de Berekhya, fils de Shimea... ; et à la gauche, les fils de Merari : Eitan, fils de Kishi, fils d'Avdi, fils de Malloukh... Leurs frères les lévites étaient chargés de tout le service du tabernacle de la maison de D.ieu.

Le service d'Aharon et les villes lévitiques (6:49-66)

Aharon et ses fils offraient sur l'autel de l'holocauste et sur l'autel de l'encens, pour toute l'œuvre du Saint des saints, afin de faire l'expiation pour Israël, selon tout ce qu'avait ordonné Moshé, serviteur de D.ieu. Voici les fils d'Aharon : Elazar, son fils ; Pinhas, son fils ; Avishoua, son fils ; Bouki, son fils ; Ouzi, son fils ; Zerahya, son fils ; Merayot, son fils ; Amarya, son fils ; Ahitouv, son fils ; Tsadok, son fils ; Ahimaats, son fils. Voici leurs habitations selon leurs enclos dans leur territoire : aux fils d'Aharon, de la famille des Kehatites, car le sort leur échut, on donna Hevron au pays de Yehouda, avec ses faubourgs alentour ; mais le territoire de la ville et ses villages, on les donna à Calev, fils de Yefouné. Aux fils d'Aharon on donna les villes de refuge : Hevron, Livna et ses faubourgs, Yatir, Eshtemoa et ses faubourgs, Hilen, Devir, Ashan et Beit-Shemesh avec leurs faubourgs ; de la tribu de Binyamin, Gueva, Alémet et Anatot avec leurs faubourgs : en tout treize villes pour leurs familles. Aux autres fils de Kehat, par le sort, dix villes de la demi-tribu de Menashé. Aux fils de Guershon, selon leurs familles, treize villes de Yissachar, d'Asher, de Naftali et de Menashé en Bashan. Aux fils de Merari, selon leurs familles, par le sort, douze villes de Reouven, de Gad et de Zevouloun. Ainsi les fils d'Israël donnèrent aux lévites ces villes avec leurs faubourgs.

Rachi (ou, à défaut, le Midrash Tehilim et le Talmud) éclaire le sens simple du texte et les enseignements des Sages.

Les chantres de David (6:16-33)

David établit Heman, Asaf et Eitan pour le service du chant ; ce sont des lévites des trois familles.

Le service d'Aharon (6:49)

Les prêtres seuls offrent les sacrifices sur les autels ; les lévites s'occupent du reste du service.

Hevron donné à Calev (6:40)

La ville est prêtre, mais le territoire revient à Calev, comme l'a décidé Yehoshoua.

Les villes de refuge (6:42)

Hevron et d'autres villes lévitiques sont aussi des villes de refuge.
