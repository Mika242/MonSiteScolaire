חַכִּים יִתְקְרֵי, וְרַבִּי לָא יִתְקְרֵי. וְאַסּוּ דְּרַבִּי עַל יָדוֹ תְּהֵא. רַבִּי וְרַבִּי נָתָן – סוֹף מִשְׁנָה. רַב אָשֵׁי וְרָבִינָא – סוֹף הוֹרָאָה.

וְסִימָנָךְ: ״עַד אָבוֹא אֶל מִקְדְּשֵׁי אֵל אָבִינָה לְאַחֲרִיתָם״.

אָמַר רַב כָּהֲנָא, אִישְׁתַּעִי לִי רַב חָמָא בַּר בְּרַתֵּיה דְּחַסָּא: רַבָּה בַּר נַחְמָנִי אַגַּב שְׁמָדָא נָח נַפְשֵׁיהּ. אֲכַלוּ בֵּיהּ קוּרְצָא בֵּי מַלְכָּא, אֲמַרוּ: אִיכָּא חַד גַּבְרָא בִּיהוּדָאֵי דְּקָא מְבַטֵּל תְּרֵיסַר אַלְפֵי גַּבְרֵי מִיִּשְׂרָאֵל יַרְחָא בְּקַיְיטָא וְיַרְחָא בְּסִתְוָא מִכְּרָגָא דְּמַלְכָּא.

שַׁדַּרוּ פְּרֵיסְתְּקָא דְמַלְכָּא בָּתְרֵיהּ וְלָא אַשְׁכְּחֵיהּ. עֲרַק וַאֲזַל מִפּוּמְבְּדִיתָא לְאַקְרָא, מֵאַקְרָא לְאַגְמָא, וּמֵאַגְמָא לְשִׁחִין, וּמִשִּׁחִין לִצְרִיפָא, וּמִצְּרִיפָא לְעֵינָא דְמַיִם, וּמֵעֵינָא דְמַיִם לְפוּמְבְּדִיתָא. בְּפוּמְבְּדִיתָא אַשְׁכְּחֵיהּ. אִיקְּלַע פְּרֵיסְתְּקָא דְמַלְכָּא לְהָהוּא אוּשְׁפִּיזָא דְּרַבָּה. קָרִיבוּ תַּכָּא קַמֵּיהּ וְאַשְׁקוּהוּ תְּרֵי כָּסֵי, וְדַלְיוּהּ לְתַכָּא מִקַּמֵּיהּ הֲדַר פַּרְצוּפֵיהּ לַאֲחוֹרֵיהּ.

אֲמַרוּ לֵיהּ: מַאי נַעֲבֵיד לֵיהּ? גַּבְרָא דְמַלְכָּא הוּא! אֲמַר לְהוּ: קָרִיבוּ תַּכָּא לְקַמֵּיהּ, וְאַשְׁקוּהּ חַד כָּסָא, וְדַלְיוּהּ לְתַכָּא מִקַּמֵּיהּ וְלִתַּסֵּי. עֲבַדוּ לֵיהּ הָכִי וְאִתַּסִּי. אֲמַר: מִידָּע יָדַעְנָא דְּגַבְרָא דְּקָא בָעֵינָא הָכָא הוּא. בְּחֵישׁ אַבָּתְרֵיהּ וְאַשְׁכְּחֵיהּ. אֲמַר: אָזֵלְינָא מֵהָא, אִי מִקְטָל קָטְלוּ לְהָהוּא גַּבְרָא – לָא מְגַלֵּינָא, וְאִי נַגֹּידֵי מְנַגְּדִין לֵיהּ – מְגַלֵּינָא.

אַתְיוּהּ לְקַמֵּיהּ, עַיְּילֵיהּ לְאִדְּרוֹנָא וְטַרְקֵיהּ לְבָבָא בְּאַנְפֵּיהּ. בְּעָא רַחֲמֵי, פְּרַק אֲשִׁיתָא עֲרַק וַאֲזַל לְאַגְמָא. הֲוָה יָתֵיב אַגִּירְדָּא דְּדִקּוּלָא וְקָא גָרֵיס. קָא מִיפַּלְגִי בִּמְתִיבְתָּא דִרְקִיעָא: אִם בַּהֶרֶת קוֹדֶמֶת לְשֵׂעָר לָבָן – טָמֵא, וְאִם שֵׂעָר לָבָן קוֹדֵם לַבַּהֶרֶת – טָהוֹר.

סָפֵק – הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹמֵר: טָהוֹר, וְכוּלְּהוּ מְתִיבְתָּא דִרְקִיעָא אָמְרִי: טָמֵא. וְאָמְרִי: מַאן נוֹכַח? נוֹכַח רַבָּה בַּר נַחְמָנִי. דְּאָמַר רַבָּה בַּר נַחְמָנִי: אֲנִי יָחִיד בִּנְגָעִים, אֲנִי יָחִיד בְּאֹהָלוֹת.

שַׁדַּרוּ שְׁלִיחָא בָּתְרֵיהּ. לָא הֲוָה מָצֵי מַלְאַךְ הַמָּוֶת לְמִקְרַב לֵיהּ, מִדְּלָא הֲוָה קָא פָסֵיק פּוּמֵּיהּ מִגִּרְסֵיהּ. אַדְּהָכִי נְשַׁב זִיקָא וַאֲוַושׁ בֵּינֵי קְנֵי, סְבַר גּוּנְדָּא דְפָרָשֵׁי הוּא. אֲמַר: תִּינַח נַפְשֵׁיהּ דְּהָהוּא גַּבְרָא, וְלָא יִמְּסַר בִּידָא דְמַלְכוּתָא.

כִּי הֲוָה קָא נָיְחָא נַפְשֵׁיהּ, אֲמַר: טָהוֹר, טָהוֹר. יָצָאת בַּת קוֹל וְאָמְרָה: אַשְׁרֶיךָ רַבָּה בַּר נַחְמָנִי שֶׁגּוּפְךָ טָהוֹר, וְיָצָאתָה נִשְׁמָתְךָ בְּטָהוֹר. נְפַל פִּתְקָא מֵרְקִיעָא בְּפוּמְבְּדִיתָא: רַבָּה בַּר נַחְמָנִי נִתְבַּקֵּשׁ בִּיְשִׁיבָה שֶׁל מַעְלָה. נְפַקוּ אַבָּיֵי וְרָבָא וְכוּלְּהוּ רַבָּנַן לְאִיעֲסוֹקֵי בֵּיהּ. לָא הֲווֹ יָדְעִי דּוּכְתֵּיהּ. אֲזַלוּ לְאַגְמָא חֲזוֹ צִפְּרֵי דִּמְטַלְּלִי וְקָיְימִי, אָמְרִי: שְׁמַע מִינַּהּ הָתָם הוּא.

סַפְדוּהּ תְּלָתָא יוֹמֵי וּתְלָתָא לֵילָוָתָא. נְפַל פִּתְקָא: כׇּל הַפּוֹרֵשׁ יְהֵא בְּנִידּוּי. סַפְדוּהּ שִׁבְעָה יוֹמֵי. נְפַל פִּתְקָא: לְכוּ לְבֵיתְכֶם לְשָׁלוֹם.

הָהוּא יוֹמָא דְּנָח נַפְשֵׁיהּ דַּלְיֵיהּ זַעְפָּא וּדְרִי לְהָהוּא טַיָּיעָא כִּי רְכִיב גַּמְלָא מֵהַאי גִּיסָא דִּנְהַר פָּפָּא וְשַׁדְיֵיהּ בְּהָךְ גִּיסָא. אֲמַר: מַאי הַאי? אָמְרִי לֵיהּ: נָח נַפְשֵׁיהּ דְּרַבָּה בַּר נַחְמָנִי. אָמַר לְפָנָיו: רִבּוֹנוֹ שֶׁל עוֹלָם! כּוּלֵּי עָלְמָא דִּידָךְ הוּא, וְרַבָּה בַּר נַחְמָנִי דִּידָךְ, אַתְּ דְּרַבָּה וְרַבָּה דִּידָךְ – אַמַּאי קָא מַחְרְבַתְּ לֵיהּ לְעָלְמָא? נָח זַעְפָּא.

רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא בַּעַל בָּשָׂר הֲוָה. יוֹמָא חַד הֲוָה חַמִּימָא לֵיהּ, הֲוָה סָלֵיק וְיָתֵיב אַשִּׁינָּא דְטוּרָא. אֲמַר לַהּ לִבְרַתֵּיהּ: בִּתִּי, הָנִיפִי עָלַי בִּמְנִיפָא, וַאֲנִי אֶתֵּן לִיךְ כִּכָּרִין דְּנֵרְדְּ. אַדְּהָכִי נְשַׁבָא זִיקָא, אֲמַר: כַּמָּה כַּכְּרִין דְּנֵרְדְּ לְמָרֵי דֵּיכִי.

הַכֹּל כְּמִנְהַג הַמְּדִינָה וְכוּ׳. הַכֹּל לְאֵתוֹיֵי מַאי? לְאֵתוֹיֵי בְּאַתְרָא דִּנְהִיגִי מִכְרַךְ רִיפְתָּא וּמִשְׁתֵּה אַנְפָּקָא. דְּאִי אָמַר לְהוּ קַדִּימוּ וְאַיְיתֵי לְכוּ, אָמְרוּ לֵיהּ: לָא כֹּל כְּמִינָךְ.

מַעֲשֶׂה בְּרַבָּן יוֹחָנָן בֶּן מַתְיָא שֶׁאָמַר לִבְנוֹ צֵא וּשְׂכוֹר וְכוּ׳. מַעֲשֶׂה לִסְתּוֹר! חַסּוֹרֵי מְחַסְּרָא וְהָכִי קָתָנֵי: וְאִם פָּסַק לָהֶם מְזוֹנוֹת

רִיבָּה לָהֶן. וּמַעֲשֶׂה נָמֵי בְּרַבִּי יוֹחָנָן בֶּן מַתְיָא שֶׁאָמַר לִבְנוֹ צֵא שְׂכוֹר לָנוּ פּוֹעֲלִים, הָלַךְ וּפָסַק לָהֶן מְזוֹנוֹת. וּכְשֶׁבָּא אֵצֶל אָבִיו, אָמַר לוֹ: בְּנִי, אֲפִילּוּ אַתָּה עוֹשֶׂה לָהֶן כִּסְעוּדַת שְׁלֹמֹה בִּשְׁעָתוֹ, לֹא יָצָאתָ יְדֵי חוֹבָתְךָ עִמָּהֶן, שֶׁהֵן בְּנֵי אַבְרָהָם יִצְחָק וְיַעֲקֹב.

לְמֵימְרָא דִּסְעוּדְתָּא דְּאַבְרָהָם אָבִינוּ עֲדִיפָא מִדִּשְׁלֹמֹה? וְהָכְתִיב: ״וַיְהִי לֶחֶם שְׁלֹמֹה לְיוֹם אֶחָד שְׁלֹשִׁים כֹּר סֹלֶת וְשִׁשִּׁים כֹּר קָמַח. עֲשָׂרָה בָּקָר בְּרִאִים וַעֲשָׂרָה בָּקָר רְעִי וּמֵאָה צֹאן לְבַד מֵאַיָּל וּצְבִי וְיַחְמוּר וּבַרְבֻּרִים אֲבוּסִים״! וְאָמַר גּוּרְיוֹן בֶּן אֲסִטְיוֹן מִשְּׁמֵיהּ דְּרַב: הַלָּלוּ לַעֲמִילָן שֶׁל טַבָּחִים. וְרַבִּי יִצְחָק אָמַר: הַלָּלוּ לְצִיקֵי קְדֵירָה.

וְאָמַר רַבִּי יִצְחָק: אֶלֶף נָשִׁים הָיוּ לִשְׁלֹמֹה, כׇּל אַחַת וְאַחַת עָשְׂתָה לוֹ בְּבֵיתָה כָּךְ. מַאי טַעְמָא – זוֹ סְבוּרָה שֶׁמָּא אֶצְלִי סוֹעֵד הַיּוֹם, וְזוֹ סְבוּרָה שֶׁמָּא אֶצְלִי סוֹעֵד הַיּוֹם. וְאִילּוּ גַּבֵּי אַבְרָהָם כְּתִיב: ״וְאֶל הַבָּקָר רָץ אַבְרָהָם וַיִּקַּח בֶּן בָּקָר רַךְ וָטוֹב״. וְאָמַר רַב יְהוּדָה אָמַר רַב: ״בֶּן בָּקָר״ – אֶחָד, ״רַךְ״ – שְׁנַיִם, ״וְטוֹב״ – שְׁלֹשָׁה!

הָתָם תְּלָתָא תּוֹרֵי לִתְלָתָא גַּבְרֵי, הָכָא לְכׇל יִשְׂרָאֵל וִיהוּדָה, שֶׁנֶּאֱמַר: ״יְהוּדָה וְיִשְׂרָאֵל רַבִּים כַּחוֹל אֲשֶׁר עַל הַיָּם״.

מַאי ״בַּרְבּוּרִים אֲבוּסִים״? אָמַר רַב: שֶׁאוֹבְסִים אוֹתָן בְּעַל כׇּרְחָן. וּשְׁמוּאֵל אָמַר: שֶׁאֲבוּסִים וְעוֹמְדִים מֵאֲלֵיהֶם. וְרַבִּי יוֹחָנָן אָמַר: מְבִיאִין תּוֹר מִמַּרְעִיתוֹ בִּדְלָא אֲנִיס, וְתַרְנְגוֹלֶת מֵאַשְׁפָּתָהּ בִּדְלָא אֲנִיסָא.

אָמַר רַבִּי יוֹחָנָן: מוּבְחָר שֶׁבַּבְּהֵמוֹת – שׁוֹר, מוּבְחָר שֶׁבָּעוֹפוֹת – תַּרְנְגוֹלֶת. אָמַר אַמֵּימָר: זָגְתָּא אוּכַּמְתָּא בֵּי בַטְנִיתָא דְּמִשְׁתַּכְחָא בֵּינֵי עַצְרֵי, דְּלָא מָצְיָא פָּסְיָא קַנְיָא.

״וְאֶל הַבָּקָר רָץ אַבְרָהָם״, אָמַר רַב יְהוּדָה אָמַר רַב: ״בֶּן בָּקָר״ – אֶחָד, ״רַךְ״ – שְׁנַיִם, ״וְטוֹב״ – שְׁלֹשָׁה. וְאֵימָא חַד, כִּדְאָמְרִי אִינָשֵׁי: רַכִּיךְ וְטָב!

אִם כֵּן, לִכְתּוֹב ״רַךְ טוֹב״, מַאי ״וָטוֹב״ – שְׁמַע מִינַּהּ לִדְרָשָׁה. אֵימָא תְּרֵי! מִדְּ״טוֹב״ לִדְרָשָׁה, ״רַךְ״ נָמֵי לִדְרָשָׁה.

מֵתִיב רַבָּה בַּר עוּלָּא, וְאִיתֵּימָא רַב הוֹשַׁעְיָא, וְאִיתֵּימָא רַב נָתָן בְּרַבִּי הוֹשַׁעְיָא: ״וַיִּתֵּן אֶל הַנַּעַר וַיְמַהֵר לַעֲשׂוֹת אֹתוֹ״, כֹּל חַד וְחַד יַהֲבֵיהּ לְנַעַר חַד. ״וַיִּקַּח חֶמְאָה וְחָלָב וּבֶן הַבָּקָר אֲשֶׁר עָשָׂה וַיִּתֵּן לִפְנֵיהֶם״, דְּקַמָּא קַמָּא דִּמְטָיא, אַיְיתִי לְקַמַּיְיהוּ.

וּלְמָה לִי תְּלָתָא? תִּסְגֵּי בְּחַד. אָמַר רַב חָנָן בַּר רָבָא: כְּדֵי לְהַאֲכִילָן שָׁלֹשׁ לְשׁוֹנוֹת בְּחַרְדָּל. אָמַר רַבִּי תַּנְחוּם בַּר חֲנִילַאי: לְעוֹלָם אַל יְשַׁנֶּה אָדָם מִן הַמִּנְהָג, שֶׁהֲרֵי מֹשֶׁה עָלָה לַמָּרוֹם וְלֹא אָכַל לֶחֶם, מַלְאֲכֵי הַשָּׁרֵת יָרְדוּ לְמַטָּה וְאָכְלוּ לֶחֶם. וְאָכְלוּ סָלְקָא דַּעְתָּךְ?! אֶלָּא אֵימָא: נִרְאוּ כְּמִי שֶׁאָכְלוּ וְשָׁתוּ.

אָמַר רַב יְהוּדָה אָמַר רַב: כׇּל מַה שֶּׁעָשָׂה אַבְרָהָם לְמַלְאֲכֵי הַשָּׁרֵת בְּעַצְמוֹ – עָשָׂה הַקָּדוֹשׁ בָּרוּךְ הוּא לְבָנָיו בְּעַצְמוֹ. וְכׇל מַה שֶּׁעָשָׂה אַבְרָהָם עַל יְדֵי שָׁלִיחַ – עָשָׂה הַקָּדוֹשׁ בָּרוּךְ הוּא לְבָנָיו עַל יְדֵי שָׁלִיחַ.

״וְאֶל הַבָּקָר רָץ אַבְרָהָם״, ״וְרוּחַ נָסַע מֵאֵת ה׳״. ״וַיִּקַּח חֶמְאָה וְחָלָב״, ״הִנְנִי מַמְטִיר לָכֶם לֶחֶם מִן הַשָּׁמָיִם״.

״וְהוּא עֹמֵד עֲלֵיהֶם תַּחַת הָעֵץ״, ״הִנְנִי עֹמֵד לְפָנֶיךָ שָּׁם עַל הַצּוּר [וְגוֹ׳]״. ״וְאַבְרָהָם הֹלֵךְ עִמָּם לְשַׁלְּחָם״, ״וַה׳ הֹלֵךְ לִפְנֵיהֶם יוֹמָם״.

״יֻקַּח נָא מְעַט מַיִם״, ״וְהִכִּיתָ בַצּוּר וְיָצְאוּ מִמֶּנּוּ מַיִם וְשָׁתָה הָעָם״.

וּפְלִיגָא דְּרַבִּי חָמָא בְּרַבִּי חֲנִינָא. דְּאָמַר רַבִּי חָמָא בְּרַבִּי חֲנִינָא, וְכֵן תָּנָא דְּבֵי רַבִּי יִשְׁמָעֵאל: בִּשְׂכַר שְׁלֹשָׁה – זָכוּ לִשְׁלֹשָׁה. בִּשְׂכַר ״חֶמְאָה וְחָלָב״ – זָכוּ לַמָּן; בִּשְׂכַר ״וְהוּא עֹמֵד עֲלֵיהֶם״ – זָכוּ לְעַמּוּד הֶעָנָן; בִּשְׂכַר ״יֻקַּח נָא מְעַט מַיִם״ – זָכוּ לִבְאֵרָהּ שֶׁל מִרְיָם.

״יֻקַּח נָא מְעַט מַיִם וְרַחֲצוּ רַגְלֵיכֶם״, אָמַר רַבִּי יַנַּאי בְּרַבִּי יִשְׁמָעֵאל, אָמְרוּ לוֹ: וְכִי בְּעַרְבִיִּים חֲשַׁדְתָּנוּ, שֶׁהֵם מִשְׁתַּחֲוִים לַאֲבַק רַגְלֵיהֶם, כְּבָר יָצָא מִמֶּנּוּ יִשְׁמָעֵאל.

״וַיֵּרָא אֵלָיו ה׳ בְּאֵלֹנֵי מַמְרֵא וְהוּא יֹשֵׁב פֶּתַח הָאֹהֶל כְּחֹם הַיּוֹם״. מַאי ״כְּחוֹם הַיּוֹם״? אָמַר רַבִּי חָמָא בְּרַבִּי חֲנִינָא: אוֹתוֹ הַיּוֹם יוֹם שְׁלִישִׁי שֶׁל מִילָה שֶׁל אַבְרָהָם הָיָה, וּבָא הַקָּדוֹשׁ בָּרוּךְ הוּא לִשְׁאוֹל בְּאַבְרָהָם, הוֹצִיא הַקָּדוֹשׁ בָּרוּךְ הוּא חַמָּה מִנַּרְתִּיקָהּ כְּדֵי שֶׁלֹּא יַטְרִיחַ אוֹתוֹ צַדִּיק בְּאוֹרְחִים.

שַׁדְּרֵיהּ לֶאֱלִיעֶזֶר לְמִיפַּק לְבָרָא. נְפַק וְלָא אַשְׁכַּח, אָמַר: לָא מְהֵימְנָא לָךְ. הַיְינוּ דְּאָמְרִי תַּמָּן: לֵית הֵימָנוּתָא בְּעַבְדֵי. נְפַק אִיהוּ חַזְיֵיהּ לְהַקָּדוֹשׁ בָּרוּךְ הוּא דְּקָאֵי אַבָּבָא. הַיְינוּ דִּכְתִיב: ״אַל נָא תַעֲבֹר מֵעַל עַבְדֶּךָ״.

כֵּיוָן דַּחֲזָא דְּקָא אָסַר וְשָׁרֵי, אֲמַר: לָאו אוֹרַח אַרְעָא לְמֵיקַם הָכָא. הַיְינוּ דִּכְתִיב: ״וַיִּשָּׂא עֵינָיו וַיַּרְא וְהִנֵּה שְׁלֹשָׁה אֲנָשִׁים נִצָּבִים עָלָיו וַיַּרְא וַיָּרׇץ לִקְרָאתָם״. מֵעִיקָּרָא אֲתוֹ קָמוּ עֲלֵיהּ, כִּי חַזְיוּהּ דַּהֲוָה לֵיהּ צַעֲרָא, אֲמַרוּ: לָאו אוֹרַח אַרְעָא לְמֵיקַם הָכָא.

מַאן נִינְהוּ שְׁלֹשָׁה אֲנָשִׁים – מִיכָאֵל וְגַבְרִיאֵל וּרְפָאֵל. מִיכָאֵל שֶׁבָּא לְבַשֵּׂר אֶת שָׂרָה, רְפָאֵל שֶׁבָּא לְרַפֵּא אֶת אַבְרָהָם, גַּבְרִיאֵל אֲזַל לְמֶהְפְּכַיהּ לִסְדוֹם. וְהָא כְּתִיב: ״וַיָּבֹאוּ שְׁנֵי הַמַּלְאָכִים סְדֹמָה בָּעֶרֶב״! דַּאֲזַל מִיכָאֵל בַּהֲדֵיהּ לְשֵׁזְבֵיהּ לְלוֹט. דַּיְקָא נָמֵי, דִּכְתִיב: ״וַיַּהֲפֹךְ אֶת הֶעָרִים הָאֵל״, וְלָא כְּתִיב ״וַיַּהַפְכוּ״ – שְׁמַע מִינַּהּ.

מַאי שְׁנָא לְגַבֵּי אַבְרָהָם, דִּכְתִיב: ״כֵּן תַּעֲשֶׂה כַּאֲשֶׁר דִּבַּרְתָּ״, וּמַאי שְׁנָא לְגַבֵּי לוֹט, דִּכְתִיב:

(Début de 86a : fin de l'énumération des grands hommes.) « Il sera appelé 'hakim (sage médecin), mais pas Rabbi ; et la guérison de Rabbi sera par lui. » J'ai aussi vu écrit là-bas : Rabbi (Yehouda HaNassi) et Rabbi Natan sont la fin de la Michna ; Rav Achi et Ravina sont la fin de l'enseignement (hora'a).

Ton moyen mnémotechnique : « Jusqu'à ce que j'entre dans les sanctuaires (mikdechei) de D.ieu, que je comprenne (avina) leur fin » (Psaumes 73:17) : mikdechei pour Rav Achi, avina pour Ravina, « leur fin » pour la fin de l'enseignement.

La mort de Rabba bar Na'hmani. Rav Kahana : Rav 'Hama, fils de la fille de 'Hasa, m'a raconté que Rabba bar Na'hmani est mort par crainte d'un décret. On l'avait dénoncé à la cour du roi : « Il y a un homme chez les Juifs qui dispense douze mille hommes d'Israël de l'impôt du roi, un mois en été et un mois en hiver. » Le roi envoya un messager à sa recherche. Rabba s'enfuit de Poumbedita à Akra, d'Akra à Agma, d'Agma à Chi'hin, de Chi'hin à Tserifa, de Tserifa à Eïn Demayim, et revint à Poumbedita. Le messager se trouva loger dans la même auberge. On lui servit deux coupes, puis on retira le plateau : son visage se retourna en arrière. Les gens de l'auberge dirent à Rabba : « Que faire ? C'est un homme du roi. » Il répondit : « Mettez-lui un plateau, donnez-lui une seule coupe, et retirez-le : il guérira. » Ainsi fut fait. Le messager comprit alors que celui qu'il cherchait était là. On le chercha, et il promit de ne pas le dénoncer. On amena Rabba ; on l'enferma dans une petite pièce ; il pria, le mur s'effondra, et il s'enfuit vers un marais, où il s'assit sur le tronc d'un palmier coupé pour étudier seul.

À ce moment, dans l'académie céleste, on discutait une loi des lésions de tsara'at : si la tache claire (baheret) précède le poil blanc, il est impur ; si le poil blanc précède, il est pur ; en cas de doute sur ce qui précéda, le Saint, béni soit-Il, dit : « Pur », et toute l'académie : « Impur ». Ils dirent : « Qui tranchera ? Rabba bar Na'hmani », car il disait : « Je suis unique en matière de lésions et unique en matière d'impureté des tentes. » On envoya un messager pour prendre son âme, mais l'ange de la mort ne pouvait l'approcher, car sa bouche ne cessait d'étudier. Un vent souffla entre les branches ; Rabba crut à une troupe de cavaliers venue l'arrêter et dit : « Que cet homme meure plutôt que d'être livré au pouvoir ! » Son âme fut alors prise. En mourant, il dit : « Pur, pur ! » Une voix céleste : « Heureux es-tu, Rabba bar Na'hmani, dont le corps est pur et dont l'âme est sortie avec le mot "pur" ! » Un billet tomba du ciel à Poumbedita : « Rabba bar Na'hmani a été appelé à l'académie céleste. » Abayé, Rava et les Sages sortirent pour l'enterrer, mais ne trouvèrent pas où il était. Ils allèrent au marais et virent des oiseaux qui faisaient de l'ombre au-dessus d'un endroit : « Il est là. » Ils le pleurèrent trois jours et trois nuits ; un billet tomba : « Celui qui s'écarte du deuil sera excommunié. » Ils le pleurèrent sept jours ; un autre billet : « Allez en paix à vos maisons. »

Le jour de sa mort, une tempête souleva un marchand arabe sur son chameau et le jeta de l'autre côté du fleuve Papa. Il demanda : « Qu'est-ce que c'est ? » On lui répondit : « Rabba bar Na'hmani est mort. » Il dit : « Maître du monde ! Le monde entier est à Toi, et Rabba aussi est à Toi. Toi à Rabba et Rabba à Toi : pourquoi détruire le monde ? » La tempête s'apaisa.

Rabbi Chimon ben 'Halafta était corpulent. Un jour de chaleur, il s'assit sur un rocher de montagne et dit à sa fille : « Ma fille, évente-moi, et je te donnerai des bottes de nard. » Un vent se leva et il dit : « Combien de bottes de nard dois-je aux maîtres de ce vent ! »

Retour à la michna (83a). « Tout est selon l'usage de la province. » Que vient ajouter « tout » ? Un lieu où l'usage est que les ouvriers mangent du pain et boivent un quart de vin (anpaka) : si l'employeur dit « levez-vous tôt et je vous l'apporterai » pour ne pas perdre de travail, ils peuvent répondre : « Tu ne peux pas nous y obliger. »

L'épisode de Rabbi Yo'hanan ben Matya. Il dit à son fils : « Va embaucher des ouvriers. » Le fils s'engagea à les nourrir. La Guemara demande : après « tout est selon l'usage de la province », comment citer cet épisode ? La michna est incomplète : si l'employeur s'est engagé à les nourrir, il a ajouté à son obligation, car s'il n'avait voulu donner que l'usage, il n'aurait rien précisé. Rabbi Yo'hanan ben Matya dit à son fils : « Même si tu leur préparais un banquet comme celui de Salomon, tu ne t'acquitterais pas, car ils sont des fils d'Abraham, Isaac et Jacob. »

La Guemara demande : le banquet d'Abraham était-il donc meilleur que celui de Salomon ? Il est écrit : « Et la provision de Salomon pour un jour : trente kor de fleur de farine et soixante kor de farine, dix bœufs gras, vingt bœufs de pâturage et cent moutons, sans compter cerfs, gazelles, daims et volailles engraissées » (I Rois 5:2-3). Guryon ben Astyon, au nom de Rav : la farine servait aux pâtes des cuisiniers placées sur les marmites pour absorber la vapeur. Rabbi Yitzhak : à un plat de viande avec farine et vin. Rabbi Yitzhak dit aussi : Salomon avait mille femmes, et chacune lui préparait chez elle un festin, pensant : « Peut-être mangera-t-il chez moi aujourd'hui. »

Mais pour Abraham : « Et Abraham courut vers le gros bétail et prit un veau tendre et bon » (Genèse 18:7). Rav Yehouda au nom de Rav : « un veau » = un ; « tendre » = un autre ; « et bon » = un autre : trois veaux, bien moins que Salomon ! Réponse : là, trois bœufs pour trois personnes ; ici, les femmes de Salomon préparaient pour tout Israël et Juda, « nombreux comme le sable de la mer, mangeant, buvant et se réjouissant » (I Rois 4:20) : le banquet d'Abraham était donc proportionnellement supérieur.

« Volailles engraissées » (avoussim) : Rav : on les gave de force ; Chemouel : elles sont naturellement grasses ; Rabbi Yo'hanan : on apportait un bœuf qui n'avait jamais travaillé et une poule qui n'avait jamais pondu. Rabbi Yo'hanan : le meilleur du gros bétail est le bœuf, le meilleur de la volaille est la poule ; Ameimar : une poule noire engraissée, trouvée près des pressoirs, qui mange tant de pépins qu'elle ne peut avancer d'une canne.

Les trois veaux. On objecte : pourquoi ne pas dire qu'il n'y avait qu'un veau, « tendre et bon » étant simplement une description ? Réponse : alors la Torah aurait écrit « tendre, bon » ; « et bon » ajoute, donc le verset est écrit pour une interprétation. Mais ne peut-on dire deux ? De même que « bon » ajoute, « tendre » ajoute aussi. Rabba bar Oulla (ou Rav Hochaya, ou Rav Natan fils de Rabbi Hochaya) objecte : « Et il le donna à l'adolescent et se hâta de le préparer » (18:7) : « le » est au singulier. Réponse : chaque veau fut donné à un serviteur. Et « le veau qu'il avait préparé » (18:8) ? Chaque veau, à mesure qu'il était prêt, fut placé devant eux.

§ The Gemara relates another story discussing the greatness of the Sages. Rav Kahana said: Rav Ḥama, son of the daughter of Ḥasa, told me that Rabba bar Naḥmani died due to the fear of a decree of religious persecution. The Gemara explains: His enemies accused him [akhalu beih kurtza] of disloyalty in the king’s palace, as they said: There is one man from among the Jews who exempts twelve thousand Jewish men from the king’s head tax two months a year, one month in the summer and one month in the winter. Since many people would study in Rabba’s study hall during the months of Adar and Elul, he was being blamed for preventing those people from working during those months.

They sent a messenger [peristaka] of the king after him, but he was not able to find him. Rabba bar Naḥmani fled and went from Pumbedita to Akra, from Akra to Agma, from Agma to Shiḥin, from Shiḥin to Tzerifa, from Tzerifa to Eina Demayim, and from Eina Demayim back to Pumbedita. Ultimately, he was found in Pumbedita, as the king’s messenger arrived by chance at that same inn where Rabba bar Naḥmani was hiding. The inn attendants placed a tray before the messenger and gave him two cups to drink. They then removed the tray from before him and his face was miraculously turned backward.

The attendants said to Rabba bar Naḥmani: What should we do with him? He is the king’s man, and we cannot leave him like this. Rabba bar Naḥmani said to them: Place a tray before him and give him one cup to drink, and then remove the tray from before him and he will be healed. They did this, and he was healed. The messenger said: I am certain that the man I seek is here, as this unnatural event must have befallen me on his account. He searched for Rabba bar Naḥmani and found out where he was. The messenger said that they should tell Rabba bar Naḥmani: I will leave this inn and will not disclose your location. Even if they will kill that man, i.e., me, I will not disclose your location. But if they will beat him, me, I will disclose your whereabouts, as I cannot bear being tortured.

With that guarantee, they brought Rabba bar Naḥmani before the messenger. They took him into a small vestibule [le’idrona] and closed the door before him. Rabba bar Naḥmani prayed for mercy, and the wall crumbled. He fled and went to hide in a swamp. He was sitting on the stump of a palm tree and studying Torah alone. At that moment, the Sages in the heavenly academy were disagreeing with regard to a halakha of leprosy. In general, a leprous spot includes two signs of impurity, a bright white spot and a white hair. The basic halakha is that if the snow-white leprous sore [baheret] preceded the white hair then the afflicted person is ritually impure, but if the white hair preceded the baheret, he is pure.

The heavenly debate concerned a case of uncertainty as to which came first, the spot or the hair. The Holy One, Blessed be He, says: The individual is pure, but every other member of the heavenly academy says: He is impure. And they said: Who can arbitrate in this dispute? They agreed that Rabba bar Naḥmani should arbitrate, as Rabba bar Naḥmani once said: I am preeminent in the halakhot of leprosy and I am preeminent in the halakhot of ritual impurity imparted by tents.

They sent a messenger from heaven after him to take his soul up to the heavenly academy, but the Angel of Death was unable to approach Rabba bar Naḥmani, as his mouth did not cease from his Torah study. In the meantime, a wind blew and howled between the branches. Rabba bar Naḥmani thought that the noise was due to an infantry battalion [gunda] about to capture him. He said: Let that man, i.e., me, die and not be given over to the hands of the government. The Angel of Death was therefore able to take his soul.

As he was dying, he said in response to the dispute in heaven: It is pure; it is pure. A Divine Voice emerged from heaven and said: Happy are you, Rabba bar Naḥmani, as your body is pure and your soul left you with the word: Pure. A note [pitka] fell from heaven and landed in the academy of Pumbedita. The note read: Rabba bar Naḥmani was summoned to the heavenly academy, i.e., he has died. Abaye and Rava and all of the other Rabbis went out to tend to his burial; however, they did not know the location of his body. They went to the swamp and saw birds forming a shade and hovering over a certain spot. The Rabbis said: We can conclude from this that he is there.

The Rabbis lamented him for three days and three nights. A note fell from heaven, upon which was written: Anyone who removes himself from the lamentations shall be ostracized. Accordingly, they lamented him for seven days. Another note fell from heaven, stating: Go to your homes in peace.

On that day when Rabba bar Naḥmani died, a hurricane lifted a certain Arab [taya’a] merchant while he was riding his camel. The hurricane carried him from one side of the Pappa River and threw him onto the other side. He said: What is this? Those present said to him: Rabba bar Naḥmani has died. He said before God: Master of the Universe! The entire world is Yours and Rabba bar Naḥmani is also Yours. You are to Rabba and Rabba is to You, i.e., you are beloved to each other. If so, why are You destroying the world on his account? The storm subsided.

The Gemara answers: There, with regard to Abraham, he prepared three oxen for three people, whereas here, in the case of Solomon, his wives would prepare a feast for the entire realms of Israel and Judah, as it is stated: “Judah and Israel were many, as the sand which is by the sea in multitude, eating and drinking and making merry” (I Kings 4:20). The Gemara asks: But why not say that the verse is referring to only one calf, as people say when describing a single item that it is tender and good?

The Gemara answers: If so, let the verse write: Tender, good. What is the significance of the term “and good,” which indicates an addition? Conclude from this that the verse is stated for the purpose of an exposition and is referring to more than one calf. The Gemara challenges: But one can still say there were only two calves. The Gemara answers: From the fact that the word “good” is written for an exposition, to include an additional calf, it may be inferred that the term “tender” is also written for an exposition and indicates yet another calf.

The Gemara responds: The verse means that as each calf arrived prepared, he brought it before them, and he did not serve all three calves at once.

Rachi sur Rabba bar Na'hmani : il était l'adversaire de Rav Yossef dans la discussion. Il est mort « par crainte du pouvoir ». Les gens se rassemblaient chez lui en Nissan et Tichri pour entendre les enseignements de Pessa'h et des fêtes, et les envoyés du roi ne les trouvaient pas chez eux au moment de percevoir l'impôt par tête. Sur « deux coupes » : les démons préposés aux nombres pairs nuisent à celui qui boit par paires, et les gens de la maison ne l'avaient pas voulu. Sur « le mur s'effondra » : la cloison tomba. Sur « le tronc coupé du palmier » : sur un tronc de palmier coupé. Sur « un régiment de cavaliers » : une troupe qui le cherche.

Rachi sur la vente : « anpaka » : récipient d'un quart de log. Sur « le festin de Salomon » : la'amilan sont des pâtes de cuisiniers posées sur la marmite pour aspirer l'écume ; tsiké kedera est un genre de plat (aroud en langue étrangère) ; barburim : des poules ; avoussim : on les gave de force ; tor : bœuf ; zagta : poule ; pitanita : « celle qui engraisse à la maison », une poule corpulente.

Rachi sur « trois langues avec de la moutarde » : mets de rois et de princes. Sur « pour la manne, le puits et la colonne » : trois dons du désert donnés par Moïse et Aharon. Sur « la colonne de nuée » : elle se tient sur eux toujours, comme il est dit « ta nuée se tient sur eux ». Sur « vous laverez vos pieds et vous vous reposerez sous l'arbre » : pour ne pas faire entrer d'idolâtrie dans sa maison, car les Arabes se prosternent devant la poussière de leurs pieds. Sur « il a fait sortir le soleil de son fourreau » : pour faire briller le soleil avec force et empêcher les voyageurs d'être là. Sur « un seul est allé renverser Sodome » : la formule est au singulier.

Tossafot sur Rabbi Chimon ben 'Halafta : il est cité ici parce que le récit précédent traitait de personnes corpulentes (Rabbi Elazar fils de Rabbi Chimon). Sur « levez-vous tôt » : ils ne voulaient pas travailler avant le lever du soleil. Sur « ils paraissaient manger » : le Séder Eliahou Rabba dit au contraire qu'ils mangèrent vraiment par respect pour Abraham, ce qui diffère de cette Guemara. Sur « le Saint, béni soit-Il, debout à la porte » : le verset « ne passe pas outre » suggère que la Présence voulait quitter Abraham, et le traité Chabbat (127a) en déduit que l'accueil des invités est plus grand que l'accueil de la Présence divine. Sur « Mikhaël venu sauver Loth » : on objecte qu'un ange ne fait pas deux missions (Bereshit Rabba 50) ; réponse : Mikhaël n'avait qu'une mission à Sodome, annoncer la naissance fut une autre mission, déjà accomplie.

Ce daf fonde plusieurs règles.

La visite des malades

Le Saint, béni soit-Il, visite Abraham le troisième jour après sa circoncision : c'est le modèle de la mitsva de visiter les malades. Voir le Choulhan Aroukh, Yoré Dé'a 335, ainsi que le daf 127 de Chabbat sur l'accueil des invités.

L'accueil des invités et l'usage du lieu

L'hospitalité d'Abraham, qui fait passer l'accueil des invités avant la Présence divine (Tossafot), est étudiée dans la fiche de la paracha Vayera. La règle « tout est selon l'usage de la province » pour le repas des ouvriers relève des lois des employeurs dans le Choulhan Aroukh ('Hochen Michpat).

Pour toute situation concrète, il faut consulter un rabbin compétent.
