בְּהֶשֵּׂג יָד כְּתִיב.

אָמַר רַב נַחְמָן: הֲלָכָה כְּדִבְרֵי חֲכָמִים, וְכַמָּה? אָמַר רַב יוֹסֵף: אֲפִילּוּ פַּטְרוֹזָא בַּר דַּנְקָא, אָמַר רָבָא: אַף אֲנַן נָמֵי תְּנֵינָא גָּדוֹל וְקָטָן.

פְּשִׁיטָא! מַהוּ דְּתֵימָא כּוּלֵּי הַאי לָא? אִי נָמֵי, פַּטְרוֹזָא לָא? קָא מַשְׁמַע לַן.

רַבִּי יְהוּדָה נְשִׂיאָה הֲוָה לֵיהּ פֶּטֶר חֲמוֹר, שַׁדְּרֵיהּ לְקַמֵּיהּ דְּרַבִּי טַרְפוֹן, אֲמַר לֵיהּ: כַּמָּה בָּעֵינָא לְמִיתַּב לְכֹהֵן? אֲמַר לֵיהּ: הֲרֵי אָמְרוּ: עַיִן יָפָה בְּסֶלַע, עַיִן רָעָה בְּשֶׁקֶל, בֵּינוֹנִית בְּרִגְיָא.

אָמַר רָבָא: הִלְכְתָא בְּרִגְיָא. וְכַמָּה? תְּלָתָא זוּזֵי. רְגִיל הָכָא וְרָגִיל הָכָא.

קַשְׁיָא הִלְכְתָא אַהִלְכְתָא! לָא קַשְׁיָא, כָּאן — בְּבָא לִימָּלֵךְ, כָּאן — בְּעוֹשֶׂה מֵעַצְמוֹ.

אָמַר רַבִּי יִצְחָק אָמַר רֵישׁ לָקִישׁ: מִי שֶׁיֵּשׁ לוֹ פֶּטֶר חֲמוֹר וְאֵין לוֹ שֶׂה לִפְדּוֹתוֹ — פּוֹדֵהוּ בְּשׇׁוְיוֹ. לְמַאן? אִילֵימָא לְרַבִּי יְהוּדָה — הָא אָמַר: הִקְפִּידָה עָלָיו תּוֹרָה בְּשֶׂה! אֶלָּא לְרַבִּי שִׁמְעוֹן.

רַב אַחָא מַתְנִי הָכִי, רָבִינָא קַשְׁיָא לֵיהּ: רַבִּי יְהוּדָה וְרַבִּי שִׁמְעוֹן — הֲלָכָה כְּרַבִּי יְהוּדָה!

וּסְתַם לַן תַּנָּא כְּרַבִּי יְהוּדָה, וְאַתְּ אָמְרַתְּ הֲלָכָה כְּרַבִּי שִׁמְעוֹן?!

אֶלָּא, אֲפִילּוּ תֵּימָא רַבִּי יְהוּדָה, לֹא יְהֵא חָמוּר מִן הַהֶקְדֵּשׁ, וְלֹא אָמְרָה תּוֹרָה ״בְּשֶׂה״ לְהַחְמִיר עָלָיו אֶלָּא לְהָקֵל עָלָיו. רַב נַחְמָן בְּרֵיהּ דְּרַב יוֹסֵף פָּרֵיק לֵיהּ בְּשִׁילְקֵי בְּשׇׁוְיוֹ.

אָמַר רַב שֵׁיזְבִי אָמַר רַב הוּנָא: הַפּוֹדֶה פֶּטֶר חֲמוֹר שֶׁל חֲבֵירוֹ, פִּדְיוֹנוֹ פָּדוּי. אִיבַּעְיָא לְהוּ: פִּדְיוֹנוֹ לַפּוֹדֶה, אוֹ דִלְמָא פִּדְיוֹנוֹ לַבְּעָלִים?

אַלִּיבָּא דְּרַבִּי שִׁמְעוֹן לָא תִּיבְּעֵי לָךְ, כֵּיוָן דְּאָמַר מוּתָּר בַּהֲנָאָה — מָמוֹנָא דִבְעָלִים הוּא, כִּי תִּיבְּעֵי אַלִּיבָּא דְּרַבִּי יְהוּדָה, דְּאָמַר אָסוּר בַּהֲנָאָה.

לְהֶקְדֵּשׁ מְדַמֵּי לֵיהּ, וְרַחֲמָנָא אָמַר: ״וְנָתַן הַכֶּסֶף וְקָם לוֹ״, אוֹ דִלְמָא, כֵּיוָן דְּקָנֵי לְהוּ בְּבֵינֵי וּבֵינֵי, לָא דָּמֵי לְהֶקְדֵּשׁ?

אָמַר רַב נַחְמָן, תָּא שְׁמַע: הַגּוֹנֵב פֶּטֶר חֲמוֹר שֶׁל חֲבֵירוֹ — מְשַׁלֵּם תַּשְׁלוּמֵי כֶפֶל לַבְּעָלִים, וְאַף עַל פִּי שֶׁאֵין לוֹ עַכְשָׁיו — יֵשׁ לוֹ לְאַחַר מִכָּאן.

מַנִּי? אִילֵימָא רַבִּי שִׁמְעוֹן, אַמַּאי אֵין לוֹ עַכְשָׁיו? אֶלָּא פְּשִׁיטָא רַבִּי יְהוּדָה.

וְאִי סָלְקָא דַעְתָּךְ לְהֶקְדֵּשׁ מְדַמֵּינַן לֵיהּ, ״וְגֻנַּב מִבֵּית הָאִישׁ״ אָמַר רַחֲמָנָא, וְלֹא מִבֵּית הֶקְדֵּשׁ! וְתוּ לָא מִידֵּי.

אַחַת בִּיכְּרָה וְאַחַת שֶׁלֹּא בִּיכְּרָה [כּוּ׳]. תָּנוּ רַבָּנַן: כֵּיצַד אָמְרוּ נִכְנַס לְדִיר לְהִתְעַשֵּׂר?

אִי אַתָּה יָכוֹל לוֹמַר בְּבָא לְיַד כֹּהֵן, שֶׁהֲרֵי שָׁנִינוּ: הַלָּקוּחַ וְשֶׁנִּיתַּן לוֹ בְּמַתָּנָה פָּטוּר מִמַּעְשַׂר בְּהֵמָה, אֶלָּא בְּיִשְׂרָאֵל שֶׁהָיוּ לוֹ עֲשָׂרָה סְפֵק פִּטְרֵי חֲמוֹרִים בְּתוֹךְ בֵּיתוֹ, שֶׁמַּפְרִישׁ עֲלֵיהֶן עֲשָׂרָה שֵׂיִין, וּמְעַשְּׂרָן, וְהֵן שֶׁלּוֹ.

מְסַיַּיע לֵיהּ לְרַב נַחְמָן, דְּאָמַר רַב נַחְמָן אָמַר רַבָּה בַּר אֲבוּהּ: יִשְׂרָאֵל שֶׁהָיוּ לוֹ עֲשָׂרָה סְפֵק פִּטְרֵי חֲמוֹרִים — מַפְרִישׁ עֲלֵיהֶן עֲשָׂרָה שֵׂיִין, וּמְעַשְּׂרָן, וְהֵן שֶׁלּוֹ.

וְאָמַר רַב נַחְמָן אָמַר רַבָּה בַּר אֲבוּהּ: יִשְׂרָאֵל שֶׁהָיוּ לוֹ עֲשָׂרָה פִּטְרֵי חֲמוֹרִים וַדָּאִין בְּתוֹךְ בֵּיתוֹ, שֶׁנָּפְלוּ מִבֵּית אֲבִי אִמּוֹ כֹּהֵן, וְאוֹתוֹ אֲבִי אִמּוֹ כֹּהֵן נָפְלוּ לוֹ מִבֵּית אֲבִי אִמּוֹ יִשְׂרָאֵל.

מַפְרִישׁ עֲלֵיהֶן עֲשָׂרָה שֵׂיִין, וּמְעַשְּׂרָן, וְהֵן שֶׁלּוֹ.

וְאָמַר רַב נַחְמָן אָמַר רַבָּה בַּר אֲבוּהּ: יִשְׂרָאֵל שֶׁהָיוּ לוֹ טְבָלִים מְמוֹרָחִין בְּתוֹךְ בֵּיתוֹ, שֶׁנָּפְלוּ מִבֵּית אֲבִי אִמּוֹ כֹּהֵן, וְאוֹתוֹ אֲבִי אִמּוֹ כֹּהֵן נָפְלוּ לוֹ מִבֵּית אֲבִי אִמּוֹ יִשְׂרָאֵל — מְעַשְּׂרָן וְהֵן שֶׁלּוֹ.

וּצְרִיכָא, דְּאִי אַשְׁמְעִינַן קַמַּיְיתָא — מִשּׁוּם דְּקָא מַפְרַישׁ וְקָאֵי.

אֲבָל הָכָא — מַתָּנוֹת שֶׁלֹּא הוּרְמוּ כְּמִי שֶׁלֹּא הוּרְמוּ דָּמְיָין, אֵימָא לָא.

וְאִי אַשְׁמְועִינַן הָכָא — דְּאֶפְשָׁר לְעַשּׂוֹרֵי מִינֵּיהּ וּבֵיהּ, דְּהָא מַנַּח, אֲבָל הָתָם — כֵּיוָן דְּשֶׂה מֵעָלְמָא בָּעֵי אֵתוֹיֵי, וּמַפְרַישׁ וְקָאֵי, אֵימָא לָא, צְרִיכָא.

אָמַר רַבִּי שְׁמוּאֵל בַּר נָתָן אָמַר רַבִּי חֲנִינָא: הַלּוֹקֵחַ טְבָלִים

מְמוֹרָחִין מִן הַגּוֹי, מְעַשְּׂרָן וְהֵן שֶׁלּוֹ.

דְּמָרְחִינְהוּ מַאן? אִילֵימָא דְּמָרְחִינְהוּ גּוֹי — ״דְּגָנְךָ״ אָמַר רַחֲמָנָא, וְלֹא דְּגַן גּוֹי!

אֶלָּא דְּמָרְחִינְהוּ יִשְׂרָאֵל מֵרְשׁוּת גּוֹי. מְעַשְּׂרָן, דְּאֵין קִנְיָן לְגוֹי בְּאֶרֶץ יִשְׂרָאֵל לְהַפְקִיעַ מִיַּד מַעֲשֵׂר, וְהֵן שֶׁלּוֹ — דְּאָמַר לֵיהּ: קָאָתֵינָא מִכֹּחַ גַּבְרָא דְּלָא מָצֵית (אִישְׁתַּעֵית) [אִישְׁתַּעוֹיֵי] דִּינָא בַּהֲדֵיהּ.

תְּנַן הָתָם: הַמַּפְקִיד פֵּירוֹתָיו אֵצֶל הַכּוּתִי וְאֵצֶל עַם הָאָרֶץ, בְּחֶזְקָתָן לַמַּעֲשֵׂר וְלַשְּׁבִיעִית.

אֵצֶל הַגּוֹי — כְּפֵירוֹתָיו, רַבִּי שִׁמְעוֹן אוֹמֵר: דְּמַאי.

אָמַר רַבִּי אֶלְעָזָר: לְהַפְרִישׁ — כּוּלֵּי עָלְמָא לָא פְּלִיגִי, כִּי פְּלִיגִי — לִיתְּנָן לְכֹהֵן. תַּנָּא קַמָּא סָבַר: וַדַּאי חַלְּפִינְהוּ, וּבָעֵי מִיתְּנַנְהוּ לְכֹהֵן, וְרַבִּי שִׁמְעוֹן סָבַר: דְּמַאי.

יָתֵיב רַב דִּימִי וְקָאָמַר לְהָא שְׁמַעְתָּא, אֲמַר לֵיהּ אַבָּיֵי: טַעְמָא דִּמְסַפְּקָא לַן אִי חַלְּפִינְהוּ אִי לָא חַלְּפִינְהוּ, הָא וַדַּאי חַלְּפִינְהוּ — דְּכוּלֵּי עָלְמָא בָּעֵי לְמִיתְּבִינְהוּ לְכֹהֵן, וְהָאָמַר רַבִּי שְׁמוּאֵל אָמַר רַבִּי חֲנִינָא: הַלּוֹקֵחַ טְבָלִים מִן הַגּוֹי מְמוֹרָחִין — מְעַשְּׂרָן וְהֵן שֶׁלּוֹ!

דִּלְמָא כָּאן בִּתְרוּמָה גְּדוֹלָה, כָּאן בִּתְרוּמַת מַעֲשֵׂר!

אַזְכַּרְתַּן מִילְּתָא דְּאָמַר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי: מִנַּיִן לַלּוֹקֵחַ טְבָלִים מְמוֹרָחִין מִן הַגּוֹי שֶׁהוּא פָּטוּר מִתְּרוּמַת מַעֲשֵׂר? שֶׁנֶּאֱמַר: ״וְאֶל הַלְוִיִּם תְּדַבֵּר וְאָמַרְתָּ אֲלֵיהֶם כִּי תִקְחוּ מֵאֵת בְּנֵי יִשְׂרָאֵל״, טְבָלִים שֶׁאַתָּה לוֹקֵחַ מִבְּנֵי יִשְׂרָאֵל — אַתָּה מַפְרִישׁ מֵהֶן תְּרוּמַת מַעֲשֵׂר וְנוֹתְנָהּ לַכֹּהֵן, טְבָלִים שֶׁאַתָּה לוֹקֵחַ מִן הַגּוֹי — אִי אַתָּה מַפְרִישׁ מֵהֶן תְּרוּמַת מַעֲשֵׂר וְנוֹתְנָהּ לַכֹּהֵן.

וְאִם מֵת נֶהֱנִים בּוֹ. דְּמִית הֵיכָא? אִילֵּימָא דְּמִית בֵּי כֹהֵן, וְנֶהֱנֶה בּוֹ כֹּהֵן — פְּשִׁיטָא, מָמוֹנָא דִּידֵיהּ הוּא! אֶלָּא, דְּמִית בֵּי בְעָלִים וְנֶהֱנֶה בּוֹ כֹּהֵן — הָא נָמֵי פְּשִׁיטָא!

סָלְקָא דַעְתָּךְ אָמֵינָא, כֹּל כַּמָּה דְּלָא מְטָא לִידֵיהּ לָא זְכָה בֵּיהּ, קָא מַשְׁמַע לַן דְּמֵעִידָּנָא דְּאַפְרְשֵׁיהּ, בִּרְשׁוּתֵיהּ דְּכֹהֵן קָאֵי.

Le verset est écrit concernant l'accessibilité financière, c'est-à-dire un cas où l'on fait vœu de donner l'évaluation d'un individu au Temple mais n'a pas les fonds suffisants pour accomplir ce vœu — dans un tel cas, il doit payer au moins un shekel. Cela ne s'applique pas à d'autres matières.

Rav Nahman dit : la halakha suit l'opinion des Sages, qui permettent de racheter un âne premier-né avec un agneau valant même moins d'un shekel. Et combien l'agneau doit-il valoir au minimum ? Rav Yossef dit : même un agneau maigre valant seulement une maa peut être utilisé. Rava dit : nous l'avons aussi appris dans la Michna (9a), qui mentionne un agneau grand ou petit, ce qui soutient l'opinion de Rav Yossef.

La Guemara demande : n'est-ce pas évident qu'un agneau maigre valant une seule maa peut être utilisé, puisque la baraïta énonce que l'agneau peut valoir n'importe quel montant ? La Guemara répond : de peur que tu ne dises que si l'agneau a si peu de valeur, on ne pourrait pas racheter avec lui, ou encore, qu'un agneau maigre ne conviendrait pas en raison de son état physique, Rav Yossef nous enseigne qu'on peut néanmoins racheter l'âne premier-né même avec un tel agneau.

Rabbi Yehouda Nessia avait un âne premier-né. Il l'envoya devant Rabbi Tarfon, et lui dit : combien dois-je donner au cohen pour son rachat ? Il lui dit : n'a-t-on pas dit : celui qui a l'œil généreux rachète son âne premier-né avec un agneau valant un séla, celui qui a l'œil mesquin le rachète avec un agneau valant un shekel, et celui du milieu le rachète avec un agneau valant un ragya ?

Rava dit : la halakha suit l'opinion exigeant un ragya. Et combien vaut un ragya ? Trois dinars, appelé ainsi car il est habitué ici et habitué là, c'est-à-dire que sa valeur se situe entre celle d'un séla (quatre dinars) et celle d'un shekel (deux dinars).

La Guemara commente : cette halakha contredit l'autre halakha, selon laquelle on peut donner un agneau de n'importe quelle valeur ! La Guemara répond : ce n'est pas contradictoire : ici, il s'agit de celui qui vient consulter, à qui l'on dit de donner un agneau valant trois dinars ; là, il s'agit de celui qui agit de lui-même, c'est-à-dire qui rachète son âne sans consultation, auquel cas il s'acquitte de son obligation même si l'agneau ne vaut qu'une maa.

§ Rabbi Yitzhak dit au nom de Reich Lakich : celui qui a un âne premier-né mais n'a pas d'agneau pour le racheter, il le rachète avec sa valeur en argent, qu'il donne au cohen. La Guemara demande : selon qui est cette opinion ? Si l'on dit que c'est selon Rabbi Yehouda — mais il dit que la Torah exige spécifiquement un agneau (voir 9a) ! C'est donc selon Rabbi Chimon.

Rav Aha l'enseigne ainsi, que l'opinion de Reich Lakich suit celle de Rabbi Chimon. Ravina objecta : Rabbi Yehouda et Rabbi Chimon — la halakha suit Rabbi Yehouda partout où ils sont en désaccord !

Et de plus, le tanna nous a enseigné une Michna anonyme en accord avec Rabbi Yehouda, et tu dis que la halakha suit Rabbi Chimon ?

C'est plutôt que, même si tu dis que c'est Rabbi Yehouda, l'âne premier-né ne devrait pas être plus sévère que les biens consacrés, et la Torah n'a pas dit « avec un agneau » pour être sévère avec le propriétaire, mais pour être clément avec lui, afin qu'il ne soit pas obligé de le racheter à sa pleine valeur, un agneau valant moins qu'un âne. Rav Nahman, fils de Rav Yossef, racheta un âne premier-né avec des légumes bouillis valant sa pleine valeur.

§ Rav Cheizvi dit au nom de Rav Houna : celui qui rachète l'âne premier-né de son prochain, son rachat est valide, malgré le fait que le propriétaire ne l'ait pas racheté lui-même. Un doute fut soulevé devant les Sages : l'objet du rachat appartient-il à celui qui rachète, ou peut-être appartient-il au propriétaire ?

Selon l'opinion de Rabbi Chimon, ne soulève pas ce doute, car puisqu'il dit qu'il est permis de tirer profit de l'âne avant son rachat, c'est la propriété du propriétaire, et donner un agneau au cohen ne change pas ce statut. Quand tu soulèves le doute, c'est selon l'opinion de Rabbi Yehouda, qui dit qu'il est interdit de tirer profit de l'âne avant son rachat.

Le compare-t-on aux biens consacrés, puisque le Miséricordieux dit : « Et il donnera l'argent et cela lui sera assuré » (voir Vayikra 27:19), ce qui indique que celui qui donne l'argent du rachat acquiert l'objet même s'il ne l'a pas consacré initialement ? Ou peut-être, puisqu'il l'acquiert par la différence entre les deux valeurs, cela ne ressemble pas aux biens consacrés, où celui qui consacre ne conserve aucune propriété de l'objet ?

Rav Nahman dit : viens et entends une résolution tirée d'une baraïta : celui qui vole l'âne premier-né de son prochain avant son rachat et se fait attraper, paie le double du principal au propriétaire (voir Chemot 22:3). Et même s'il n'a pas de droit sur l'âne maintenant, il en a après le rachat.

La Guemara analyse la baraïta : selon qui est-ce ? Si l'on dit que c'est selon Rabbi Chimon, pourquoi n'a-t-il pas de droit sur l'âne maintenant ? Rabbi Chimon soutient qu'il est permis de tirer profit de l'âne même avant son rachat. C'est donc évidemment selon Rabbi Yehouda.

Et si tu penses qu'on le compare aux biens consacrés, le Miséricordieux ne dit-il pas : « Et cela fut volé de la maison de l'homme » (Chemot 22:6), indiquant : mais non de la maison du Temple ? Dans un tel cas, le voleur ne paie pas le double. Et il n'y a rien de plus à dire, ceci étant une preuve claire.

§ La Michna mentionne le cas où l'une des ânesses avait déjà mis bas et l'autre non. Les Sages enseignèrent dans une baraïta : comment ont-ils dit qu'il entre dans l'enclos pour être dîmé ?

Tu ne peux pas dire qu'il s'agit d'un cas où l'agneau est arrivé entre les mains du cohen, car nous avons appris : celui qui est acheté ou donné en cadeau est exempté de la dîme animale. C'est plutôt le cas d'un Israélite qui avait dix ânes premiers-nés douteux dans sa maison, qui met de côté pour eux dix agneaux, les dîme, et ils sont à lui.

Cela soutient l'opinion de Rav Nahman, qui dit au nom de Rabba bar Avouh : un Israélite qui avait dix ânes premiers-nés douteux met de côté pour eux dix agneaux, les dîme, et ils sont à lui, car un cohen ne peut prouver qu'il a droit à l'un quelconque des dix.

Et Rav Nahman dit au nom de Rabba bar Avouh : un Israélite qui avait dix ânes premiers-nés certains dans sa maison, qui lui échurent de la maison de son grand-père maternel cohen, et ce grand-père maternel cohen les avait lui-même hérités de la maison de son propre grand-père maternel israélite — les ânes nécessitaient initialement un rachat, et le cohen aurait dû les racheter avec des agneaux, bien qu'il lui fût permis de garder ces agneaux pour lui-même.

Il met de côté pour eux dix agneaux, les dîme, et ils sont à lui, comme si son grand-père avait lui-même désigné les agneaux et les lui avait légués.

Et Rav Nahman dit au nom de Rabba bar Avouh : un Israélite qui avait des produits non dîmés déjà lissés dans sa maison, qui lui échurent de la maison de son grand-père maternel cohen, et ce grand-père maternel cohen les avait lui-même hérités de la maison de son propre grand-père maternel israélite — il les dîme et ils sont à lui, comme si son grand-père cohen avait lui-même dîmé les produits, acquis la dîme, et la lui avait léguée en héritage.

Et cela était nécessaire, les deux décisions de Rav Nahman — celle sur les ânes premiers-nés et celle sur les produits non dîmés. Car si Rav Nahman ne nous avait enseigné que le premier cas, celui des ânes premiers-nés, on aurait pu penser que c'est parce qu'ils avaient déjà été mis de côté par le cohen.

Mais ici, concernant les produits non dîmés, les dons qui n'ont pas été prélevés sont comme s'ils n'avaient pas été prélevés, même s'ils étaient déjà en possession du cohen. Dis donc que non, ce n'est pas considéré comme si le grand-père cohen avait acquis la teroumah.

Et si Rav Nahman ne nous avait enseigné que ce cas-ci, celui des produits non dîmés, on aurait pu penser que c'est parce qu'il est possible de le dîmer de lui-même, puisque les produits non dîmés et la teroumah qui en sera prélevée se trouvent ensemble. Mais là-bas, concernant l'âne premier-né, puisqu'il faut apporter un agneau de l'extérieur pour le racheter, et le mettre de côté, dis que non, le principe ne s'applique pas. C'était donc nécessaire d'énoncer les deux cas.

déjà lissés d'un non-Juif, les dîme, et ils sont à lui.

La Guemara demande : qui les a lissés ? Si l'on dit que c'est le non-Juif qui les a lissés — le Miséricordieux ne dit-il pas « ton grain » (Devarim 12:17, 18:4), indiquant que seul le grain dont le traitement est achevé par un Juif est soumis aux lois de teroumah et de dîme, mais non le grain d'un non-Juif !

C'est plutôt qu'un Juif les a lissés alors qu'ils étaient encore en la possession du non-Juif avant leur achat. Il les dîme, car un non-Juif n'a pas de pouvoir d'acquisition de la terre d'Israël qui abrogerait l'obligation de dîmer ses produits, et ils sont à lui, car il peut dire au cohen : « je les possède en vertu du droit d'un homme avec qui tu ne peux engager de litige », c'est-à-dire un non-Juif, auprès de qui je les ai achetés ; puisqu'il n'est pas obligé de donner teroumah ou dîme au cohen, je ne suis pas non plus tenu de les donner.

§ La Guemara cite une discussion supplémentaire liée à ce sujet : nous avons appris là-bas dans une Michna (Demaï 3:4) : celui qui dépose ses produits chez un Samaritain ou chez un homme du peuple peu fiable en matière de dîmes, quand ceux-ci les lui rendent, ils conservent leur présomption de statut concernant la dîme et l'année sabbatique, qu'il faut retirer de sa propriété. Le Samaritain ou l'homme du peuple n'est pas suspecté d'avoir échangé les produits déposés contre des produits non dîmés ou de l'année sabbatique.

Mais chez un non-Juif, ils sont traités comme ses propres produits, présumés échangés avec les siens. Rabbi Chimon dit : ils sont considérés comme doublement douteux [demaï], car il est incertain si le non-Juif les a échangés ou non.

Rabbi Elazar dit : concernant l'obligation de prélever teroumah et dîmes, tous s'accordent à dire qu'elle s'applique aux produits que le non-Juif a rendus. Quand ils sont en désaccord, c'est concernant la remise de ces prélèvements au cohen. Le premier tanna soutient que le non-Juif les a certainement échangés, et qu'il faut donc les donner au cohen. Et Rabbi Chimon soutient que les produits sont traités comme demaï, et le cohen doit donc apporter une preuve pour les recevoir.

Rav Dimi était assis et énonçait cette halakha de Rabbi Elazar. Abaye lui dit : la raison pour laquelle Rabbi Chimon est en désaccord est que nous sommes incertains si le non-Juif les a échangés ou s'il ne les a pas échangés. Mais s'il les a certainement échangés, tous s'accordent à dire qu'il faut les donner au cohen. Pourquoi ? Rabbi Chmouel ne dit-il pas au nom de Rabbi Hanina : celui qui achète des produits non dîmés d'un non-Juif, déjà lissés, c'est-à-dire complètement traités, les dîme, mais ils sont à lui, et il n'est pas tenu de donner la teroumah au cohen ?

Rav Dimi lui dit : peut-être qu'ici, dans la Michna qui indique qu'il faut donner la teroumah au cohen, il s'agit de la grande teroumah, celle prélevée avant les dîmes, tandis que là, l'affirmation de Rabbi Hanina selon laquelle il dîme mais les garde pour lui, se réfère à la teroumah prélevée sur la dîme.

Abaye lui dit : tu m'as rappelé une chose que Rabbi Yehochoua ben Lévi a dite : d'où déduit-on que celui qui achète des produits non dîmés déjà lissés d'un non-Juif est exempté de prélever la teroumah de la dîme ? Cela se déduit d'un verset, ainsi qu'il est dit : « Et tu parleras aux lévites, et tu leur diras : quand vous prendrez la dîme des enfants d'Israël... vous en prélèverez une offrande pour l'Éternel, une dîme de la dîme » (Bamidbar 18:26). On en déduit que concernant les produits non dîmés que tu achètes des enfants d'Israël, tu en prélèves la teroumah de la dîme et la donnes au cohen, mais concernant les produits non dîmés que tu achètes d'un non-Juif, tu n'en prélèves pas la teroumah de la dîme pour la donner au cohen.

Si l'on dit qu'il est mort dans la maison du cohen, et que la Michna signifie que le cohen peut en tirer profit, n'est-ce pas évident ? L'agneau est sa propriété. C'est plutôt peut-être qu'il est mort dans la maison du propriétaire avant d'être donné au cohen, et la Michna enseigne que le cohen peut en tirer profit. Mais cela aussi n'est-il pas évident ?

La Guemara répond : il aurait pu te venir à l'esprit de dire que, tant que l'agneau n'est pas arrivé en la possession du cohen, celui-ci ne l'a pas acquis. Elle nous enseigne donc que, depuis le moment où le propriétaire l'a mis de côté, il se trouve déjà en la possession du cohen.

The Gemara answers: Lest you say that if the lamb is of such little value one may not redeem with it, Rav Yosef states that one may redeem with it. Rabbi Yehuda Nesia had a firstborn donkey. He sent it before Rabbi Tarfon, and said to him: How much do I need to give a priest for its redemption, i.e., how much must the lamb used for its redemption be worth? Rabbi Tarfon said to him: Didn’t the Sages say: One who has a generous disposition redeems his firstborn donkey with a lamb worth a sela, one who has a miserly disposition redeems his donkey with a lamb worth a shekel, and one who has an intermediate disposition redeems his donkey with a lamb worth a ragya?

Rava said: The halakha is that one must redeem the firstborn donkey with a lamb that is worth at least a ragya. The Gemara asks: And how much is the value of a ragya? The Gemara answers: A ragya is worth three dinars, and is called a ragya because it is close to [ragil] here and close to there, i.e., its value is between that of a sela, which is worth fours dinars, and a shekel, which is worth two dinars.

The Gemara comments: The contradiction between this halakha, that one must give a lamb worth three dinars, and the halakha stated by Rav Naḥman, that one may give a lamb of any monetary value, poses a difficulty. The Gemara responds: That is not difficult, as here the reference is to one who comes to consult; he is told to give a lamb worth three dinars. There, the reference is to one who acts on his own, i.e., one who redeems his donkey without consultation, in which case he has fulfilled his obligation even if the lamb is worth only a ma’a.

§ Rabbi Yitzḥak says that Reish Lakish says: One who has a firstborn donkey but does not have a lamb with which to redeem it may redeem the donkey with a sum of money equal to the donkey’s value and give it to a priest. The Gemara asks: In accordance with whose opinion is this statement? If we say that it is in accordance with the opinion of Rabbi Yehuda, doesn’t he say that the Torah is particular that the donkey be redeemed with a lamb (see 9a)? Rather, clearly it is in accordance with the opinion of Rabbi Shimon.

The Gemara comments: Rav Aḥa teaches it in this way, that the statement of Reish Lakish was made in accordance with the opinion of Rabbi Shimon. Ravina found this difficult, as wherever there is a dispute between Rabbi Yehuda and Rabbi Shimon, the halakha is in accordance with the opinion of Rabbi Yehuda.

And furthermore, the tanna taught us an unattributed mishna in accordance with the opinion of Rabbi Yehuda. The mishna on 9a states that in a case where the firstborn status of a donkey is uncertain, the owner must redeem it with a lamb to abrogate its forbidden status, and he may then keep the lamb. The Gemara on 9b explains that this is in accordance with the opinion of Rabbi Yehuda that a firstborn donkey is forbidden before its redemption. And you say that Reish Lakish rules that the halakha is in accordance with the opinion of Rabbi Shimon?

Ravina concludes: Rather, the statement of Reish Lakish is correct even if you say that the halakha is in accordance with the opinion of Rabbi Yehuda, as the halakha of a firstborn donkey should not be more stringent than that of consecrated items. One who redeems a consecrated animal from its consecrated status redeems it with any movable property that is worth its value. And the Torah did not state that one must redeem a firstborn donkey with a lamb in order to be stringent with the owner, but in order to be lenient with the owner such that he is not obligated to redeem it at its value, as a lamb is worth less than a donkey. The Gemara relates that Rav Naḥman, son of Rav Yosef, redeemed a firstborn donkey with boiled vegetables that were worth its value.

§ Rav Sheizevi says that Rav Huna says: In the case of one who redeems the firstborn donkey of another, his object of redemption, i.e., the firstborn donkey, is redeemed, despite the fact that the owner did not redeem it. A dilemma was raised before the Sages: Does his object of redemption belong to the one who redeems it, or perhaps his object of redemption belongs to the owner?

The Gemara clarifies: According to the opinion of Rabbi Shimon you should not raise the dilemma, since he says that prior to the redemption it is permitted for the owner to derive benefit from the firstborn donkey by using it for labor or renting it out. Therefore, it is the property of the owner, and giving a lamb to the priest does not change this status. When you raise the dilemma, it should be according to the opinion of Rabbi Yehuda, who says that it is prohibited to derive benefit from the firstborn donkey prior to its redemption. If so, it does not belong to the owner initially, and only redeeming the donkey with the lamb allows him to take possession of it.

Consequently, one can ask whether the firstborn donkey is compared to consecrated property; and the Merciful One says with regard to redeeming consecrated property: “And he will give the money and it will be assured to him” (see Leviticus 27:19), which indicates that the one who gives the redemption money acquires the item even if he did not consecrate it initially. Or perhaps since the owner may redeem the firstborn donkey with a lamb that is worth less than the donkey, he owns the difference between the price of the donkey and the price of the lamb, and therefore it is not similar to consecrated property, where the one who consecrated it does not retain any ownership of the item.

Rav Naḥman said: Come and hear a resolution from a baraita: In the case of one who steals the firstborn donkey of another before it is redeemed and he is then caught, he pays the payment of double the principal to the owner (see Exodus 22:3). And even though the owner does not have rights to the donkey now, he does have rights to it afterward, i.e., after the redemption.

The Gemara analyzes the baraita: Whose opinion does this follow? If we say that it is in accordance with the opinion of Rabbi Shimon, why does the owner not have rights to it now? Rabbi Shimon holds that it is permitted to derive benefit from a firstborn donkey even before its redemption. Rather, it is obvious that it is in accordance with the opinion of Rabbi Yehuda, who deems it prohibited to derive benefit from a firstborn donkey prior to its redemption.

And if it enters your mind that we compare it to consecrated property, then why must the thief pay double the principal? Doesn’t the Merciful One state with regard to the double payment of a thief: “And it be stolen from the man’s house” (Exodus 22:6), indicating: But not when it is stolen from the Temple treasury? In such a case the thief does not pay double. Evidently, a firstborn donkey does not have the status of consecrated property, and if another person redeemed it, it still belongs to the owner. The Gemara concludes: And nothing more remains to be said about this issue, as this is a clear proof.

§ The mishna states that if a person owned two donkeys, and one had previously given birth and one had not previously given birth, and they gave birth to a male and a female, and it is not clear which donkey gave birth to the male and which to the female, he designates one lamb as redemption due to the uncertainty, as perhaps the one that had not previously given birth was the one that gave birth to the male. The mishna also states that this lamb is tithed with the other non-sacred animals. With regard to this point, the Gemara states that the Sages taught in a baraita: In what case did the Sages say that this lamb enters the pen in order to be tithed?

You cannot say that this is referring to a case where the lamb had already entered the possession of a priest, as we learned in a mishna (55b): An animal that was purchased by a person or that was given to him as a present is exempt from animal tithe, and this lamb was given as a present to the priest. Rather, it is referring to an Israelite who has ten donkeys in his house whose firstborn status is uncertain, who designates ten lambs for them in order to redeem them due to the uncertainty, and he tithes those lambs as well, and they are his.

The Gemara comments that this baraita supports the statement of Rav Naḥman, as Rav Naḥman says that Rabba bar Avuh says: An Israelite who has in his house ten donkeys whose firstborn status is uncertain designates ten lambs to redeem them, and tithes the lambs, separating one as a tithe, and they all belong to him, as a priest cannot prove that he is entitled to any of the ten.

And in addition, Rav Naḥman says that Rabba bar Avuh says: In the case of an Israelite who has ten certain firstborn donkeys in his house that were bequeathed to him from the household of his mother’s father, who was a priest and was therefore exempt from the redemption of firstborn donkeys, but that mother’s father who was a priest had the donkeys bequeathed to him from the household of his mother’s father, who was an Israelite, the donkeys initially required redemption, and the priest should have redeemed them with lambs, though it was permitted for him to keep the lambs for himself.

Therefore, his grandson, who inherits these donkeys and is an Israelite, designates ten lambs for them and tithes them and they are his, as if his grandfather had designated the lambs himself and bequeathed them to him.

And likewise, Rav Naḥman says that Rabba bar Avuh says: In the case of an Israelite who has smoothed piles of untithed produce in his house, from which one is obligated to separate teruma for the priests and tithes for the Levites, and it is produce that was bequeathed to him from the household of his mother’s father, who was a priest, and that mother’s father who was a priest previously had it bequeathed to him from the household of his mother’s father, who was an Israelite, he separates teruma and tithes from the produce and it is his, just as if his grandfather who was a priest had tithed the produce, acquired the tithe, and bequeathed it to him as an inheritance.

The Gemara comments that both rulings of Rav Naḥman, i.e., the one concerning firstborn donkeys and the one concerning untithed produce, are necessary. Because had he taught us only the first case, i.e., that of the firstborn donkeys, one could claim that perhaps only in that case the Israelite grandson may keep the lambs; it is considered as if his grandfather who was a priest designated the lambs and acquired them because they stand separated. An act of separation would have been unnecessary, as the priest could have simply declared that his own lambs were serving to redeem the donkeys.

But here, with regard to the untithed produce, one must separate the teruma and declare it as such in order for the priest to acquire it. Therefore, perhaps gifts to which members of the priesthood are entitled that were not separated are treated as if they were not separated, even if they were already in the possession of a priest. If so, say that it is not considered as if his grandfather who was a priest acquired the teruma.

And had Rav Naḥman taught us only here, with regard to untithed produce, that it is treated as if his grandfather who was a priest separated the teruma and acquired it, one might claim that this is only because it is possible to tithe the produce from itself, as both the untithed produce and the teruma that will be separated from it lie together. But there, with regard to a firstborn donkey, since one needs to bring a lamb from elsewhere to redeem it, say that the principle that the lamb stands separated and does not require additional separation is not applied. Therefore, it is necessary for both cases to be stated.

§ The Gemara discusses a variation of the previous case. Rabbi Shmuel bar Natan says that Rabbi Ḥanina says: With regard to one who purchases untithed produce

that is gathered in smoothed piles, from a gentile, he tithes the piles but they are his, as he is not required to give the teruma to a priest or the tithes to a Levite.

The Gemara asks: Who smoothed the piles? If we say that a gentile smoothed them, doesn’t the Merciful One state: “Your grain” (Deuteronomy 12:17, 18:4), with regard to teruma and tithes, indicating that only grain whose processing is completed by a Jew is subject to the rules of teruma and tithes, but not the grain of a gentile?

Rather, clearly a Jew smoothed them while they were in the possession of the gentile before purchasing them. Therefore, he tithes them, as a gentile has no capability of acquisition of land in Eretz Yisrael that would cause the abrogation of the sanctity of the land, thereby removing it from the obligation to tithe its produce. But the tithes are still the Jew’s, as he says to the priest: I have come to own this produce on the basis of the rights of a man with whom you cannot engage in litigation, i.e., a gentile, from whom I bought the produce. Since he is not obligated to give teruma or tithes to the priest, I am not required to give them either.

§ The Gemara cites an additional discussion involving this issue: We learned in a mishna elsewhere (Demai 3:4) that with regard to one who deposits his produce with a Samaritan or with one who is unreliable with regard to tithes [am ha’aretz], when they return it to him, the produce retains its presumptive status with regard to the halakhot of tithe and produce of the Sabbatical Year, both of which must be removed from one’s property. The Samaritan or am ha’aretz is not suspected of having exchanged the deposited produce with untithed produce or produce of the Sabbatical Year.

But in the case of one who deposits his produce with a gentile, the produce returned to him is treated as the produce of the gentile, as he presumably exchanged it with his own. Rabbi Shimon says: It is treated as doubtfully tithed produce [demai], as it is uncertain whether the gentile exchanged the produce or not.

The mishna continues: Rabbi Elazar says: Everyone agrees that one is obligated to separate teruma and tithes from the produce that the gentile returned to him. When they disagree it is with regard to giving it to the priest following the separation. The first tanna holds that the gentile certainly exchanged the produce, and therefore the owner needs to give the teruma to the priest. And Rabbi Shimon holds that the produce is treated as demai, so in order to receive teruma, the priest must bring proof that the produce requires tithing.

Rav Dimi sat and stated this halakha of Rabbi Elazar. Abaye said to him: The reason that Rabbi Shimon disagrees is that we are uncertain whether the gentile exchanged it or whether he did not exchange it. But if he definitely exchanged it, everyone agrees that he is required to give the teruma and tithes to a priest. Why? But doesn’t Rabbi Shmuel say that Rabbi Ḥanina says: In the case of one who purchases untithed produce from a gentile that is gathered in smoothed piles, i.e., it was completely processed, he tithes and separates teruma from the piles of produce but they are his, and he is not required to give the teruma to a priest?

Rav Dimi said to him: Perhaps here, in the mishna that indicates that he must give the teruma to a priest, it is referring to teruma gedola, i.e., the teruma that is separated from the produce before tithes, whereas there, Rabbi Ḥanina’s statement that he separates tithes from them but they are his is referring to the teruma that is separated from the tithe.

Abaye said to him: You reminded me of a statement that Rabbi Yehoshua ben Levi said: From where is it derived that one who purchases from a gentile untithed produce that is gathered in smoothed piles is exempt from separating the teruma of the tithe? It is derived from a verse, as it is stated: “And you shall speak to the Levites, and say to them: When you take the tithe from the children of Israel…then you shall set apart of it a gift for the Lord, even a tithe of the tithe” (Numbers 18:26). It can be inferred from here that concerning untithed produce that you purchase from the children of Israel, you separate teruma of the tithe from it and give the teruma of the tithe to a priest, but concerning untithed produce that you purchase from a gentile, you do not separate teruma of the tithe from it and give it to a priest.

§ The mishna teaches with regard to the lamb that is used to redeem the firstborn donkey: And if it dies, one may derive benefit from it. The Gemara asks: Where did it die? If we say that it died in the house of the priest, and the mishna means that the priest may derive benefit from it, isn’t it obvious? The lamb is his property. Rather, perhaps the mishna means that it died in the house of the owner before it was given to the priest, and teaches that the priest may derive benefit from it. Isn’t this also obvious?

The Gemara responds: It might enter your mind to say that as long as the lamb has not reached the possession of the priest, the priest has not acquired it. Therefore, the mishna teaches us that from the time that the Israelite separated it, it stands in the possession of the priest.

Contexte : ce daf poursuit la discussion du daf 10 sur le montant minimal de rachat de l'âne premier-né, avant d'aborder trois nouveaux sujets successifs : le rachat par un tiers (et à qui appartient l'agneau ainsi donné), le cas pratique de dix ânes premiers-nés douteux hérités dans une même maison, et enfin une longue digression sur les produits agricoles non dîmés achetés à un non-Juif — avant de revenir, en toute fin de daf, à la question du statut de l'agneau du rachat s'il meurt avant d'atteindre les mains du cohen.

1. Combien faut-il donner concrètement ? Trois réponses selon le contexte. Le daf ouvre sur la conclusion du débat entamé au daf 10 : Rav Nahman tranche que la halakha suit l'avis permissif des Sages — un agneau de n'importe quelle valeur, même infime, suffit pour le rachat. Rav Yossef précise que même un agneau "maigre" valant à peine une maa (une très petite pièce) est valable, et Rava trouve un appui à cela dans la Michna elle-même, qui autorise indifféremment un agneau grand ou petit. Mais la Guemara nuance immédiatement cette permissivité par un cas concret : quand Rabbi Yehouda Nessia demande conseil à Rabbi Tarfon sur le montant à donner, ce dernier lui rapporte une échelle de générosité à trois niveaux — un séla (4 dinars) pour l'"œil généreux", un shekel (2 dinars) pour l'"œil mesquin", et un ragya (3 dinars, la valeur intermédiaire) pour la personne "du milieu". Rava fixe alors la pratique concrète sur ce dernier montant, le ragya. Cette apparente contradiction entre "n'importe quel montant suffit" et "il faut donner un ragya" est résolue par une distinction fine : la loi stricte n'exige effectivement rien de plus qu'un montant symbolique, mais celui qui vient demander conseil reçoit, par courtoisie ou par volonté d'encourager une pratique digne, la recommandation du montant intermédiaire — tandis que celui qui agit "de son propre chef", sans consulter personne, s'acquitte pleinement de son obligation même avec un agneau de valeur minimale.

2. Peut-on racheter en argent plutôt qu'avec un agneau, à défaut d'en avoir un ? Reich Lakich introduit un cas pratique : que faire si l'on possède un âne premier-né mais aucun agneau disponible pour le racheter ? Sa réponse — payer la valeur en argent — semble à première vue devoir suivre l'opinion permissive de Rabbi Chimon (puisque Rabbi Yehouda exige, lui, spécifiquement un agneau et rien d'autre). Ravina soulève une objection méthodologique importante : dans tout désaccord entre Rabbi Yehouda et Rabbi Chimon, la halakha suit toujours Rabbi Yehouda — et la Michna elle-même semble déjà avoir tranché en ce sens ailleurs dans le traité. Comment alors Reich Lakich pourrait-il s'appuyer sur l'opinion minoritaire ? La résolution de Ravina est élégante : même selon Rabbi Yehouda, on peut accepter un rachat en argent, car l'exigence d'un agneau n'a jamais eu pour but de compliquer la vie du propriétaire (comme c'est le cas pour les biens consacrés au Temple, dont le rachat suit des règles strictes), mais au contraire de la lui faciliter — puisqu'un agneau vaut généralement moins qu'un âne. Priver le propriétaire de cette option en cas d'absence d'agneau irait donc à l'encontre de l'esprit même de la loi.

3. Qui possède l'agneau quand un tiers rachète l'âne d'autrui ? Un nouveau cas se présente : si quelqu'un rachète, de sa propre initiative, l'âne premier-né appartenant à une autre personne, l'objet racheté appartient-il à celui qui a payé, ou reste-t-il la propriété du propriétaire original de l'âne ? La Guemara précise d'emblée que cette question ne se pose vraiment que selon Rabbi Yehouda (qui interdit tout profit avant rachat, donc l'âne "n'appartenait à personne" au sens plein tant qu'il n'était pas racheté) — selon Rabbi Chimon, la question ne se pose même pas, puisque l'âne appartenait déjà pleinement au propriétaire avant tout rachat. Deux analogies opposées sont envisagées : soit on compare ce cas à un bien consacré au Temple, où celui qui paie le rachat en devient automatiquement propriétaire (par le principe scripturaire "il donnera l'argent et cela lui sera assuré", Vayikra 27:19) ; soit on objecte que, puisque le propriétaire garde toujours un intérêt financier dans l'opération (la différence entre la valeur de l'âne et celle de l'agneau, moins chère), ce cas ne ressemble finalement pas à un bien consacré au Temple, où le donateur d'origine ne conserve aucun droit résiduel.

La résolution de ce doute vient d'une baraïta sur le vol : si quelqu'un vole l'âne premier-né d'un tiers avant son rachat, il doit payer le double au propriétaire d'origine — preuve indirecte mais solide que le propriétaire conserve bien des droits sur l'animal, même avant rachat, puisque la loi du double paiement pour vol (Chemot 22:6) exclut explicitement le vol de biens appartenant au Temple ("volé de la maison de l'homme", pas "de la maison consacrée"). L'âne premier-né n'est donc pas assimilable à un bien consacré, et cette preuve est jugée suffisamment solide pour clore définitivement le débat.

4. Le cas concret des dix ânes hérités — et pourquoi il fallait le préciser deux fois. La Guemara développe ensuite, via Rav Nahman au nom de Rabba bar Avouh, une série de cas pratiques liés à l'héritage. Premier cas : un Israélite possède dix ânes dont le statut de premier-né est douteux (issus de naissances multiples incertaines, comme vu au daf 9). Il met de côté dix agneaux correspondants, les dîme comme n'importe quel autre bétail lui appartenant, et les garde tous pour lui — car aucun cohen individuel ne peut prouver un droit spécifique sur l'un ou l'autre de ces dix agneaux précis. Deuxième cas, plus subtil : des ânes premiers-nés certains (non douteux) hérités à travers trois générations — d'un arrière-grand-père israélite, à son gendre cohen, puis au petit-fils israélite actuel. Puisque le grand-père cohen aurait eu le droit de garder les agneaux du rachat pour lui-même (étant cohen), son héritier israélite hérite de ce même droit : il peut mettre de côté les agneaux, les dîmer, et les garder — comme si son grand-père les avait lui-même déjà mis de côté avant de les lui léguer. Un troisième cas, structurellement identique mais appliqué à des produits agricoles non dîmés plutôt qu'à des animaux, confirme le même principe.

Pourquoi la Guemara a-t-elle besoin d'énoncer ces trois cas séparément, alors qu'ils semblent répéter le même principe ? Parce que chacun aurait pu, pris isolément, laisser penser que le principe ne s'applique que dans ce cas précis et non dans l'autre :

  • Si l'on n'avait connu que le cas des ânes premiers-nés, on aurait pu penser que le principe ne fonctionne que parce que les agneaux étaient déjà "mis de côté" et physiquement distincts — une opération qui n'aurait pas été nécessaire pour de simples produits agricoles.
  • Si l'on n'avait connu que le cas des produits agricoles, on aurait pu penser que le principe ne fonctionne que parce que la teroumah peut être prélevée "de la chose elle-même" (le tas de grain et sa teroumah sont ensemble) — un raisonnement qui ne s'appliquerait pas à l'âne, où l'agneau du rachat doit être amené "de l'extérieur".

Les deux cas se complètent donc mutuellement, chacun couvrant une objection que l'autre ne pourrait à lui seul écarter.

5. Une longue digression sur les produits achetés à un non-Juif. Le daf s'engage alors dans un développement plus long sur un sujet parallèle : celui qui achète des produits agricoles déjà "lissés" (complètement traités, prêts à consommer) auprès d'un non-Juif doit-il en prélever la teroumah et la dîme, et si oui, doit-il les remettre au cohen et au lévite, ou peut-il les garder ? La réponse de Rabbi Hanina est que l'acheteur doit bien dîmer ces produits (car la sainteté de la Terre d'Israël elle-même impose cette obligation, indépendamment du vendeur), mais il peut garder la teroumah et la dîme pour lui-même — puisqu'un non-Juif, n'étant lui-même soumis à aucune obligation de ce type, ne peut transmettre à l'acheteur une dette qu'il n'avait pas lui-même. Cette règle se justifie même par un verset (Bamidbar 18:26) qui, par contraste implicite, limite l'obligation de remise au cohen aux seuls produits achetés à d'autres Israélites.

Une Michna parallèle sur le dépôt de produits (chez un Samaritain, chez un Israélite peu fiable, ou chez un non-Juif) introduit une nuance supplémentaire : Rabbi Elazar clarifie que le désaccord entre le tanna anonyme et Rabbi Chimon dans cette Michna ne porte pas sur l'obligation de prélever la teroumah (tout le monde s'accorde là-dessus), mais uniquement sur l'obligation de la remettre effectivement au cohen. Abaye soulève alors une tension apparente avec la règle de Rabbi Hanina énoncée plus haut, résolue finalement par une distinction entre deux types de teroumah — la "grande teroumah" (prélevée avant toute dîme) et la "teroumah de la dîme" (prélevée sur la dîme elle-même) — chacune suivant des règles de remise légèrement différentes.

6. Retour au sujet principal : le statut de l'agneau mort avant remise. Le daf se referme en revenant au texte même de la Michna, qui précise qu'on peut tirer profit de l'agneau du rachat s'il meurt. La Guemara s'interroge sur la portée exacte de cette précision, qui semble à première vue superflue dans deux scénarios évidents (l'agneau meurt une fois déjà chez le cohen — évidemment sa propriété ; ou meurt chez le propriétaire avant d'être physiquement remis — là encore, apparemment évident). La réponse finale révèle la véritable portée de cet enseignement : sans cette précision, on aurait pu penser que la propriété de l'agneau ne se transfère au cohen qu'au moment où il le reçoit physiquement entre ses mains. La Michna vient donc enseigner un principe juridique important : dès l'instant où le propriétaire met de côté l'agneau destiné au rachat, celui-ci appartient déjà, légalement, au cohen — même s'il ne lui a pas encore été matériellement livré.

Rachi précise, sur l'échelle des trois montants de rachat (séla/shekel/ragya), que cette hiérarchie ne reflète pas une exigence légale stricte mais une gradation morale : la loi elle-même se contente d'un montant symbolique, mais les Sages ont voulu offrir un cadre de référence pour ceux qui souhaitent accomplir la mitsva avec une générosité particulière, sans pour autant imposer cette générosité comme une obligation.

Sur la distinction entre l'âne premier-né et les biens consacrés au Temple, Rachi souligne que cette comparaison — acceptée puis nuancée par la Guemara — traverse tout ce passage : chaque fois qu'une halakha de l'âne premier-né semble ambiguë, la Guemara teste d'abord l'hypothèse d'une analogie avec la sainteté du Temple, avant de l'écarter ou de la confirmer selon les preuves disponibles, la preuve du vol (payant le double, contrairement au vol de biens du Temple) étant ici jugée décisive.

Tossefot s'attarde sur le principe des "deux preuves nécessaires" (le cas de l'âne et celui des produits agricoles), notant qu'il s'agit d'une méthode d'argumentation récurrente dans le Talmud : lorsque deux cas semblent illustrer le même principe général, la Guemara vérifie systématiquement qu'aucun des deux ne pourrait, à lui seul, s'expliquer par une raison plus étroite et spécifique à ce cas précis — ce n'est que lorsque cette explication alternative est exclue pour les deux cas que le principe général peut être considéré comme véritablement établi.

D'après Rashi, Tosafot
⚖️ Loi pratique

La halakha retient qu'un âne premier-né peut légalement être racheté avec un agneau de valeur minime, mais la pratique recommandée — pour celui qui souhaite accomplir la mitsva dans de bonnes conditions — reste de donner un agneau d'une valeur intermédiaire (l'équivalent moderne du ragya), ni excessive ni symbolique.

Le principe qu'un objet mis de côté pour un rachat appartient immédiatement à son destinataire légal, même avant remise physique, reste un principe juridique fondamental appliqué bien au-delà du seul cas de l'âne premier-né — il informe de nombreuses questions de droit des biens et de transferts de propriété dans la halakha contemporaine.
